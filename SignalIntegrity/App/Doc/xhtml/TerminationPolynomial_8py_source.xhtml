26class TerminationPolynomial(object):
27 """Base class for terminations for calibration standards"""
28 def __init__(self,a0=0.0,a1=0.0,a2=0.0,a3=0.0):
29 """Constructor
30 @param a0 (optional) float polynomial term for f^0 (defaults to 0.0).
31 @param a1 (optional) float polynomial term for f^1 (defaults to 0.0).
32 @param a2 (optional) float polynomial term for f^2 (defaults to 0.0).
33 @param a3 (optional) float polynomial term for f^3 (defaults to 0.0).
34 """
35 self.a0=a0
36 self.a1=a1
37 self.a2=a2
38 self.a3=a3
39 def Polynomial(self,f):
40 """Evaluates polynomial.
41 @param f float frequency
42 @return float polynomial evaluated at frequency f
43 """
44 return ((self.a3*f+self.a2)*f+self.a1)*f+self.a0
