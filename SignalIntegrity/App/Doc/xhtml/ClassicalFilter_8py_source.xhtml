25class ClassicalFilter(SParameters):
26 """Classical filter"""
27 def __init__(self,f,z,p,k,f0=1.,Z0=50.):
28 self.z=z
29 self.p=p
30 self.k=k
31 self.f0=f0
32 SParameters.__init__(self,f,None,Z0)
33 def _Evaluate(self,f):
34 """Evaluate filter
35 @param f float frequency
36 @return filter evaluated at f
37 """
38 s=1j*f/self.f0
39 D=np.prod([s-self.p[k] for k in range(len(self.p))]); N=np.prod([s-self.z[k] for k in range(len(self.z))])
40 return self.k*N/D
41 def __getitem__(self,n):
42 return [[1.,0],[2.*self._Evaluate(self.m_f[n]),-1.]]

44class BesselLowPassFilter(ClassicalFilter):
45 """Bessel low pass filter"""
46 def __init__(self,f,order,f0,Z0=50.):
47 """Constructor
48 @param f list of float frequencies
49 @param order int filter order
50 @param f0 float cutoff frequency (Hz)
51 @param Z0 float (optional, defaults to 50) reference impedance
52 @note This device requires installation of the scipy package
53 """
54 try:
55 from scipy.signal import besselap
56 except: raise SignalIntegrityExceptionSParameters('Bessel filters require the scipy package to be installed') # pragma: no cover
57 (z,p,k)=besselap(order,'mag')
58 # this filter has -3 dB point at w=1 radian
59 ClassicalFilter.__init__(self,f,z,p,k,f0,Z0)
