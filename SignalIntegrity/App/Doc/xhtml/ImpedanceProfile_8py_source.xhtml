30class ImpedanceProfile(list):
31 """ Impedance profile generated using the peeling algorithm. After construction, it is
32 a list of rho values"""
33 rhoLimit=0.99
34 ZLimit=10e3
35 def __init__(self,sp,sections,port):
36 """ Constructor--
37 Generates a list of rho values for a specified number of sections by peeling the
38 s-parameters provided from the port provided.
39 @param sp instance of class SParameters
40 @param sections integer number of sections to compute
41 @param port port of the SParameters instance to peel from
42 """
43 list.__init__(self,[])
44 N = len(sp)-1
45 self.m_Td = 1./(4.*sp.f()[N])
46 self.m_Z0 = sp.m_Z0
47 fr=sp.FrequencyResponse(port,port)
48 self.m_fracD=fr._FractionalDelayTime()
49 fr=fr._DelayBy(-self.m_fracD)
50 S11 = fr.Values()
51 zn1 = [cmath.exp(-1j*math.pi*n/N) for n in range(N+1)]
52 finished=False
53 rho=0.0
54 for _ in range(sections):
55 if finished:
56 self.append(rho)
57 continue
58 rho = 1/(2.*N)*(S11[0].real + S11[N].real +
59 sum([2.*S11[n].real for n in range(1,N)]))
60 rho=max(-self.rhoLimit,min(rho,self.rhoLimit))
61 self.append(rho)
62 if abs(rho)==self.rhoLimit:
63 finished=True
64 continue
65 rho2=rho*rho
66 S11=[(-S11[n]+S11[n]*rho2*zn1[n]-rho*zn1[n]+rho)/
67 (rho2+S11[n]*rho*zn1[n]-S11[n]*rho-zn1[n])
68 for n in range(N+1)]
69 def Z(self):
70 """impedance list
71 @return list of impedances, in ohms, for each section.
72 """
73 return [max(0.,min(self.m_Z0*(1+rho)/(1-rho),self.ZLimitZLimit))
74 for rho in self]
75 def DelaySection(self):
76 """time length of each section
77 @return float time length of each section.
78 """
79 return self.m_Td
80 def SParameters(self,f):
81 """s-parameters
82 @param f list of frequencies for the s-parameters
83 @return instance of class SParameters calculated by aggregating the list of rho
84 values using lossless transmission line sections with the rho calculated for each
85 section.
86 """
87 N = len(f)-1
88 Gsp=[None for n in range(N+1)]
89 gamma=[1j*2.*math.pi*f[n]*self.m_Td for n in range(N+1)]
90 for n in range(N+1):
91 tacc=array([[1.,0.],[0.,1.]])
92 for m in range(len(self)):
93 tacc=tacc.dot(array(S2T(IdealTransmissionLine(self[m],gamma[n]))))
94 Gsp[n]=T2S(tacc.tolist())
95 sp = SParameters(f,Gsp,self.m_Z0)
96 return sp
