26class RLGCFitFromFile(object):
27 """fits a two-port RLGC model to s-parameters from a file"""
28 def __init__(self,f,filename,scale=1,Z0=50.,**kwargs):
29 """Constructor
30 @param f list of float frequencies
31 @param filename string name of s-parameter file or project that produces s-parameters
32 @param scale float (optional, defaults to 1.0) scaling to be applied on resulting fit
33 @param Z0 float (optional, defaults to 50) reference impedance
34 @param **kwargs dict (optional, defaults to {}) dictionary of arguments for the file
35 @note fitting is not performed until an item from the fitted model is requested
36 """
37 self.scale=scale
38 self.Z0=Z0
39 self.m_f=f
40 self.spfile=filename
41 self.RLGC=None
42 self.kwargs=kwargs
43 if scale == 0.0:
44 self.RLGC=TLineLossless(self.m_f,2,50.0,0.0,self.Z0)
45 def Fit(self):
46 """Fits a two-port RLGC model for the specified s-parameter file
47 @see see RLGC
48 """
49 sp=SParameterFile(self.spfile,self.Z0,None,**self.kwargs)
50 stepResponse=sp.FrequencyResponse(2,1).ImpulseResponse().Integral()
51 threshold=(stepResponse[len(stepResponse)-1]+stepResponse[0])/2.0
52 for k in range(len(stepResponse)):
53 if stepResponse[k]>threshold: break
54 dly=stepResponse.Times()[k]
55 rho=sp.FrequencyResponse(1,1).ImpulseResponse().Integral(scale=False).Measure(dly)
56 Z0=sp.m_Z0*(1.+rho)/(1.-rho)
57 L=dly*Z0; C=dly/Z0; guess=[0.,L,0.,C,0.,0.]
58 fitter=RLGCFitter(sp,guess)
59 (R,L,G,C,Rse,df)=fitter.Solve().Results()
60# print "series resistance: "+ToSI(R,'ohm')
61# print "series inductance: "+ToSI(L,'H')
62# print "shunt conductance: "+ToSI(G,'S')
63# print "shunt capacitance: "+ToSI(C,'F')
64# print "skin-effect resistance: "+ToSI(Rse,'ohm/sqrt(Hz)')
65# print "dissipation factor: "+ToSI(df,'')
66 s=self.scale
67 self.RLGC=TLineTwoPortRLGCAnalytic(self.m_f, R*s, Rse*s, L*s, G*s, C*s, df, Z0=self.Z0)
68 return self.RLGC
69 def __getitem__(self,item):
70 """overloads [n]
71 @return list of list s-parameter matrix for the nth frequency element
72 @note if a fit has not yet been performed, it is performed prior to obtaining the element
73 """
74 if self.RLGC is None:
75 self.RLGC = self.Fit()
76 return self.RLGC[item]
