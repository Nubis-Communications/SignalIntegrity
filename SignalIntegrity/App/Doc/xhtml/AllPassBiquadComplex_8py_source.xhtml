24class Gain(object):
25 def __init__(self,ω,G,fix_gain=True):
26 self.ω=ω
27 self.G=G
28 self.H=G

32class Delay(object):
33 def __init__(self,ω,Td,fix_delay=False):
34 self.ω=ω
35 self.Td=Td
36 self.H=np.exp(-1j*ω*Td)
37 self.pHpTd=10000 if fix_delay else -1j*ω*self.H

39class AllPassBiquadSection(object):
40 def __init__(self,ω,ω0,Q):
41 """
42 biquad section is expressed as H(s) = (s^2+ω0/-Q*s+ω0^2)/(s^2+ω0/Q*s+ω0^2) = (α+jβ)/(α-jβ)
43 """
44 self.ω=ω
45 self.ω0=ω0
46 self.Q=Q
47 ω02=ω0*ω0
48 ω2=ω*ω
49 Q2=Q*Q
50 α=ω02-ω2
51 α2=α*α
52 β=-ω0/Q*ω
53 β2=β*β
54 α2pβ2=α2+β2
55 β2mα2=β2-α2
56 α2pβ22=α2pβ2*α2pβ2
57 self.H=(α+1j*β)/(α-1j*β) # H(ω)
58 pαpω0=2*ω0 # ∂α/∂ωz
59 pαpQ=0 # ∂α/∂Qz
60 pβpω0=-ω/Q # ∂β/∂ωz
61 pβpQ=ω*ω0/Q2 # ∂β/∂Qz
62 RepHpα=4*α*β2/α2pβ22 # Re(∂H/∂α)
63 RepHpβ=-4*α2*β/α2pβ22 # Re(∂H/∂β)
64 ImpHpα=2*β*β2mα2/α2pβ22 # Im(∂H/∂α)
65 ImpHpβ=-2*α*β2mα2/α2pβ22 # Im(∂H/∂β)
66
67 RepHpω0=RepHpα*pαpω0+RepHpβ*pβpω0 # Re(∂H/∂ωz)
68 RepHpQ=RepHpα*pαpQ+RepHpβ*pβpQ # Re(∂H/∂Qz)
69 ImpHpω0=ImpHpα*pαpω0+ImpHpβ*pβpω0 # Im(∂H/∂ωz)
70 ImpHpQ=ImpHpα*pαpQ+ImpHpβ*pβpQ # Im(∂H/∂Qz)
71
72 self.pHpω0=RepHpω0+1j*ImpHpω0 # ∂H/∂ωz
73 self.pHpQ=RepHpQ+1j*ImpHpQ # ∂H/∂Qz

75class TransferFunctionOneFrequency(object):
76 def __init__(self,ω,variable_list):
77 """
78 The variable list is assumed to be in the following format:
79
80 Gain G and time delay Td followed by two variables per allpass biquad section in order
81 of ω0, Q.
82 """
83 self.ω=ω
84 self.variable_list = variable_list.reshape(variable_list.shape[0],)
85 num_biquads = (len(self.variable_list)-2)//4
86 self.gain = Gain(ω,self.variable_list[0])
87 self.delay = Delay(ω,self.variable_list[1])
88 self.biquad_list = [AllPassBiquadSection(ω,
89 self.variable_list[s*2+2+0], # ω0
90 self.variable_list[s*2+2+1]) # Q
91 for s in range(num_biquads)]
92 bq=math.prod([self.biquad_list[s].H for s in range(len(self.biquad_list))])
93 gd=self.gain.H*self.delay.H
94 self.H=gd*bq
95 self.pd=[]
96 self.pd.append(self.gain.pHpG*self.delay.H*bq)
97 # self.pd[0]=1e9
98 self.pd.append(self.gain.H*self.delay.pHpTd*bq)
99 # self.pd[0]=1e9
100 for biquad in self.biquad_list:
101 this_pd=[]
102 gd_bq_product_others=self.H/biquad.H # the product of all other biquad sections with gain and delay
103 this_pd.append(biquad.pHpω0*gd_bq_product_others)
104 this_pd.append(biquad.pHpQ*gd_bq_product_others)
105 self.pd.extend(this_pd)

107class TransferFunction(object):
108 def __init__(self,ω_list,variable_list):
109 self.fF=[]
110 self.fJ=[]
111 for ω in ω_list:
112 tf=TransferFunctionOneFrequency(ω,variable_list)
113 self.fF.append(tf.H)
114 self.fJ.append(tf.pd)

116class TransferFunctionBiquadVectorized(object):
117 def __init__(self,ω_list,variable_list):
118 """
119 The variable list is assumed to be in the following format:
120
121 Gain G and time delay Td followed by two variables per allpass biquad section in order
122 of ω0, Q.
123 """
124 self.ω_list = np.array(ω_list)
125 self.variable_list = variable_list.reshape(variable_list.shape[0],)
126 num_biquads = (len(self.variable_list)-2)//4
127 self.gain = Gain(self.ω_list,self.variable_list[0])
128 self.delay = Delay(self.ω_list,self.variable_list[1])
129 self.biquad_list = [AllPassBiquadSection(ω_list,
130 self.variable_list[s*2+2+0], # ωz
131 self.variable_list[s*2+2+1]) # Qz
132 for s in range(num_biquads)]
133
134 self.H=self.gain.H*self.delay.H*math.prod([bq.H for bq in self.biquad_list])
135 self.pd=[]
136 self.pd.append(self.gain.pHpG/self.gain.H*self.H)
137 self.pd.append(self.delay.pHpTd/self.delay.H*self.H)
138 for bq in self.biquad_list:
139 this_pd=[]
140 this_pd.append(bq.pHpω0/bq.H*self.H)
141 this_pd.append(bq.pHpQ/bq.H*self.H)
142 self.pd.extend(this_pd)
143 self.fF=self.H.tolist()
144 self.fJ=np.array(self.pd).T.tolist()
