68 If the optional time descriptor is supplied, the resulting impulse response is resampled
69 onto that time descriptor.
70 @note
71 internally, the frequency response is either evenly spaced or not.
72

74 whether to adjust delay determines all possibilities.
75
76 | evenly spaced | time descriptor | adjust delay | Situation |
77 |:------------: |:---------------:|:------------:|:--------------------------------------------------- |
78 | False | False | X | Cannot be done |
79 | False | True | X | Spline resamples to time descriptor |
80 | True | False | False | generic impulse response |
81 | True | False | True | impulse response with delay adjusted |
82 | True | True | X | CZT resamples to td - ad forced to T |
83
84 Much of these options are meant for internal use. Mostly you should simply use ImpulseResponse()
85 with the default arguments.
86 @remark td can also be supplied as a float or int. In this situation, the TimeDescriptor corresponding
87 to the internal FrequencyList is used, but the td supplied supplants the sample rate of the TimeDescriptor.
88 In this way, only the sample rate can be specified in the resampling, and all processing as shown in the
89 table above will assume that a time descriptor has been supplied of this calculated type.
90 """

144 def _Pad(self,P):
145 """Pads the frequency response
146 @param P int number of frequency points to pad to (-1)
147 @note N+1 is the number of points in the selfs frequency response.
148 if P==N, the original response is returned.
149 if P<N, the response is truncated to P+1 frequency points.
150 if P>N, the response is zero padded to P+1 frequency points.
151 """
152 fd=self.FrequencyList()
153 if P == fd.N: X=self.Response()
154 elif P < fd.N: X=self.Response()[:P+1]
155 else: X=self.Response()+[0]*(P-fd.N)
156 return FrequencyResponse(EvenlySpacedFrequencyList(P*fd.Fe/fd.N,P),X)
157 def _Decimate(self,D):
158 """decimates the frequency response
159 @param D integer decimation factor.
160 @note D is assumed >= 1.
161 N+1 is the number of points in the selfs frequency response.
162 N/D+1 is the number of points in the decimated frequency response.
163 """
164 fd=self.FrequencyList()
165 R=self.Response()
166 X=[R[n*D] for n in range(fd.N//D+1)]
167 return FrequencyResponse(EvenlySpacedFrequencyList(fd.N//D*D//fd.N*fd.Fe,fd.N//D),X)

187 def Resample(self,fdp):
188 """Resamples to a different set of frequencies
189 @param fdp instance of class FrequencyList to resample to
190 @return instance of class FrequencyResponse containing resampled self
191 @remark
192 Resampling first attempts to find a ratio of numbers of points
193 to resample to. If a reasonable ratio is found, pure DFT and IDFT
194 methods are utilized along with padding and decimation.
195
196 Otherwise, the chirp z transform is used to resample.
197
198 If the points are unevenly spaced, there is no choice but to resample with
199 splines.
200
201 @see FrequencyResponse.ResampleCZT()
202 @see Spline
203 """
204 fd=self.FrequencyList()
205 # pragma: silent exclude
206 from SignalIntegrity.Lib.FrequencyDomain.FrequencyList import GenericFrequencyList
207 if not fd.CheckEvenlySpaced():
208 if GenericFrequencyList([0]+fd).CheckEvenlySpaced():
209 # only the DC point is missing. Restore that first
210 DCRestored=self._SplineResample(GenericFrequencyList([0]+fd))
211 return DCRestored.Resample(fdp)
212 # pragma: include
213 evenlySpaced = fd.CheckEvenlySpaced() and fdp.CheckEvenlySpaced()
214 if not evenlySpaced: return self._SplineResample(fdp)
215 R=Rat(fd.Fe/fdp.Fe*fdp.N); ND1=R[0]; D2=R[1]
216 if ND1 < fd.N: R=Rat(fd.Fe/fdp.Fe*fdp.N/fd.N); ND1=R[0]*fd.N; D2=R[1]
217 if ND1 > 50000: return self.ResampleCZT(fdp)
218 if ND1 == fd.N: fr=self
219 else: fr=self.ImpulseResponse()._Pad(2*ND1).FrequencyResponse(None,False)
220 if D2*fdp.N != ND1: fr=fr._Pad(D2*fdp.N)
221 if D2==1: return fr
222 else: return fr._Decimate(D2)
223 def _FractionalDelayTime(self):
224 ir = self.ImpulseResponse(None,adjustDelay=False)
225 idx = ir.Values('abs').index(max(ir.Values('abs')))
226 TD = ir.td[idx] # the time of the main peak
227 # calculate the frequency response with this delay taken out
228 # the fractional delay is based on the minimum adjustment to the phase of
229 # the last point to make that point real
230 theta = self._DelayBy(-TD).Response('rad')[self.FrequencyList().N]
231 # do not make this adjustment if the phase is of a tiny magnitude!
232 if self.Response('dB')[self.FrequencyList().N]<-90: theta=0.
233 if theta < -math.pi/2.: theta=-(math.pi+theta)
234 elif theta > math.pi/2.: theta = math.pi-theta
235 else: theta = -theta
236 # calculate the fractional delay
237 TD=theta/2./math.pi/self.FrequencyList()[-1]
238 return TD
