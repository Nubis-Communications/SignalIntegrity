51 def FrequencyResponse(self,fd=None,adjustLength=True):
52 """produces the frequency response
53
54 impulse responses and frequency responses are equivalent and can be converted back
55 and forth into each other. This method converts the impulse response to its corresponding
56 frequency domain equivalent.
57
58 @attention users of this function should only either supply the fd argument or not. Not
59 supplying the argument provides the generic frequency response associated with this impulse
60 response such that self.FrequencyResponse().ImpulseResponse() provides the same answer.
61
62 Supplying fd supplies the resulting frequency response resampled onto another frequency scale.
63
64 @param fd (optional) instance of class FrequencyList (defaults to None).
65 @param adjustLength (optional) bool whether to adjust the length. (defaults to True).
66
67 @note All impulse responses are evenly spaced
68
69 @note whether a frequency descriptor is specified and whether to adjust length determines
70 all possibilities of what can happen:
71
72 | fd | adjustLength | Generates: |
73 |:---------:|:------------:|:---------------------------------------:|
74 | None | False | generic frequency response |
75 | None | True | frequency response with length adjusted |
76 | provided | don't care | CZT resamples to fd (adjusts length) |
77
78 The frequency descriptor, if provided, provides the frequency points to resample to, otherwise
79 the frequency descriptor associated with the internal time descriptor inherent to the impulse
80 response is used.
81
82 Length adjustment means that although the impulse response may start at time zero, or some
83 other time, the assumption is that there are an equal number of points for negative and positve
84 time and that time zero is in the center of these points.
85
86 This assumption for length adjustment is what allows an impulse response to be filled in with
87 positive only times, but allow an equality to exist between all frequency responses and impulse
88 responses.
89
90 Basically, the time adjustment can be seen as calculating the number of points before time zero
91 and the number of points after time zero and calculating the total number of points in the waveform
92 (for the purposes of frequency response calculation) as the max of these two numbers multiplied by two.
93 """
94 # pragma: silent exclude
95 from SignalIntegrity.Lib.FrequencyDomain.FrequencyResponse import FrequencyResponse
96 # pragma: include
97 if not fd and not adjustLength:
98 X=fft.fft(self.Values())
99 fd=self.td.FrequencyList()
100 return FrequencyResponse(fd,[X[n] for n in range(fd.N+1)]).\
101 _DelayBy(self.td.H)
102 if not fd and adjustLength:
103 return self._AdjustLength().FrequencyResponse(None,adjustLength=False)
104 if fd:
105 return self.FrequencyResponse().Resample(fd)
106 def _AdjustLength(self):
107 """adjusts the length of the impulse response
108
109 Calculates the number of points in the impulse response as twice the maximimum of the positive and
110 negative time-domain points.
111
112 @return instance of class ImpulseResponse with the impulse centered at time zero, an even number of
113 points, and an equal number of points before and after time zero.
114 """
115 td = self.td
116 PositivePoints = int(max(0,math.floor(td.H*td.Fs+td.K+0.5)))
117 NegativePoints = int(max(0,math.floor(-td.H*td.Fs+0.5)))
118 P=max(PositivePoints,NegativePoints)*2
119 return self._Pad(P)
120 def _Pad(self,P):
121 """Pads the impulse response
122
123 The impulse response is padded to P points with an equal number of points before and after
124 time zero.
125
126 @param P integer number of points to pad the impulse response to.
127 @return an instance of class ImpulseResponse with the zero padded points.
128 @attention P must be even - not checked - it must add equal points to the left
129 and right of the impulse response.
130 @note
131 if K is the number of points in the selfs frequency response then:
132
133 | P and K | outcome
134 |:--------:|:--------------------------------------------:|
135 | P==K | the original response is returned |
136 | P<K | the response is truncated to P time points |
137 | P>K | the response is zero padded to P time points |
138 """
139 K=len(self)
140 if P==K: x = self.Values()
141 elif P<K: x=[self[k] for k in range((K-P)//2,K-(K-P)//2)]
142 else:
143 x=[0 for p in range((P-K)//2)]
144 x=x+self.Values()+x
145 td = self.td
146 return ImpulseResponse(TimeDescriptor(td.H-(P-K)/2./td.Fs,P,td.Fs),x)
147 def _FractionalDelayTime(self):
148 """fractional delay time of impulse response
149
150 returns the fraction of the sample time since time zero containing the first positive point in the
151 impulse response.
152
153 @return fractional number of samples of delay.
154
155 @note this can be a positive or negative number and will be the between -0.5 and 0.5
156 """
157 td=self.td
158 TD=-(-td.H*td.Fs-math.floor(-td.H*td.Fs+0.5))/td.Fs
159 return TD

168 def TrimToThreshold(self,threshold):
169 """truncates an impulse response, keeping only the values above or equal to specified threshold.
170
171 This is useful for reducing computational complexity in processing.
172
173 @param threshold float threshold to apply to the values in the waveform.
174 @attention the threshold specified is a fraction of the maximum absolute value of the values
175 in the waveform.
176 @note that absolute values are utilized in value/threshold comparison.
177 @note the trimming is performed by finding the lowest and highest time value above or equal
178 to the threshold - all values between these times are retained.
179 """
180 x=self.Values()
181 td=self.td
182 maxabsx=max(self.Values('abs'))
183 minv=maxabsx*threshold
184 for k in range(len(x)):
185 if abs(x[k]) >= minv:
186 startidx = k
187 break
188 for k in range(len(x)):
189 ki = len(x)-1-k
190 if abs(x[ki]) >= minv:
191 endidx = ki
192 break
193 if (endidx-startidx+1)//2*2 != endidx-startidx+1:
194 # the result would not have an even number of points
195 if endidx < len(x)-1:
196 # keep a point at the end if possible
197 endidx = endidx + 1
198 elif startidx > 0:
199 # keep a point at the beginning if possible
200 startidx = startidx - 1
201 else:
202 # append a zero to the end and calculate number of
203 # points with endidx+1
204 return ImpulseResponse(TimeDescriptor(td[startidx],
205 (endidx+1)-startidx+1,td.Fs),
206 [x[k] for k in range(startidx,endidx+1)]+[0.])
207 return ImpulseResponse(TimeDescriptor(td[startidx],
208 endidx-startidx+1,td.Fs),
209 [x[k] for k in range(startidx,endidx+1)])
