42
43 @param Sk instance of class SParameters containing the s-parameters of

45
46 calculates unknown devices in the system, effectively deembedding the
47 surrounding network
48 """
49 Bmsd=self.PortANames()
50 Amsd=self.PortBNames()
51 Adut=self.DutANames()
52 Bdut=self.DutBNames()
53 # pragma: silent exclude
54 K=len(self.PortANames())
55 U=len(self.DutANames())
56 if U > K: # underconstrained
57 raise SignalIntegrityExceptionNumeric('under-constrained system')
58 # pragma: include
59 Internals=self.OtherNames(Bmsd+Amsd+Adut+Bdut)
60 G14=-array(self.WeightsMatrix(Bmsd,Amsd))
61 G15=-array(self.WeightsMatrix(Bmsd,Bdut))
62 G24=-array(self.WeightsMatrix(Adut,Amsd))
63 G25=-array(self.WeightsMatrix(Adut,Bdut))
64 if len(Internals)>0:# internal nodes
65 G13=-array(self.WeightsMatrix(Bmsd,Internals))
66 G23=-array(self.WeightsMatrix(Adut,Internals))
67 # pragma: silent exclude
68 try:
69 # pragma: include outdent
70 G33=array(identity(len(Internals)))-\
71 array(self.WeightsMatrix(Internals,Internals))
72 G34=-array(self.WeightsMatrix(Internals,Amsd))
73 G35=-array(self.WeightsMatrix(Internals,Bdut))
74 F11=self.Dagger(G33,Left=G13,Right=G34,Mul=True)-G14
75 F12=self.Dagger(G33,Left=G13,Right=G35,Mul=True)-G15
76 F21=self.Dagger(G33,Left=G23,Right=G34,Mul=True)-G24
77 F22=self.Dagger(G33,Left=G23,Right=G35,Mul=True)-G25
78 # pragma: silent exclude indent
79 except:
80 raise SignalIntegrityExceptionNumeric('cannot invert G33')
81 # pragma: include
82 else:# no internal nodes
83 F11=-G14; F12=-G15; F21=-G24; F22=-G25
84 # pragma: silent exclude
85 try:
86 # pragma: include outdent
87 #if long and skinny F12 then
88 #F12.getI()=(F12.transpose()*F12).getI()*F12.transpose()
89 #if short and fat F12, F12.getI() is wrong
90 B=self.Dagger(F12,Right=(Sk-F11),Mul=True)
91 # pragma: silent exclude indent
92 except:
93 raise SignalIntegrityExceptionNumeric('cannot invert F12')
94 # pragma: include
95 A=F21+F22.dot(B)
96 AL=self.Partition(A)# partition for multiple unknown devices
97 BL=self.Partition(B)
98 # pragma: silent exclude
99 try:
100 # pragma: include outdent
101 Su=[self.Dagger(AL[u],Left=BL[u],Mul=True).tolist() for u in range(len(AL))]
102 # pragma: silent exclude indent
103 except:
104 raise SignalIntegrityExceptionNumeric('cannot invert A')
105 # pragma: include
