20from SignalIntegrity.Lib.SParameters.SParameters import SParameters
21
22import cmath
23import numpy as np
24
25class FFE(SParameters):
26 """Linear feed-forward equalizer"""
27 def __init__(self,f,Td,taps,numPreCursorTaps,Z0=50.):
28 """Constructor
29 @param f list of float frequencies
30 @param Td float time delay of each tap
31 @param taps list of float values of the taps
32 @param numPreCursorTaps integer number of precursor taps
33 @param Z0 float (optional, defaults to 50) reference impedance
34 @remark Usually Td is the reciprocal of the baud rate or sometimes, half the reciprocal.
35 """
36 self.Td=Td
37 self.taps=taps
38 self.PC=numPreCursorTaps
39 SParameters.__init__(self,f,None,Z0)
40 def _Evaluate(self,f):
41 """Evaluate FFE
42 @param f float frequency
43 @return FFE evaluated at f
44 """
45 K=len(self.taps)
46 return sum([self.taps[k]*cmath.exp(1j*2.*cmath.pi*f*-(k-self.PC)*self.Td) for k in range(K)])
47 def _ResponseVector(self):
48 """Frequency response evaluated at all frequencies.
49
50 The FFE frequency response is the discrete-time Fourier transform of the
51 tap sequence. Evaluating it as a scalar sum over the taps at every
52 frequency is O(frequencies x taps), which becomes very slow when the tap
53 count is large (e.g. long-risetime risetime filters). This computes the
54 whole response at once: an FFT-based evaluation (with tap folding) when the
55 frequency list is evenly spaced from DC and aligns with the tap delay, and
56 a vectorized (chunked) evaluation otherwise. The result is cached.
57 @return numpy array of the response at each frequency.
58 """
59 try:
60 return self._responseVector
61 except AttributeError:
62 pass
63 taps=np.asarray(self.taps,dtype=complex)
64 K=len(taps)
65 f=np.asarray(self.m_f,dtype=float)
66 Np=len(f); Td=self.Td; PC=self.PC
67 H=None
68 if Np>=2 and K>0 and Td!=0:
69 df=f[1]-f[0]
70 if df>0 and np.allclose(np.diff(f),df,rtol=1e-9,atol=0.0) and abs(f[0])<=1e-9*df:
71 Mfloat=1.0/(df*Td)
72 M=int(round(Mfloat))
73 if M>=1 and abs(Mfloat-M)<=1e-6*M:
74 # H[n]=exp(j2*pi*n*PC/M)*sum_k taps[k]*exp(-j2*pi*n*k/M); the inner
75 # sum is the length-M DFT of the taps folded modulo M.
76 g=np.zeros(M,dtype=complex)
77 np.add.at(g,np.arange(K)%M,taps)
78 G=np.fft.fft(g)
79 n=np.arange(Np)
80 H=np.exp(1j*2.*np.pi*n*PC/M)*G[n%M]
81 if H is None:
82 # general vectorized fallback, chunked over frequency to bound memory
83 kv=np.arange(K)-PC
84 base=-1j*2.*np.pi*Td
85 H=np.empty(Np,dtype=complex)
86 chunk=max(1,4000000//max(K,1))
87 for s in range(0,Np,chunk):
88 e=min(s+chunk,Np)
89 H[s:e]=np.exp(base*np.outer(f[s:e],kv)).dot(taps)
90 self._responseVector=H
91 return H
92 def __getitem__(self,n):
93 return [[1.,0],[2.*complex(self._ResponseVector()[n]),-1.]]
