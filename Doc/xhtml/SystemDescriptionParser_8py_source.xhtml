83 def _ProcessLine(self,line,exclusionList):
84 """processes the lines of a netlist
85 Lines that can be processed at this level are processed and lines that
86 are unknown are place in a list of unknown lines for upstream processing. This
87 enables derived classes to benefit from what this class knows how to process and
88 to simply add specific functionality. As a simple example, a derived simulator class
89 needs to add output probes, and this simple system description class knows nothing of
90 this.
91 netlist lines that are handled at this level are:
92 - 'device' - addition of devices.
93 - 'connect' - handles device connections.
94 - 'port' - adds system ports for s-parameter calculation.
95 @param exclusionList list of strings representing commands to exclude. These are either
96 commands that are removed from the functionality (i.e. it would not be write to add a system
97 port in a simulation), or commands that are withheld until further processing at a later time
98 to enforce the order that commands are handled in. For example, we like the netlist to be agnostic
99 about the order of devices listed and device connections, but we cannot connect device ports until
100 a device has been declared.
101 """
102 # pragma: silent exclude
103 from SignalIntegrity.Lib.Parsers.Devices.DeviceParser import DeviceParser
104 from SignalIntegrity.Lib.Helpers.LineSplitter import LineSplitter
105 # pragma: include
106 lineList=self.ReplaceArgs(LineSplitter(line))
107 if len(lineList) == 0: return
108 if self.ProcessVariables(lineList): return
109 elif lineList[0] in exclusionList: self.m_ul.append(line)
110 elif lineList[0] == 'device':
111 argList = lineList[3:]
112 if [lineList[2]]+argList in self.m_spcl:
113 dev = DeviceParser(self.m_f,int(lineList[2]),self.callback,None,
114 Z0=self.m_Z0)
115 dev.m_spf = self.m_spc[self.m_spcl.index([lineList[2]]+argList)][1]
116 else: dev=DeviceParser(self.m_f,int(lineList[2]),self.callback,argList,
117 Z0=self.m_Z0)
118 self.m_sd.AddDevice(lineList[1],int(lineList[2]),dev.m_sp)
119 if not dev.m_spf is None:
120 self.m_spc.append((lineList[1],dev.m_spf))
121 self.m_spcl.append([lineList[2]]+argList)
122 elif lineList[0] == 'connect':
123 # pragma: silent exclude
124 numConnections=(len(lineList)-1)//2
125 if numConnections > 2 and self.MultiPortTee:
126 # more than two device ports being connected -- use a multiport tee
127 teeName=self.m_sd.m_UniqueDevice.Name()
128 self.m_sd.AddDevice(teeName,numConnections)
129 self.m_sd.AssignSParameters(teeName,Tee(numConnections))
130 for i in range(1,len(lineList),2):
131 # hook the tee up to all of the connections
132 self.m_sd.ConnectDevicePort(lineList[i],int(lineList[i+1]),teeName, (i-1)//2+1)
133 else:
134 # pragma: silent include outdent
135 for i in range(3,len(lineList),2):
136 self.m_sd.ConnectDevicePort(lineList[1],int(lineList[2]),
137 lineList[i],int(lineList[i+1]))
138 # pragma: indent
139 elif lineList[0] == 'port':
140 i=1
141 while i < len(lineList):
142 port=int(lineList[i])
143 # pragma: silent exclude
144 if lineList[i+1] == 'td':
145 if not hasattr(self,'delayDict'): self.delayDict={}
146 self.delayDict[port]=float(lineList[i+2])
147 i=i+2
148 # pragma: include
149 dev=lineList[i+1]; devPort=int(lineList[i+2])
150 self.m_sd.AddPort(dev,devPort,port,self.m_addThru)
151 i=i+3
152 else: self.m_ul.append(line)
153 def _ProcessLines(self,exclusionList=[]):
154 """processes all of the lines in a netlist
155 @see _ProcessLine() for explanation of parameters and functionality.
156 """
157 # pragma: silent exclude
158 from SignalIntegrity.Lib.SystemDescriptions.SystemDescription import SystemDescription
159 from SignalIntegrity.Lib.Helpers.LineSplitter import LineSplitter
160 # pragma: include
161 self.m_sd=SystemDescription()
162 self.m_spc=[]; self.m_spcl=[]; self.m_ul=[]
163 if not self.known is None:
164 for key in self.known.keys():
165 self.m_spcl.append(LineSplitter(key))
166 self.m_spc.append((None,self.known[key].Resample(self.m_f)))
167 for line in self.m_lines: self._ProcessLine(line,exclusionList)
168 return self
