24from SignalIntegrity.Lib.FrequencyDomain.FrequencyList import FrequencyList

26class SParametersParser(SParameters):
27 """parses a list of commands to process s-parameters"""
28 def __init__(self,sp,lines,efl=None):
29 """Constructor
30 Takes a set of s-parameters along with a list of command lines to process the s-parameters
31 @param sp instance of SParameters to process
32 @param lines list of lines with commands indicating the processing, in order
33 @param efl (optional, defaults to None) instance of class FrequencyList containing the
34 evenly spaced frequency list to use when time-domain transformations are required.
35 @remark For causality checks, the threshold for causality is 10e-6 (-100 dB).
36 The maximum number of iterations is 30.
37 the Largest singular value is 1.
38 @see EnforceCausality
39 @see EnforcePassivity
40 @see EnforceReciprocity
41 @see EnforceBothPassivityAndReciprocity
42 @see EnforceAll
43 @see LimitImpulseResponseLength
44 @see SetReferenceImpedance
45 @see RemoveImpulseResponseOffset
46 """
47 SParameters.__init__(self,sp.m_f,sp.m_d,sp.m_Z0)
48 self.preserveDC = False
49 # evenlySpaced=FrequencyList(self.m_f).EvenlySpaced()
50 # todo - probably should use whether evenly spaced frequency list is supplied (i.e. not None for this
51 # determination
52 self.efl=efl
53 self.uefl = None if efl is None else sp.m_f
54 for line in lines:
55 tokens=line.split()
56 if len(tokens)>=0:
57 if tokens[0]!='post': continue
58 else: # pragma: no cover
59 continue
60 try:
61 if tokens[1]=='enforce':
62 if tokens[2]=='causality':
63 self.EnforceCausality(preserveDC=self.preserveDC)
64 elif tokens[2]=='passivity':
65 self.EnforcePassivity()
66 elif tokens[2]=='reciprocity':
67 self.EnforceReciprocity()
68 elif tokens[2]=='both':
69 self.EnforceBothPassivityAndCausality(causalityThreshold=10e-6,
70 maxIterations=30,
71 maxSingularValue=1.,
72 preserveDC=self.preserveDC)
73 elif tokens[2]=='all':
74 self.EnforceAll(causalityThreshold=10e-6,
75 maxIterations=30,
76 maxSingularValue=1.,
77 preserveDC=self.preserveDC)
78 else:
79 raise IndexError
80 elif tokens[1] == 'preserve':
81 if tokens[2] in ['dc','DC']:
82 self.preserveDC = True
83 else:
84 raise IndexError
85 elif tokens[1]=='limit':
86 self.LimitImpulseResponseLength((float(tokens[2]) if tokens[2]!='none' else -1e15,
87 float(tokens[3]) if tokens[3]!='none' else +1e15))
88 elif tokens[1]=='offset':
89 if len(tokens)<= 2:
90 self.RemoveImpulseResponseOffset()
91 elif len(tokens)<=3:
92 self.RemoveImpulseResponseOffset((float(tokens[2]) if tokens[2]!='none' else -1e15,+1e15))
93 else:
94 self.RemoveImpulseResponseOffset((float(tokens[2]) if tokens[2]!='none' else -1e15,
95 float(tokens[3]) if tokens[3]!='none' else +1e15))
96 elif tokens[1]=='reference':
97 if tokens[2]=='impedance':
98 try:
99 self.SetReferenceImpedance(float(tokens[3]))
100 except:
101 raise IndexError
102 else:
103 raise IndexError
104 elif tokens[1]=='port':
105 if tokens[2] == 'reorder':
106 try:
107 order=tokens[3]
108 sp=self.PortReorder([int(p) for p in order.split(',')])
109 SParameters.__init__(self,sp.m_f,sp.m_d,sp.m_Z0)
110 except:
111 raise IndexError
112 else:
113 raise IndexError
114 elif tokens[1] == 'taper':
115 if len(tokens) > 2:
116 max_frequency_of_interest = float(tokens[2])
117 if len(tokens) == 4:
118 taper_frequency = float(tokens[3])
119 else:
120 taper_frequency = None
121 self.Taper(from_frequency=max_frequency_of_interest,
122 to_frequency=taper_frequency)
123 else:
124 raise IndexError
125 elif tokens[1] == 'wavelet':
126 if tokens[2] == 'denoise':
127 try:
128 self.WaveletDenoise(float(tokens[3]))
129 except:
130 raise IndexError
131 elif tokens[1][0]=='!':
132 self.header.append(line[len('post !'):])
133 else:
134 raise IndexError
135 except:
136 raise SignalIntegrityExceptionPostProcessing('not understood: '+line)
