32class SpectralDensity(FrequencyDomain):
33 """SpectralDensity
34 Spectral density view of frequency-domain content.
35
36 Stores per-bin amplitude spectral density values (V/sqrt(Hz)) across an
37 instance of class FrequencyList. This class is the spectral-density
38 counterpart of FrequencyResponse and FrequencyContent: it represents the
39 noise (or signal) spectral density distribution, and provides conversions
40 to other common units (dBm/Hz, per-bin rms, per-bin dBm) as well as
41 integrated quantities (total rms, total dBm), random time-domain noise
42 realization, resampling onto a different FrequencyList, and a WhiteNoise()
43 factory.
44 @see FrequencyResponse
45 @see FrequencyContent
46 @see DFTUtilities
47 """
48 R=50.0
49 P=1e-3
50 LogRP10=10.*math.log10(R*P)

85 def ReadFromCSV(self,fileName):
86 """reads a spectral density from a CSV file
87 @param fileName string name of file to read
88 @return self
89 @remark
90 The CSV file is assumed to contain frequency in Hz in the first column
91 and spectral density in V/sqrt(Hz) in the second column. Any
92 non-numeric lines before or within the data are skipped until valid
93 comma-separated frequency,density pairs are found.
94 """
95 frequencies=[]
96 density=[]
97 with open(fileName,'rU' if sys.version_info.major < 3 else 'r') as f:
98 for row in csv.reader(f):
99 if len(row) < 2:
100 continue
101 try:
102 frequency=float(row[0].strip())
103 value=float(row[1].strip())
104 except (TypeError,ValueError):
105 continue
106 if not math.isfinite(frequency) or not math.isfinite(value):
107 continue
108 frequencies.append(frequency)
109 density.append(value)
110 if len(frequencies) == 0:
111 raise ValueError('no spectral density data found in '+str(fileName))
112 fd=FrequencyList(frequencies)
113 fd.CheckEvenlySpaced()
114 SpectralDensity.__init__(self,fd,density,getattr(self,'Keven',True))
115 return self
116 def ReadFromJson(self,fileName):
117 """reads a spectral density from a JSON file
118 @param fileName string name of file to read
119 @return self
120 @remark
121 The JSON data is assumed to contain frequency in Hz and spectral density
122 in V/sqrt(Hz), typically as arrays keyed by 'x' and 'y'.
123 """
124 with open(fileName,'rU' if sys.version_info.major < 3 else 'r') as f:
125 text=f.read()
126
127 decoder=json.JSONDecoder()
128 index=0
129 while index < len(text) and text[index].isspace():
130 index=index+1
131 data,_=decoder.raw_decode(text,index)
132
133 if not isinstance(data,dict):
134 raise ValueError('spectral density json must decode to a dictionary')
135
136 if 'x' in data and 'y' in data:
137 frequencies=data['x']
138 density=data['y']
139 elif 'frequency' in data and 'density' in data:
140 frequencies=data['frequency']
141 density=data['density']
142 elif 'frequencies' in data and 'densities' in data:
143 frequencies=data['frequencies']
144 density=data['densities']
145 else:
146 raise ValueError('spectral density json must contain x/y arrays')
147
148 if len(frequencies) != len(density):
149 raise ValueError('frequency and density arrays must be the same length')
150 if len(frequencies) == 0:
151 raise ValueError('no spectral density data found in '+str(fileName))
152
153 parsed_frequencies=[]
154 parsed_density=[]
155 for frequency,value in zip(frequencies,density):
156 f_hz=float(frequency)
157 rho=float(value)
158 if not math.isfinite(f_hz) or not math.isfinite(rho):
159 raise ValueError('spectral density json contains non-finite values')
160 parsed_frequencies.append(f_hz)
161 parsed_density.append(rho)
162
163 fd=FrequencyList(parsed_frequencies)
164 fd.CheckEvenlySpaced()
165 SpectralDensity.__init__(self,fd,parsed_density,getattr(self,'Keven',True))
166 return self

208 def Values(self,unit=None):
209 """spectral density values
210 @param unit (optional) string containing the unit for the values desired.
211 @return list of float values in the unit specified.
212 @remark
213 Valid spectral-density units are:
214 - 'V/sqrt(Hz)' - amplitude spectral density (default if no unit).
215 - 'dBm/Hz' - power spectral density in dBm/Hz (50 ohm, 1 mW).
216
217 If no unit is specified, the magnitudes of the spectral density values
218 (V/sqrt(Hz)) are returned.
219
220 If no spectral-density unit matches, the call defers to the
221 FrequencyDomain base class for generic complex-value units such as
222 'dB', 'mag', 'rad', 'deg', 'real', and 'imag'.
223 @see FrequencyDomain
224 """
225 if unit is None or unit == 'V/sqrt(Hz)':
226 return [abs(v) for v in list(self)]
227 elif unit in ['dBm/Hz','dBmPerHz']:
228 return [-3000. if r < 1e-15 else 20.*math.log10(r)-self.LogRP10
229 for r in self.ValuesValues('V/sqrt(Hz)')]
230 else:
231 return FrequencyDomain.Values(self,unit)
232 def TotalRMS(self):
233 """Total RMS across the spectrum
234 @return float total rms = sqrt(sum(rms[n]^2)).
235 """
236 deltaf = DFTUtilities.DeltaFrequency((len(self)-1)*2, self.Frequencies()[-1]*2)
237 return DFTUtilities.TotalSpectralContentRMS(
238 DFTUtilities.rho_to_rms(
239 self.ValuesValues(),
240 deltaf))
241 def TotaldBm(self,reference='voltage'):
242 """Total power across the spectrum
243 @param reference (optional) string either 'voltage' (default) or
244 'current' selecting the power law used to convert the integrated rms
245 to power. For a voltage quantity P = rms^2 / R (R divides); for a
246 current quantity P = rms^2 * R (R multiplies).
247 @return float total power in dBm.
248 @note
249 The two references differ by exactly 20*log10(R) dB for the same
250 numeric rms.
251 """
252 deltaf = DFTUtilities.DeltaFrequency((len(self)-1)*2, self.Frequencies()[-1]*2)
253 rms = DFTUtilities.rho_to_rms(self.ValuesValues(), deltaf)
254 if reference == 'current':
255 # P = rms^2 * R (current through the reference impedance)
256 total_power = sum(r*r for r in rms) * self.R
257 return -3000. if total_power <= 1e-300 else 10.*math.log10(total_power/self.P)
258 return DFTUtilities.TotalSpectralContentdBm(
259 DFTUtilities.rms_to_dBm(rms))
260 def SNRdB(self, other, other_is_signal_or_noise):
261 """Calculates SNR in dB from integrated spectral-density powers.
262 @param other instance of class SpectralDensity.
263 @param other_is_signal_or_noise string either 'signal' or 'noise'.
264 When set to 'signal', other is treated as signal and self as noise.
265 When set to 'noise', other is treated as noise and self as signal.
266 @return float SNR in dB computed as signal_dBm - noise_dBm.
267 """
268 if not isinstance(other, SpectralDensity):
269 raise TypeError('other must be an instance of SpectralDensity')
270
271 if other_is_signal_or_noise not in ['signal', 'noise']:
272 raise ValueError("other_is_signal_or_noise must be 'signal' or 'noise'")
273
274 if other_is_signal_or_noise == 'signal':
275 signal_dbm = other.TotaldBm()
276 noise_dbm = self.TotaldBm()
277 else:
278 signal_dbm = self.TotaldBm()
279 noise_dbm = other.TotaldBm()
280
281 return signal_dbm - noise_dbm
282 def SalzSNRdB(self, other, other_is_signal_or_noise, noise_floor=1e-300):
283 """Calculates Salz SNR in dB using bin-wise PSD ratios.
284 @param other instance of class SpectralDensity.
285 @param other_is_signal_or_noise string either 'signal' or 'noise'.
286 When set to 'signal', other is treated as signal and self as noise.
287 When set to 'noise', other is treated as noise and self as signal.
288 @param noise_floor (optional) minimum linear noise PSD required for a
289 bin to be included in the average. Defaults to 1e-300 (effectively no filtering).
290 @return float Salz SNR in dB.
291 @remark
292 other is resampled to self's frequency list before the computation.
293 The implemented definition is:
294 SalzSNR = 10*log10(mean(PSD_signal[n]/PSD_noise[n]))
295 where PSD = rho^2 and rho is in V/sqrt(Hz).
296 Bins whose noise PSD is below noise_floor are excluded from the mean.
297 """
298 if not isinstance(other, SpectralDensity):
299 raise TypeError('other must be an instance of SpectralDensity')
300
301 if other_is_signal_or_noise not in ['signal', 'noise']:
302 raise ValueError("other_is_signal_or_noise must be 'signal' or 'noise'")
303
304 if noise_floor <= 0:
305 raise ValueError('noise_floor must be positive')
306
307 other_on_self = other.Resample(self.FrequencyList())
308
309 if other_is_signal_or_noise == 'signal':
310 signal_rho = other_on_self.Values('V/sqrt(Hz)')
311 noise_rho = self.ValuesValues('V/sqrt(Hz)')
312 else:
313 signal_rho = self.ValuesValues('V/sqrt(Hz)')
314 noise_rho = other_on_self.Values('V/sqrt(Hz)')
315
316 # Compute PSD ratios, filtering by noise floor and excluding zero noise values
317 psd_ratios = []
318 for s, n in zip(signal_rho, noise_rho):
319 n_psd = n * n
320 if n_psd >= noise_floor and n_psd > 0:
321 psd_ratios.append((s*s) / n_psd)
322
323 if len(psd_ratios) == 0:
324 # If no bins pass the noise floor filter, try without floor (excluding only zero noise).
325 # This handles cases where noise spectral density is extremely small
326 # (near machine epsilon) across the entire spectrum.
327 for s, n in zip(signal_rho, noise_rho):
328 n_psd = n * n
329 if n_psd > 0:
330 psd_ratios.append((s*s) / n_psd)
331
332 if len(psd_ratios) == 0:
333 # If still no valid ratios, all noise is zero or undefined.
334 # This represents infinite SNR, represented as very large dB value.
335 return 3000.0 # Represent infinite SNR as very large positive dB
336
337 salz_linear = sum(psd_ratios) / float(len(psd_ratios))
338
339 return -3000. if salz_linear < 1e-300 else 10.*math.log10(salz_linear)
340 def NoiseWaveform(self,td=None):
341 """Generates a random time-domain noise realization from the spectral density.
342 @param td (optional) instance of class TimeDescriptor describing the time
343 descriptor of the waveform to produce.
344 @return instance of class Waveform containing a random noise realization
345 whose expected spectrum matches self.
346 @remark
347 Each interior bin is assigned an independent uniformly distributed phase;
348 the DC bin (and the Nyquist bin when Keven) are kept real so that the
349 inverse DFT yields a real-valued waveform.
350 @note
351 If td is None, the time descriptor corresponding to self's FrequencyList
352 is used.
353 """
354 # pragma: silent exclude
355 from SignalIntegrity.Lib.TimeDomain.Waveform.Waveform import Waveform
356 # pragma: include
357 fd=self.FrequencyList()
358 delta_f=fd.Fe/fd.N
359 rho=self.ValuesValues('V/sqrt(Hz)')
360 rms=DFTUtilities.rho_to_rms(rho,delta_f,self.Keven)
361 A=DFTUtilities.rms_to_A(rms,self.Keven)
362 X=DFTUtilities.A_to_X(A,self.Keven,random_phase=True)
363 F=DFTUtilities.Half_to_Full(X,self.Keven)
364 x=[v.real for v in fft.ifft(F).tolist()]
365 wf_td=fd.TimeDescriptor(Keven=self.Keven)
366 wf=Waveform(wf_td,x)
367 if td is not None:
368 wf=wf.Adapt(td)
369 return wf
370 def Resample(self,fdp):
371 """Resamples the spectral density to a different frequency list.
372 @param fdp instance of class FrequencyList to resample to.
373 @return instance of class SpectralDensity containing self resampled onto
374 fdp using linear interpolation of the V/sqrt(Hz) values.
375 @remark Resampling is performed on the magnitude in V/sqrt(Hz); points
376 beyond self's end frequency are clamped to zero.
377 """
378 if fdp is None:
379 return SpectralDensity(self.FrequencyList(),self.ValuesValues(),self.Keven)
380 from numpy import interp
381 fd=self.FrequencyList()
382 old_f=list(fd)
383 old_rho=self.ValuesValues('V/sqrt(Hz)')
384 new_f=list(fdp)
385 interpolated=interp(new_f,old_f,old_rho).tolist()
386 new_rho=[float(v) if f<=old_f[-1] else 0.0
387 for f,v in zip(new_f,interpolated)]

390 def WhiteNoise(fd,specification_type,value,noise_bandwidth=None,Keven=True):
391 """Constructs a flat (white) noise SpectralDensity.
392 @param fd instance of class FrequencyList for the frequency descriptor.
393 @param specification_type string one of 'dBm/Hz', 'V/sqrt(Hz)', or 'Vrms'.
394 @param value float the noise level expressed in the specification_type:
395 - 'dBm/Hz' : power spectral density in dBm/Hz
396 - 'V/sqrt(Hz)' : amplitude spectral density in V/sqrt(Hz)
397 - 'Vrms' : total rms voltage over noise_bandwidth
398 @param noise_bandwidth (optional) float bandwidth in Hz over which the
399 total rms value is specified. Required only when
400 specification_type == 'Vrms'.
401 @param Keven (optional) bool whether the corresponding time-record length
402 is even. Defaults to True.
403 @return instance of class SpectralDensity containing a flat spectral
404 density evaluated at each frequency in fd.
405 """
406 if specification_type == 'V/sqrt(Hz)':
407 rho=value
408 elif specification_type == 'dBm/Hz':
409 rho=math.sqrt(SpectralDensity.R*SpectralDensity.P*
410 10.**(value/10.))
411 elif specification_type == 'Vrms':
412 if noise_bandwidth is None or noise_bandwidth <= 0:
413 raise ValueError(
414 'noise_bandwidth must be positive for Vrms specification')
415 rho=value/math.sqrt(noise_bandwidth)
416 else:
417 raise ValueError(
418 "specification_type must be 'dBm/Hz', 'V/sqrt(Hz)', or 'Vrms'")
419 return SpectralDensity(fd,[rho for _ in range(fd.N+1)],Keven)
