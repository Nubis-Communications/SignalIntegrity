28class TransferMatrices(list,CallBacker):
29 """Class that is used for processing waveforms in simulation."""
30 cacheResponses=True
31 def __init__(self,f,d):
32 """Constructor
33 @param f instance of class FrequencyList
34 @param d list of list of list matrices
35 @remark
36 The list of list of list matrices in d are such that each element in the list
37 represents a list of list matrix for a given frequency. The list of list matrix
38 for a frequency is the frequency response that converts inputs (columns) to outputs
39 (rows). If d is an N+1 element list of RxC matrices, then for n in 0..N, d[n][r][c]
40 represents the frequency response for a filter used to convert input C into output
41 r.
42
43 Generally, you don't deal with the structure of d as it is produced completely by the
44 two classes SimulatorNumericParser and VirtualProbeNumericParser, and an element of d
45 for a frequency is produced by the two classes SimulatorNumeric and VirtualProbeNumeric.
46 """
47 self.f=FrequencyList(f)
48 list.__init__(self,d)
49 CallBacker.__init__(self)
50 self.Inputs=len(d[0][0])
51 self.Outputs=len(d[0])
52 self.fr=None
53 self.ir=None
54 self.td=None
55 def SParameters(self):
56 """SParameters
57 @return list of list of lists representing the transfer matrices as s-parameters.
58 @note that transfer matrices are actually very much like s-parameters because they
59 provide the complex frequency-domain relationship between an input and output at the
60 given frequencies. The only actual difference is that an s-parameter matrix must be
61 square while transfer parameter matrices can be rectangular.
62
63 s-parameters make it convenient for writing and viewing transfer matrices, so this
64 format is provided. The transfer matrices converted to s-parameters will have a square
65 matrix whose row and column length is the maximum of the row and column length of a
66 transfer matrix.
67 """
68 # pragma: silent exclude
69 from SignalIntegrity.Lib.SParameters.SParameters import SParameters
70 # pragma: include
71 if self.Inputs == self.Outputs:
72 return SParameters(self.f,self)
73 else:
74 squareMatrices=[]
75 P=max(self.Inputs,self.Outputs)
76 for transferMatrix in self:
77 squareMatrix=zeros((P,P),complex).tolist()
78 for r in range(len(transferMatrix)):
79 for c in range(len(transferMatrix[0])):
80 squareMatrix[r][c]=transferMatrix[r][c]
81 squareMatrices.append(squareMatrix)
82 return SParameters(self.f,squareMatrices)
83 def FrequencyResponse(self,o,i):
84 """frequency response of one filter
85 @param o integer index of output
86 @param i integer index of input
87 @return instance of class FrequencyResponse corresponding to the frequency response of
88 a filter used to convert input i to output o.
89 """
90 # pragma: silent exclude
91 from SignalIntegrity.Lib.FrequencyDomain.FrequencyResponse import FrequencyResponse
92 # pragma: include
93 if not self.cacheResponses or self.fr == None:
94 return FrequencyResponse(self.f,[Matrix[o-1][i-1]
95 for Matrix in self])
96 else:
97 return copy.deepcopy(self.fr[o-1][i-1])
98 def FrequencyResponses(self):
99 """frequency responses of filters
100 @return list of list of instances of class FrequencyResponse
101 @remark
102 The return is a list of list like a matrix where each element in the matrix M is
103 such that M[o][i] is the frequency response of a filter that would convert the
104 input i to an output o.
105 @see FrequencyResponse()
106 """
107 if not self.cacheResponses or self.fr==None:
108 fr = [[None for s in range(self.Inputs)] for o in range(self.Outputs)]
109 for o in range(self.Outputs):
110 for s in range(self.Inputs):
111 fr[o][s] = self.FrequencyResponse(o+1,s+1)
112 if not self.CallBack((o*self.Inputs+s)/
113 (self.Inputs*self.Outputs)*100.0):
114 return None
115 if not self.cacheResponses:
116 return fr
117 self.fr = fr
118 return copy.deepcopy(self.fr)
119 def ImpulseResponses(self,td=None):
120 """impulse responses of filters
121 @return list of list of instances of class ImpulseResponse
122 @remark
123 The return is a list of list like a matrix where each element in the matrix M is
124 such that M[o][i] is the impulse response of a filter that would convert the
125 input i to an output o.
126 """
127 fr = self.FrequencyResponses()
128 if td is None or isinstance(td,float) or isinstance(td,int):
129 td = [td for _ in range(self.Inputs)]
130 if fr == None:
131 return None
132 if self.cacheResponses and self.td == td and self.ir != None:
133 return self.ir
134 ir = [[None for s in range(self.Inputs)] for o in range(self.Outputs)]
135 for o in range(self.Outputs):
136 for s in range(self.Inputs):
137 ir[o][s] = fr[o][s].ImpulseResponse(td[s])
138 if not self.CallBack((o*self.Inputs+s)/
139 (self.Inputs*self.Outputs)*100.0):
140 return None
141 if not self.cacheResponses:
142 return ir
143 self.ir = ir
144 self.td = td
145 return copy.deepcopy(self.ir)
146 def Resample(self,fdp):
147 """Resamples to a different set of frequencies
148 @param fdp instance of class FrequencyList to resample to
149 @return instance of class FrequencyResponse containing resampled self
150 @remark
151 Resampling first attempts to find a ratio of numbers of points
152 to resample to. If a reasonable ratio is found, pure DFT and IDFT
153 methods are utilized along with padding and decimation.
154
155 Otherwise, the chirp z transform is used to resample.
156
157 If the points are unevenly spaced, there is no choice but to resample with
158 splines.
159
160 @see FrequencyResponse.ResampleCZT()
161 @see Spline
162 """
163 fr = self.FrequencyResponses()
164 fr = [[fr[o][s].Resample(fdp)
165 for s in range(self.Inputs)]
166 for o in range(self.Outputs)]
167 d = [[[fr[o][s][n]
168 for s in range(self.Inputs)]
169 for o in range(self.Outputs)]
170 for n in range(len(fdp))]
171 return TransferMatrices(fdp,d)
172 def Remove(self,outputs,sources,outputs_to_remove,sources_to_remove):
173 """Removes specified outputs and sources from the transfer matrices.
174 @param outputs list of output names corresponding to the rows of the transfer matrices.
175 @param sources list of source names corresponding to the columns of the transfer matrices.
176 @param outputs_to_remove list of output names to remove.
177 @param sources_to_remove list of source names to remove.
178 @return tuple (new_outputs, new_sources, new_transfer_matrices) where:
179 - new_outputs is the list of outputs with outputs_to_remove removed
180 - new_sources is the list of sources with sources_to_remove removed
181 - new_transfer_matrices is a new instance of TransferMatrices with the
182 corresponding rows and columns removed from each frequency's matrix.
183 @remark
184 The transfer matrices are a list of list of list where each element corresponds
185 to a frequency and is a list of list matrix M such that, for that frequency,
186 outputs = M * sources. After removal, the new matrix at each frequency contains
187 only the rows for outputs not in outputs_to_remove and only the columns for
188 sources not in sources_to_remove.
189 """
190 output_indices_to_keep=[o for o in range(len(outputs))
191 if outputs[o] not in outputs_to_remove]
192 source_indices_to_keep=[s for s in range(len(sources))
193 if sources[s] not in sources_to_remove]
194 new_outputs=[outputs[o] for o in output_indices_to_keep]
195 new_sources=[sources[s] for s in source_indices_to_keep]
196 new_d=[[[self[n][o][s] for s in source_indices_to_keep]
197 for o in output_indices_to_keep]
198 for n in range(len(self))]
199 return (new_outputs,new_sources,TransferMatrices(self.f,new_d))
200 def Keep(self,outputs,sources,outputs_to_keep,sources_to_keep):
201 """Keeps only the specified outputs and sources in the transfer matrices.
202 @param outputs list of output names corresponding to the rows of the transfer matrices.
203 @param sources list of source names corresponding to the columns of the transfer matrices.
204 @param outputs_to_keep list of output names to keep.
205 @param sources_to_keep list of source names to keep.
206 @return tuple (new_outputs, new_sources, new_transfer_matrices) where:
207 - new_outputs is the list of outputs containing only those in outputs_to_keep,
208 in the order they appear in outputs.
209 - new_sources is the list of sources containing only those in sources_to_keep,
210 in the order they appear in sources.
211 - new_transfer_matrices is a new instance of TransferMatrices with only the
212 corresponding rows and columns retained in each frequency's matrix.
213 @remark
214 The transfer matrices are a list of list of list where each element corresponds
215 to a frequency and is a list of list matrix M such that, for that frequency,
216 outputs = M * sources. After keeping, the new matrix at each frequency contains
217 only the rows for outputs in outputs_to_keep and only the columns for sources
218 in sources_to_keep.
219 """
220 output_indices_to_keep=[o for o in range(len(outputs))
221 if outputs[o] in outputs_to_keep]
222 source_indices_to_keep=[s for s in range(len(sources))
223 if sources[s] in sources_to_keep]
224 new_outputs=[outputs[o] for o in output_indices_to_keep]
225 new_sources=[sources[s] for s in source_indices_to_keep]
226 new_d=[[[self[n][o][s] for s in source_indices_to_keep]
227 for o in output_indices_to_keep]
228 for n in range(len(self))]
229 return (new_outputs,new_sources,TransferMatrices(self.f,new_d))
