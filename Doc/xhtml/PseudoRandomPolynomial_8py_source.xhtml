63 def __init__(self,polynomial):
64 """constructor
65 @param polynomial integer number or list of integer (1 or 0) of polynomial coefficients
66 @remark if the polynomial parameter given is an integer, the polynomial is generated for that prbs integer number
67 (i.e. prbs7 if 7 given).
68 @remark the first element of the polynomial is for the zero power and is always 1.
69
70 The second element is for power 1, the third element is power 2, etc.
71
72 polynomials are found at https://en.wikipedia.org/wiki/Linear-feedback_shift_register
73 """
74 if isinstance(polynomial,int):
75 number=polynomial
76 if not number in self.polynomials:
77 raise SignalIntegrityException('PseudoRandomPolynomial','pseudo-random polynomial not found')
78 polynomial=[0 for _ in range(number+1)]
79 for c in self.polynomials[number]: polynomial[c]=1
80 list.__init__(self,polynomial)
81 def Order(self):
82 """Order
83 @return integer polynomial order
84
85 This is the length of the polynomial - 1
86 """
87 return len(self)-1
88 def PatternLength(self):
89 """PatternLength
90 @return integer pattern length generated by polynomial.
91
92 This is two raised to the polynomial order minus 1
93 @see Order
94 """
95 return 2**self.Order()-1
96 def Pattern(self,bits=None):
97 """generate a list of pseudo-random bits according to the polynomial
98 @param bits (optional, defaults to None) integer number of bits to return.
99 If bits are specified, then the pattern returned will contain the number of bits specified.
100 @return returns a list of integer (1 or 0) corresponding to the pattern.
101
102 for an order P polynomial, the list of bits is 2^P-1 elements long, assuming bits are specified as None.
103 """
104 order = self.Order()
105 length=self.PatternLength() if bits == None else min(self.PatternLength(),bits)
106 taps=[]
107 for b in range(1,len(self)):
108 if self[b] == 1:
109 taps.append(b-1)
110
111 lfsr = [1] * order
112 pattern = []
113
114 for _ in range(length):
115 pattern.append(lfsr[-1])
116 feedback = 0
117 for tap in taps:
118 feedback ^= lfsr[tap]
119 lfsr = [feedback] + lfsr[:-1]
120
121 pattern = [(b+1)%2 for b in pattern]
122 return pattern
