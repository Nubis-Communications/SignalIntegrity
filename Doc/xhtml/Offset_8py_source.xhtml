22import cmath

24from SignalIntegrity.Lib.SParameters.SParameters import SParameters
25
26class Offset(SParameters):
27 """class providing the s-parameters of an offset portion of the various calibration standards
28 for a calibration kit."""
29 # pragma: silent exclude
30 calcZc=False
31 # pragma: include
32 def __init__(self,fList,offsetDelay,offsetZ0,offsetLoss,f0=1e9,Z0=50.):
33 """Constructor
34 @param fList list of frequencies
35 @param offsetDelay (optional) float electrical length of offset in s (defaults to 0 s)
36 @param offsetZ0 (optional) float real characteristic impedance of offset (defaults to 50 ohms)
37 @param offsetLoss (optional) float loss due to skin-effect defined in ohms/s at f0 (defaults to 0).
38 @param f0 (optional) float frequency where the offset loss is defined (defaults to 1e9).
39 @param Z0 float (optional, defaults to 50.) reference impedance for the calculation.
40 The result is that the class becomes the base-class SParameters with the s-parameters
41 of an offset, which is a common portion of a calibration standard.
42 """
43 data=[]
44 Zc=offsetZ0
45 Td=offsetDelay
46 R0=offsetLoss
47 L=Td*Zc
48 C=Td/Zc
49 G=0
50 for f in fList:
51 R=R0*Td*cmath.sqrt(f/f0)*(1+1j)
52 Z=R+1j*2*cmath.pi*f*L
53 Y=G+1j*2*cmath.pi*f*C
54 y=cmath.sqrt(Z*Y)
55 # pragma: silent exclude
56 try:
57 if self.calcZc:
58 Zc=cmath.sqrt(Z/Y)
59 except:
60 Zc=offsetZ0
61 # pragma: include
62 rho=(Zc-Z0)/(Zc+Z0)
63 D=(1-rho*rho*cmath.exp(-2*y))
64 S11=rho*(1-cmath.exp(-2*y))/D
65 S21=(1-rho*rho)*cmath.exp(-y)/D
66 data.append([[S11,S21],[S21,S11]])
67 SParameters.__init__(self,fList,data,Z0)
