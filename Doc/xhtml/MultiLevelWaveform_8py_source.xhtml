28class MultiLevelWaveform(Waveform):
29 """a PRBS multi-level waveform with a given PRBS polynomial"""
30 def __init__(self,polynomial,baudrate,bitsPerSymbol=1,amplitude=1.0,risetime=0.,delay=0.,td=None):
31 """constructor
32 @param polynomial integer polynomial number
33 @param baudrate, amplitude, risetime, delay, td all pertain to the derived SerialDataWaveform class
34 @see SerialDataWaveform
35 @param bitsPerSymbol integer (defaults to 1) are the number of bits per symbol and determines the number
36 of levels in the waveform. bitsPerSymbol = 1, means NRZ, 2 means PAM-4, 3 means PAM-8 etc.
37 @return self, a waveform.
38 @throw SignalIntegrityWaveform exception is raised if the polynomial number cannot be found
39 @see PseudoRandomPolynomial
40 @note the pseudo-random bits are grouped for each symbol.
41 """
42 symbols=int(math.ceil(td.Duration()*baudrate)); levels=pow(2,bitsPerSymbol)
43 pattern=PseudoRandomPolynomial(polynomial).Pattern(symbols*bitsPerSymbol) if isinstance(polynomial,int) else polynomial
44 wf=sum([SerialDataWaveform([pattern[(i*bitsPerSymbol+k)%(len(pattern))] for i in range(symbols)],
45 baudrate,pow(2.,(bitsPerSymbol-k-1.))/(levels-1)*amplitude,risetime,delay,td) for k in range(bitsPerSymbol)])
46 Waveform.__init__(self,wf)
