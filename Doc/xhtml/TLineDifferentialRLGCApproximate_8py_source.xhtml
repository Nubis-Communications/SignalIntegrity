27class TLineDifferentialRLGCApproximate(SParameters):
28 """s-parameters of differential RLGC (telegrapher's) transmission line
29 calculated by approximating distributed parameters with a finite number
30 of sections specified."""
31 rtFraction=.01
32 def __init__(self,f, Rp, Rsep, Lp, Gp, Cp, dfp, Rn, Rsen, Ln, Gn, Cn, dfn,
33 Cm, dfm, Gm, Lm, Z0=50., K=0, scale=1.):
34 """Constructor
35
36 ports are 1,2,3,4 is +1,-1, +2, -2
37
38 @param f list of float frequencies
39 @param Rp float DC resistance of positive leg (ohms)
40 @param Rsep float skin-effect resistance of positive leg (ohms/sqrt(Hz))
41 @param Lp float inductance of positive leg (H)
42 @param Gp float DC conductance of positive leg to ground (S)
43 @param Cp float capacitance of positive leg to ground (F)
44 @param dfp float dissipation factor (loss-tangent) of capacitance of positive leg to ground
45 @param Rn float DC resistance of negative leg (ohms)
46 @param Rsen float skin-effect resistance of negative leg (ohms/sqrt(Hz))
47 @param Ln float inductance of negative leg (H)
48 @param Gn float DC conductance of negative leg to ground (S)
49 @param Cn float capacitance of negative leg to ground (F)
50 @param dfn float dissipation factor (loss-tangent) of capacitance of negative leg to ground
51 @param Cm float mutual capacitance (F)
52 @param dfm float dissipation factor (loss-tangent) of mutual capacitance (F)
53 @param Gm float mutual conductance (S)
54 @param Lm float mutual inductance (H)
55 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
56 @param K (optional) float number of sections (defaults to 0)
57 @param scale (optional) float amount to scale the line by (assuming all other values are per unit)
58 @note If K=0 is specified, it is modified to a value that will provided a very good numerical
59 approximation.
60
61 The calculation is such that round-trip propagation time (twice the electrical length)
62 of any one small section is no more than one percent of the fastest possible risetime.
63 """
64 Rp=Rp*scale; Rsep=Rsep*scale; Lp=Lp*scale; Gp=Gp*scale; Cp=Cp*scale; dfp=dfp
65 Rn=Rn*scale; Rsen=Rsen*scale; Ln=Ln*scale; Gn=Gn*scale; Cn=Cn*scale; dfn=dfn
66 Cm=Cm*scale; dfm=dfm; Gm=Gm*scale; Lm=Lm*scale
67 K=K*scale
68 if K==0:
69 """max possible electrical length and fastest risetime"""
70 Td=math.sqrt((max(Lp,Ln)+Lm)*(max(Cp,Cn)+2*Cm)); Rt=0.45/f[-1]
71 """sections such that fraction of risetime less than round trip electrical
72 length of one section"""
73 K=Td*2/(Rt*self.rtFraction)
74 self.m_K=K
