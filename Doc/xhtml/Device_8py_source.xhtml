58 def SymbolicMatrix(Name,Rows,Columns=-1):
59 """creates a symbolic matrix
60 The matrix is of a simple form. It is a list of list of strings where the
61 string is either:
62 - rows=columns=1 - [[name]]
63 - rows <= 9 and columns <= 9 - list of list of name+'_'+str(r+1)+str(c+1)
64 @note if a row or column goes above 9, a comma is inserted between the port numbers
65 to remove ambiguity.
66 @param Name string name of device
67 @param Rows integer number of rows in the matrix
68 @param Columns (optional) integer number of columns in the matrix (defaults
69 to the number of rows).
70 """
71 if Columns == -1:
72 Columns = Rows
73 if Rows == 1 and Columns == 1:
74 return [[Name]]
75 return [[Name+'_'+(str(r+1)+str(c+1) if r<9 and c<9 else str(r+1)+','+str(c+1))
76 for c in range(Columns)] for r in range(Rows)]
