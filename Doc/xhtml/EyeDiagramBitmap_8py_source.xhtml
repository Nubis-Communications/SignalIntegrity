40 def Hits(self,r,c):
41 R=self.rows; C=self.columns
42 lowestr=r-0.5
43 rl=int(math.floor(lowestr))
44 highestr=r+0.5
45 rh=int(math.floor(highestr))
46 lrh=lowestr-rl
47 lrl=1-lrh
48 lowestc=c-0.5
49 cl=int(math.floor(lowestc))
50 highestc=c+0.5
51 ch=int(math.floor(highestc))
52 lch=lowestc-cl
53 lcl=1-lch
54 rlValid=(0<=rl<R)
55 rhValid=(0<=rh<R)
56 if cl<0:
57 cl=cl+C
58 if ch>=C:
59 ch=ch-C
60 clValid=(0<=cl<C)
61 chValid=(0<=ch<C)
62 results=[]
63 if rlValid and clValid: results+=[((rl,cl),(lrl,lcl),lrl*lcl/self.step)]
64 if rhValid and clValid: results+=[((rh,cl),(lrh,lcl),lrh*lcl/self.step)]
65 if rlValid and chValid: results+=[((rl,ch),(lrl,lch),lrl*lch/self.step)]
66 if rhValid and chValid: results+=[((rh,ch),(lrh,lch),lrh*lch/self.step)]

71 def Results(self,ri,ci,rf,cf,useFirst=False):
72 C=self.columns
73 steps=self.step if not self.auto else max(1,min(int(math.ceil(2*abs(rf-ri))),self.step))
74 if ci>cf: cf=cf+C
75 cspan=cf-ci
76 rspan=rf-ri

289 def ApplyJitterNoise(self,
290 NoiseSigma=0.,
291 JitterSigma=0.,
292 DeterministicJitter=0.,
293 MaxPixelsKernel=100000):
294 """Applies jitter and/or noise to the eye diagram bitmap
295 @param NoiseSigma float, defaults to 0, amount of rms vertical noise to apply.
296 @param JitterSigma float, defaults to 0, amount of rms horizontal jitter to apply.
297 @param DeterministicJitter float, defaults to 0, amount of deterministic jitter to apply
298 horizontally.
299 @param MaxPixelsKernel int, defaults to 100000, maximum number of pixels in kernel
300
301 Noise and jitter are applied by constructing a kernel and then convolving this kernel with
302 the eye diagram bitmap. The kernel itself is a convolution of a vertical histogram, containing
303 the Gaussian distributed vertical noise and a horizontal histogram, containing the jitter.
304 The jitter histogram is also, itself, a convolution of a dual-Dirac, whose pk-pk separation is
305 twice the amount of deterministic jitter specified, with a Guassian distribution representing
306 the jitter specified.
307 The kernel generated covers 10 standard deviations in both vertical and horizontal direction.
308 As a protection against absurd numbers for jitter and noise, a check is made that the number of
309 points in the kernel does not exceed a gigantic number. If the kernel exceeds this amount, it
310 is truncated.
311 When this function concludes, the side effect is that the rawBitmap member variable is updated
312 with a new bitmap representing the noise and jitter addition effect.
313 @remark The shape of the bitmap is adjusted by padding rows based on the amount of vertical noise
314 added.
315 """
316 self.NoiseSigma=NoiseSigma
317 UI=1./self.BaudRate
318 deltaT=UI/self.Cols
319 deltaY=(self.maxV-self.minV)/self.RowsSpecified
320
321 WH=int(math.floor(math.floor(2.*10.*(DeterministicJitter+JitterSigma)/deltaT/2.)*2.)+1.)
322 WV=int(math.floor(math.floor(2.*10.*NoiseSigma/deltaY/2.)*2.)+1.)
323
324 if WH*WV > MaxPixelsKernel:
325 WH=int(math.floor((WH*math.sqrt(float(MaxPixelsKernel)/float(WH*WV))-1)/2))*2+1
326 WV=int(math.floor((WV*math.sqrt(float(MaxPixelsKernel)/float(WH*WV))-1)/2))*2+1

328 def cnorm(x,mu,sigma):
329 if sigma==0.:
330 if x < mu:
331 return 0.
332 if x==mu:
333 return 0.5
334 if x > mu:
335 return 1.0
336 else:
337 res= (1.+math.erf((x-mu)/sigma/math.sqrt(2.)))/2.
338 return res
339
340 def dnorm(x,mu,sigma):
341 res = cnorm(x+1/2.,mu,sigma)-cnorm(x-1/2.,mu,sigma)
342 return res
343
344 # make the Gaussian kernel
345 bitmaparray=np.array(self.rawBitmap)
346 kernelHarray=np.array([[(dnorm(wh-(WH-1)//2,-DeterministicJitter/deltaT,JitterSigma/deltaT)+dnorm(wh-(WH-1)//2,DeterministicJitter/deltaT,JitterSigma/deltaT))/2.0 for wh in range(WH)]])
347 kernelVarray=np.array([[dnorm(wv-(WV-1)//2,0,NoiseSigma/deltaY)] for wv in range(WV)])
348
349 # trim away the edges of the vertical part of the kernel that is containing zeros to save time
350 for vertStart in range(kernelVarray.shape[0]):
351 if kernelVarray[vertStart][0]!=0:
352 break
353 for vertEnd in range(kernelVarray.shape[0]):
354 if kernelVarray[kernelVarray.shape[0]-1-vertEnd][0]!=0:
355 vertEnd=kernelVarray.shape[0]-1-vertEnd
356 break
357 kernelVarray=kernelVarray[vertStart:vertEnd,:]
358
359 # trim away the edges of the horizontal part of the kernel that is containing zeros to save time
360 for horStart in range(kernelHarray.shape[1]):
361 if kernelHarray[0][horStart]!=0:
362 break
363 for horEnd in range(kernelHarray.shape[1]):
364 if kernelHarray[0][kernelHarray.shape[1]-1-horEnd]!=0:
365 horEnd=kernelHarray.shape[1]-1-horEnd
366 break
367 kernelHarray=kernelHarray[:,horStart:horEnd]
368
369 # pad the bitmap array to encompass the vertical size of the kernel
370 rowsToPad=min(kernelVarray.shape[0],self.RowsSpecified)
371 if rowsToPad > 0:
372 padding=np.zeros((rowsToPad,self.Cols))
373 bitmaparray=np.vstack((padding,bitmaparray,padding))
374 #self.RowsSpecified+=rowsToPad*2
375 self.minV-=rowsToPad*deltaY
376 self.maxV+=rowsToPad*deltaY
377
378 try:
379 from scipy.ndimage import convolve
380 except ImportError:
381 from scipy.ndimage.filters import convolve
382 bitmap=bitmaparray
383 if kernelHarray.shape[1]!=0: bitmap=convolve(bitmaparray,kernelHarray,mode='wrap')
384 if kernelVarray.shape[0]!=0: bitmap=convolve(bitmap,kernelVarray,mode='constant')
385 bitmap=bitmap/np.sum(bitmap)*self.Cols
386 self.rawBitmap=bitmap

388 def AutoAlign(self,
389 BERForAlignment=-3, # Exponent of probability contour to align on
390 AlignmentMode='Horizontal', # can be 'Horizontal' or 'Vertical'
391 HorizontalAlignment='Middle', # 'Middle' or 'Max' (vertical eye) - alignment will be the horizontal midpoint of one of these two eye possibilities
392 VerticalAlignment='MaxMin', # 'MaxMin' (maximum minimum opening) or 'Max' (maximum opening)
393 ):
394 """Automatically align the eye diagram
395 @param BERForAlignment float, defaults to -3, exponent of the probability contour to align on.
396 @param AlignmentMode string, either 'Horizontal' or 'Vertical', defaults to 'Horizontal', defining the alignment mode.
397 @param HorizontalAlignment string, either 'Middle' or 'Max' , defaults to 'Middle', defining the horizontal alignment mode.
398 @param VerticalAlignment string, either 'MaxMin' or 'Max', defaults to MaxMin, defining the vertical alignment mode.
399
400 Alignment is made by considering contours that define the probability specified by the BERForAlignment. While not technically
401 a BER, it is the probability contour inside of which, all hits are lower probability. All measurements of vertical height and
402 horizontal width are taken on these contours produced.
403 Alignment modes are a combination of the three alignment mode specifications:
404 | AlignmentMode | HorizontalAlignment | VerticalAlignment | Definition |
405 |:-------------:|:-------------------:|:-----------------:|:--------------------------------------------------------------------------------------------|
406 | Horizontal | Middle | N/A | The center of the eye is found by first computing the widest horizontal opening of the middle eye, then computing the horizontal midpoint.|
407 | Horizontal | Max | N/A | The center of the eye is found by first computing the widest horizontal opening of the widest eye, then computing the horizontal midpoint.|
408 | Vertical | N/A | MaxMin | The center of the eye is found by finding the horizontal location where the vertical opening is maximized for the narrowest eye. |
409 | Vertical | N/A | Max | The center of the eye is found by finding the horizontal location where the vertical opening is maximized for any eye. |
410 The side effect of this function is for the rawBitmap member variable to be replaced with a new bitmap containing the aligned eye diagram.
411 """

413 bitmap=self.rawBitmap.copy()

419 maxSat=1.
420

422 m=(maxSat-minSat)/(maxBER-minBER)
423 b=minSat-minBER*m
424 bitmapLog=np.array([[math.log10(max(bitmap[r][c],minValueLog)) for c in range(C)] for r in range(R)])
425 bitmapLog=bitmapLog*m+b
426 bitmapLog=np.array([[0. if bitmapLog[r][c] < minSat else (1 if bitmapLog[r][c] > maxSat else bitmapLog[r][c]) for c in range(C)] for r in range(R)])

433 counter=None

435 for r in range(1,R):

444

448 bestOpenings=None
449 indexOfBestOpening=None
450 bestMinimumOpening=None
451 indexOfBestMinimumOpening=None

455 continue

457 if bestMinimumOpening is None:
458 bestMinimumOpening=thisMinimumOpening
459 indexOfBestMinimumOpening=c
460 elif thisMinimumOpening > bestMinimumOpening:
461 bestMinimumOpening=thisMinimumOpening
462 indexOfBestMinimumOpening=c
463 if bestOpenings is None:
464 bestOpenings=np.flip(np.sort(dc))
465 indexOfBestOpening=c
466 else:
467 dc=np.flip(np.sort(dc))

471 bestOpenings=dc
472 indexOfBestOpening=c

474 elif dc[e]==bestOpenings[e]:

477 Continue=False
478 e+=1

486 rowOfMaxEyeWidth=None
487 startColumnofMaxEyeWidth=None
488 endColumnofMaxEyeWidth=None

491 eyeExtents=[self.eyeExtentsForAlignment[(self.Levels)//2-1]]
492 else: # 'Max', meaning horizontal midpoint of widest eye
493 eyeExtents=self.eyeExtentsForAlignment

495 for eye in eyeExtents:
496 for r in range(eye[0],eye[1]+1):
497 c=indexOfBestOpening
498 count=1 # count the pixel we're on
499 startColumn=c
500 endColumn=c
501 found=False
502 while count != C and not found:
503 c=(c-1)%C
504 if bitmapLog[r][c]!=0:
505 found=True
506 else:
507 count+=1
508 startColumn=c
509 found=False
510 c=(indexOfBestOpening+1)%C
511 while count != C and not found:
512 c=(c+1)%C
513 if bitmapLog[r][c]!=0:
514 found=True
515 else:
516 count+=1
517 endColumn=c

519 maxEyeWidth=count
520 rowOfMaxEyeWidth=r
521 startColumnofMaxEyeWidth=startColumn

531 else: # 'Max', meaning max vertical opening of all eyes

533

541 self.sampledWf=self.sampledWf.DelayBy((columnAtEyeCenter-C//2)/C/self.BaudRate)
542 self.sampledWf=self.sampledWf.DelayBy(1./self.BaudRate/2.)

547 DecisionMode='Mid',
548 WaveformType='V'
549 ):
550 """Computes measurements on the eye diagram
551 @param BERForMeasure float, defaults to -6, exponent of probability contour in eye to make measurements on.
552 @param DecisionMode string, 'Mid' or 'Best', defaults to 'Mid' to determine where the decision point is.
553 If 'Mid', the decision point is the vertical middle of the eye. If 'Best' is specified, it is the least likey
554 point in the middle of the eye. If the probability goes to zero in the middle, it is the midpoint of the middle
555 zero locations.
556 @param WaveformType string (optional, defaults to 'V') indicator of the type of waveform being measured.
557 Valid waveform types are:
558 * 'V' -- voltage, in volts
559 * 'A' -- current, in amperes
560 * 'W' -- power, in watts
561 * 'FW' -- fractional power, in fractions of watts
562 * 'FA' -- fractional current, in fractions of amperes
563 * 'FV' -- fractional voltage, in fractions of volts
564
565 Measurements are made by first determining a probablity contour in the eye such that everything inside the contour
566 is lower probability than the contour edge, which is above the probability of the BERForMeasure. The vertical extents
567 of the contour (the eye height) is the length of the vertical line in the exact center of the eye diagram. The horizontal
568 extents of the contour (the eye width) are the lengths of the horizontal line at the decision point, specified by the
569 DecisionMode.
570
571 Upon completion, the member variable measDict contains a dictionary with the following elements:
572 * R - Rows in the bitmap.
573 * C - Columns in the bitmap.
574 * MinV - Voltage defined by row 0.
575 * MaxV - Voltage defined by the maximum row.
576 * MinT - Time defined by the column 0.
577 * MaxT - Time defined by the last column.
578 * BERForMeasure - BER, or probability, for contour for all of the eye measurements.
579 * WaveformType - Type of waveform as either V, A, W, FW, AW, VW
580 * Eye - An array of parameters, one for each eye containing:
581 - Start - a dictionary containing Bin and Time defining the pixel column of the horizontal start of the eye.
582 - End - a dictionary containing Bin and Time defining the pixel column of the horizontal end of the eye.
583 - Width - a dictionary containing the Bin and Time width of the eye.
584 - High - a dictionary containing the Bin and Volt defining the pixel row of the vertical top of the eye.
585 - Low - a dictionary containing the Bin and Volt defining the pixel row of the vertical bottom of the eye.
586 - Mid - a dictionary containing the Bin and Volt defining the pixel row of the vertical middle of the eye.
587 - Best - a dictionary containing the Bin and Volt defining the pixel row of the best decision point for the eye.
588 - Height - a dictionary containing the Bin and Volt defining the vertical height of the eye.
589 - AV - a dictionary containing Volt defining the volts between the level extents at each edge of the eye.
590 * Level - An array of parameters, one for each level of the eye (the number of eyes plus one) containing:
591 - Min - a dictionary containing the Bin and Volt of the minimum edge of the level extent.
592 - Max - a dictionary containing the Bin and Volt of the maximum edge of the level extent.
593 - Delta - a dictionary containing the Bin and Volt vertical distance between the edeges of the level extent.
594 - Mean - a dictionary containing the Bin and Volt of the mean vertical location of the level extent.
595 * Linearity - The eye linearity calculated as the Min(AV)/Max(AV). This is 100% for NRZ waveforms.
596 * RLM - Relative Level Mismatch - only defined for PAM-4.
597 * RMS - Rms value of the waveform
598 * Noise - Rms value of the noise
599 * NoiseResidual - Rms value of the residual noise (ISI and linearity only - no noise added).
600 * SDR - Ratio of sampled ideal signal to residual error (noise) in dB (without noise added).
601 * SNR - Signal to noise ratio.
602 * SNDR - Signal to noise and distortion ratio.
603 * VerticalResolution - the resolution, in V, for each row of the eye.
604 * HorizontalResolution - the resolution in seconds, for each column of the eye.
605 """
606 from SignalIntegrity.Lib.TimeDomain.Waveform import Waveform
607 bitmap=self.rawBitmap.copy()
608 numberOfEyes=int(self.Levels-1)
609 (R,C)=bitmap.shape
610 minValueLog=pow(10.,-20)
611 minBER=BERForMeasure; maxBER=BERForMeasure+0.001
612 m=1./(maxBER-minBER)
613 b=0-minBER*m
614 bitmapLog=np.array([[math.log10(max(bitmap[r][c],minValueLog)) for c in range(C)] for r in range(R)])
615 bitmapLog=bitmapLog*m+b
616 bitmapLog=np.array([[0. if bitmapLog[r][c] < 0 else (1 if bitmapLog[r][c] > 1 else bitmapLog[r][c]) for c in range(C)] for r in range(R)])
617
618 # develop the extents of the eye openings for each eye in a column (vertical slice)
619 # The assumption is that there is signal or the exterior in the first row. Stepping into
620 # a dark area from a non-dark area starts an extent and the stepping from a dark area to a
621 # non-dark area ends an extent.
622

624 counter=None
625 darkExtents=[]
626 for r in range(1,R):
627 if bitmapLog[r][c]==0. and bitmapLog[r-1][c]!=0.:
628 counter=r
629 elif bitmapLog[r][c]!=0 and bitmapLog[r-1][c]==0:

633 # now, for each eye, walk backwards and forwards at each row slice within these extents
634 # to find the extents of each eye opening
635 deltaV=self.maxV-self.minV
636 eyeHeights=[(f-e)/R*deltaV for (e,f) in eyeVerticalExtentsForMeasurement]
637 eyeMidBins=[int(round((f+e)/2)) for (e,f) in eyeVerticalExtentsForMeasurement]
638 eyeMids=[b/R*deltaV+self.minV for b in eyeMidBins]
639 # find the best decision points based on probability
640 eyeDecisionPoints=[]
641 for extent in eyeVerticalExtentsForMeasurement:
642 minIndex=None
643 minValue=None
644 count=0
645 for r in range(extent[0],extent[1]+1):
646 value=self.rawBitmap[r][c]
647 if (minValue == None) or (value<minValue):
648 minValue=value
649 minIndex=r
650 count=1
651 elif value==0:
652 count+=1
653 eyeDecisionPoints.append(int(minIndex+count//2))
654 # the threshold extents are based off the eye extents, except for the bottom and top
655 thresholdExtents=[(None if t==0 else eyeVerticalExtentsForMeasurement[t-1][1]+1,
656 None if t==numberOfEyes else eyeVerticalExtentsForMeasurement[t][0]-1)
657 for t in range(numberOfEyes+1)]
658 # fill in the bottom threshold extent
659 for r in range(R):
660 if bitmapLog[r][c] > 0:
661 thresholdExtents[0]=(r,thresholdExtents[0][1])
662 break
663 # fill in the top threshold extent
664 for r in range(R):
665 if bitmapLog[R-1-r][c] > 0:
666 thresholdExtents[-1]=(thresholdExtents[-1][0],R-1-r)
667 break
668 eyeAverageLevels=[]
669 acc=0
670 pacc=0
671 L=0
672 for r in range(R+1):
673 if ((L < len(eyeMidBins)) and (r < eyeMidBins[L])) or ((L >= len(eyeMidBins)) and (r < R)):
674 pacc+=self.rawBitmap[r][c]
675 acc+=self.rawBitmap[r][c]*(r/R*deltaV+self.minV)
676 else:
677 eyeAverageLevels.append(acc/pacc)
678 acc=0; pacc=0; L+=1
679 AV=[eyeAverageLevels[l+1]-eyeAverageLevels[l] for l in range(len(eyeAverageLevels)-1)]
680 EyeLinearity=min(AV)/max(AV)
681 if self.Levels == 4:
682 (V0,V1,V2,V3)=(eyeAverageLevels[0],eyeAverageLevels[1],eyeAverageLevels[2],eyeAverageLevels[3])
683 Vmid=(V0+V3)/2
684 ES1=(V1-Vmid)/(V0-Vmid)
685 ES2=(V2-Vmid)/(V3-Vmid)
686 RLM=min(3.*ES1,3.*ES2,2.-3.*ES1,2.-3.*ES2)
687 else:
688 RLM=None
689 eyeHorizontalExtentsForMeasure=[]
690 eyeWidths=[]
691 UI=1./self.BaudRate
692 for i in range(len(eyeVerticalExtentsForMeasurement)):
693 decision=eyeDecisionPoints[i]
694 eye=eyeVerticalExtentsForMeasurement[i]
695 r=(eye[0]+eye[1])//2 if DecisionMode == 'Mid' else decision
696 c=C//2
697 count=1 # count the pixel we're on
698 startColumn=c
699 endColumn=c
700 found=False
701 while count != C and not found:
702 c=(c-1)%C
703 if bitmapLog[r][c]!=0:
704 found=True
705 else:
706 count+=1
707 startColumn=c
708 found=False
709 c=C//2
710 while count != C and not found:
711 c=(c+1)%C
712 if bitmapLog[r][c]!=0:
713 found=True
714 else:
715 count+=1
716 endColumn=c
717 if (count < C):
718 eyeHorizontalExtentsForMeasure.append((startColumn,endColumn))
719 eyeWidths.append(count/C*UI)
720 def precision_round(number, digits=3):
721 if number == None: return number
722 power = "{:e}".format(number).split('e')[1]
723 return round(number, -(int(power) - digits))
724 # calculate the SDR -- this would be the power of the ideal sampled signal with respect to the power of the residual noise
725 levels = [eyeAverageLevels[i] for i in range(numberOfEyes+1)]
726 noiseWf=Waveform(self.sampledWf.td)
727 signalWf=Waveform(self.sampledWf.td)
728 for k in range(len(self.sampledWf)):
729 x=self.sampledWf[k]
730 distance=[abs(level-x) for level in levels] # list of minimum distances from levels
731 l=distance.index(min(distance)) # index of the ideal level
732 noiseWf[k]=x-levels[l] # error
733 signalWf[k]=levels[l] # signal (ideal)
734 # calculate the power of each
735 mean=1./len(signalWf)*sum(signalWf)
736 signalRms=math.sqrt(1./len(signalWf)*sum([(x-mean)**2 for x in signalWf]))
737 noiseResidual=math.sqrt(1./len(noiseWf)*sum([x**2 for x in noiseWf]))
738 sdr=20.*math.log10(signalRms/noiseResidual)
739 if self.NoiseSigma==0.:
740 snr=None
741 else:
742 snr=20*math.log10(signalRms/self.NoiseSigma)
743 sndr=20.*math.log10(signalRms/math.sqrt(noiseResidual**2+self.NoiseSigma**2))
744 self.measDict={'R':R,'C':C,'MinV':self.minV,'MaxV':self.maxV,'MinT':-C//2*UI,'MaxT':(C-1)//2*UI,'BERForMeasure':BERForMeasure,'WaveformType':WaveformType,
745 'Eye':{i:{'Start':{'Bin':eyeHorizontalExtentsForMeasure[i][0],
746 'Time':precision_round((eyeHorizontalExtentsForMeasure[i][0]-C/2)/C*UI)},
747 'End':{'Bin':eyeHorizontalExtentsForMeasure[i][1],
748 'Time':precision_round((eyeHorizontalExtentsForMeasure[i][1]-C/2)/C*UI)},
749 'Width':{'Bin':(eyeHorizontalExtentsForMeasure[i][1]-eyeHorizontalExtentsForMeasure[i][0]+1),
750 'Time':precision_round(eyeWidths[i])},
751 'Low':{'Bin':eyeVerticalExtentsForMeasurement[i][0],
752 'Value':precision_round(eyeVerticalExtentsForMeasurement[i][0]/R*deltaV+self.minV)},
753 'High':{'Bin':eyeVerticalExtentsForMeasurement[i][1],
754 'Value':precision_round(eyeVerticalExtentsForMeasurement[i][1]/R*deltaV+self.minV)},
755 'Mid':{'Bin':int((eyeVerticalExtentsForMeasurement[i][0]+eyeVerticalExtentsForMeasurement[i][1])/2),
756 'Value':precision_round(eyeMids[i])},
757 'Best':{'Bin':eyeDecisionPoints[i],
758 'Value':precision_round(eyeDecisionPoints[i]/R*deltaV+self.minV)},
759 'Height':{'Bin':(eyeVerticalExtentsForMeasurement[i][1]-eyeVerticalExtentsForMeasurement[i][0]+1),
760 'Value':precision_round(eyeHeights[i])},
761 'AV':{'Value':precision_round(AV[i])}} for i in range(numberOfEyes)},
762 'Level':{i:{'Min':{'Bin':thresholdExtents[i][0],
763 'Value':precision_round(thresholdExtents[i][0]/R*deltaV+self.minV)},
764 'Max':{'Bin':thresholdExtents[i][1],
765 'Value':precision_round(thresholdExtents[i][1]/R*deltaV+self.minV)},
766 'Delta':{'Bin':(thresholdExtents[i][1]-thresholdExtents[i][0]),
767 'Value':precision_round((thresholdExtents[i][1]-thresholdExtents[i][0])/R*deltaV)},
768 'Mean':{'Bin':int((eyeAverageLevels[i]-self.minV)/deltaV*R+0.5),
769 'Value':precision_round(eyeAverageLevels[i])}} for i in range(numberOfEyes+1)},
770 'Linearity':precision_round(EyeLinearity*100.)/100.,
771 'RLM':precision_round(RLM),
772 'RMS':precision_round(signalRms),
773 'Noise':precision_round(self.NoiseSigma),
774 'NoiseResidual':precision_round(noiseResidual),
775 'SDR':precision_round(sdr),
776 'SNR':precision_round(snr),
777 'SNDR':precision_round(sndr),
778 'VerticalResolution':precision_round((self.maxV-self.minV)/R),
779 'HorizontalResolution':precision_round(UI/C)}
780 for i in range(numberOfEyes):
781 self.measDict['Eye'][i]['Decision']=self.measDict['Eye'][i]['Mid'] if DecisionMode=='Mid' else self.measDict['Eye'][i]['Best']
782 self.BitmapLog = bitmapLog

786 """Computes optical measurements on the eye diagram
787 @param WaveformType string definition of waveform type. Can be 'W', 'FW', 'AW', 'VW', which stands for Power, Fractional power, Current proportional
788 to power, and Voltage proportional to power. If 'V' or 'A' is specified, for voltage and current waveforms, this function exits with no results.
789
790 @param PinW float, optional defaults to None, input power in Watts, if known and applicable (i.e. is a transmitter measurement)
791
792 Computes optical measurements assuming an input waveform. The most straightforward is for waveforms in Watts.
793
794 Upon completion, the 'Optical' field in the member variable measDict contains a dictionary with the following elements:
795 * PL - low level.
796 - W - value in Watts.
797 - dBm - value in dBm.
798 * PH - High level.
799 - W - value in Watts.
800 - dBm - value in dBm.
801 * OMA - Optical modulation amplitude.
802 - W - value in Watts.
803 - dBm - value in dBm.
804 * Pavg - Average power.
805 - W - value in Watts.
806 - dBm - value in dBm.
807 * ER - Extinction ration.
808 - Linear - value as ratio.
809 - dB - value in dB.
810 (if input power supplied):
811 * Pin - Input power.
812 - W - value in Watts.
813 - dBm - value in dBm.
814 * TP - Transmission penalty.
815 - Linear - value as ratio.
816 - dB - value in dB.
817 * IL - Insertion loss
818 - Linear - value as ratio.
819 -dB - value in dB.
820 """
821 def W2dBm(W): return 10.*math.log10(W/1e-3)
822 def dBm2W(dBm): return 10.**(dBm/10.)*1e-3
823 def dB2Lin(dB): return 10**(dB/10.)
824 def Lin2dB(L): return 10*math.log10(L)
825
826 if not WaveformType in ['W','FW','AW','VW']:
827 del self.measDict['Optical']
828 return
829
830 verticalUnit={'W':'W','FW':'','AW':'A','VW':'V'}[WaveformType]
831 dBmUnit = {'W':'dBm','FW':'dB','AW':None,'VW':None}[WaveformType]
832
833 if WaveformType != 'W': PinW = None
834
835 def W2dBmUnit(W):
836 if W==None: return None
837 elif dBmUnit=='dBm': return W2dBm(W)
838 elif dBmUnit=='dB': return Lin2dB(W)
839 else: return None
840
841 PLWraw=self.measDict['Level'][0]['Mean']['Value']
842 PHWraw=self.measDict['Level'][len(self.measDict['Level'])-1]['Mean']['Value']
843 if WaveformType in ['AW','VW']:
844 PLWraw=abs(self.measDict['Level'][0]['Mean']['Value'])
845 PHWraw=abs(self.measDict['Level'][len(self.measDict['Level'])-1]['Mean']['Value'])
846 PHW=max(PLWraw,PHWraw)
847 PLW=min(PLWraw,PHWraw)
848 else:
849 PHW=PHWraw
850 PLW=PLWraw
851
852 PLdBm=W2dBmUnit(PLW)
853 PHdBm=W2dBmUnit(PHW)
854 OMA=PHW-PLW
855 OMAdBm=W2dBmUnit(OMA)
856
857 PavgW=np.mean(self.prbswf)
858 if WaveformType in ['AW','VW']:
859 PavgW=abs(PavgW)
860
861 PavgdBm=W2dBmUnit(PavgW)
862 ER=PHW/PLW
863 ERdB=Lin2dB(ER)
864
865 self.measDict['Optical']={'PL':{'Linear':{'Unit':verticalUnit,'Value':PLW},'Log':{'Unit':dBmUnit,'Value':PLdBm}},
866 'PH':{'Linear':{'Unit':verticalUnit,'Value':PHW},'Log':{'Unit':dBmUnit,'Value':PHdBm}},
867 'OMA':{'Linear':{'Unit':verticalUnit,'Value':OMA},'Log':{'Unit':dBmUnit,'Value':OMAdBm}},
868 'Pavg':{'Linear':{'Unit':verticalUnit,'Value':PavgW},'Log':{'Unit':dBmUnit,'Value':PavgdBm}},
869 'ER':{'Linear':{'Unit':'','Value':ER},'Log':{'Unit':'dB','Value':ERdB}}}
870
871 if WaveformType == 'FW':
872 PinW = 1.
873
874 if PinW != None:
875 PindBm=W2dBmUnit(PinW)
876 TP=PinW/OMA
877 TPdB=Lin2dB(TP)
878 ILdB=PindBm-PHdBm
879 IL=dB2Lin(ILdB)
880 LossdB=PindBm-PavgdBm
881 LossW=dB2Lin(LossdB)
882 self.measDict['Optical'].update({'Pin':{'Linear':{'Unit':verticalUnit,'Value':PinW},'Log':{'Unit':dBmUnit,'Value':PindBm}},
883 'TP':{'Linear':{'Unit':'','Value':TP},'Log':{'Unit':'dB','Value':TPdB}},
884 'IL':{'Linear':{'Unit':'','Value':IL},'Log':{'Unit':'dB','Value':ILdB}},
885 'Loss':{'Linear':{'Unit':'','Value':LossW},'Log':{'Unit':'dB','Value':LossdB}}})
886
887 numLevels=self.Levels
888 numEyes=numLevels-1
889 numInnerLevels=numLevels-2
890 numOuterLevels=numLevels-numInnerLevels
891 numErrorCases=numOuterLevels+2*numInnerLevels
892 # expected BER is 1/BPS * 1/levels * cases * 1/2 * erfc( outer oma / eyes / sqrt(2) / 2 / sigma of noise )
893 # for NRZ this is 1/1 * 1/2 * 2 * 1/2 *erfc( OMA/1/sqrt(2)/2/sigma) = 1/2 * erfc( OMA/sqrt(2)/2/sigma)
894 # for PAM-4 this is 1/2 * 1/4 * 6 * 1/2 * erfc( OMA/3/sqrt(2)/2/sigma) = 3/8 * erfc( OMA/6/sqrt(2)/sigma )
895 # for PAM-8 this is 1/3 * 1/8 * 14 *1/2 * erfc( OMA/7/sqrt(2)/2/sigma ) = 7/24 * erfc( OMA/14/sqrt(2)/sigma )
896 # we simplify this to F * erfc( 1./sqrt(2) * OMA/2/D/sigma )
897 # where OMA/2/D/sigma is effectively the SNR
898 F= 1./self.BitsPerSymbol * 1./numLevels * numErrorCases * 1./2
899 (FN,FD)=Rat(F)
900 D = numEyes
901
902 if 'Probabilities' in self.measDict.keys():
903 import scipy.special
904 BERmeas=self.measDict['Probabilities']['ErrorRate']['Bit']['Gray']['Measured']
905 QFactor=min(scipy.special.erfcinv(BERmeas/F)*math.sqrt(2),1e9)
906 QFactordB=Lin2dB(QFactor)
907
908 self.measDict['Optical']['Q']={'QFactor':QFactor,'QFactordB':QFactordB,'BERMeasured':BERmeas,
909 'Levels':{'Total':numLevels,'Inner':numInnerLevels,'Outer':numOuterLevels},
910 'Eyes':numEyes,'ErrorCases':numErrorCases,
911 'F':{'Numerical':F,'Numerator':FN,'Denominator':FD},'D':D}
912 if self.NoiseSigma > 0.:
913 QFactorExpected=OMA/2/D/self.NoiseSigma
914 BERexpected=F*scipy.special.erfc(1./math.sqrt(2.)*QFactorExpected)
915 QFactorExpecteddB=Lin2dB(QFactorExpected)
916 Penalty=QFactorExpecteddB-QFactordB
917 self.measDict['Optical']['Q'].update({'QFactorExpected':QFactorExpected,'QFactorExpecteddB':QFactorExpecteddB,
918 'BERExpected':BERexpected,'TxPenalty':Penalty})

920 def Bathtub(self,
921 DecadesFromJoin=0.25,

923 ):
924 """Generates bathtub curves on the eye diagram and calculates the error probabilities and rates
925 @param DecadesFromJoin float, defaults to 0.25, number of decades above the point where the PDFs join together to begin fitting
926 the histogram tails. A bit further up, like 0.5-1 decade is ideal, but will cause algorithm failure for very low BER systems.

928 @param MinPointsForFit integer, defaults to 6, minimum number of points to use to fit the tail of the PDF.
929 After processing, the measDict member variable has an additional dictionary added called 'Bathtub' and 'Probabilities'. Inside
930 the 'Bathtub' dictionary are the following items:
931 Vertical - a dictionary containing:
932 * Data - contains two vectors called:
933 - x - an x axis containing one element per row containing the voltage.
934 - y - a y axis containing one element per row containing the probability.
935 * Level - a dictionary array (each element in the dictionary is 0, 1, 2, etc.) containing, per level:
936 - LeftEst - a dictionary containing information on the left estimate (the PDF tail below the level in the histogram shown as a vertical slice:
937 - Start - a dictionary containing Bin which is the starting row for the estimate of the PDF.
938 - End - a dictionary containing Bin which is the ending row for the estimate of the PDF.
939 - Est - a dictionary of the estimatated polynomial:
940 - Coef - list of the three coefficients a0, a1, a2 for the polynomial a0+a1*x+a2*x^2 where x is the natural log of the probability.
941 - Valid - if the estimate is valid.
942 - Wf - a dictionary of the estimated PDF tail where:
943 - x - is the x axis containing the voltage.
944 - y - is the y axis containing the probability.
945 - RightEst - a dictionary containing information on the right estimate (the PDF tail above the level in the histogram shown as a vertical slice:
946 - Start - a dictionary containing Bin which is the starting row for the estimate of the PDF.
947 - End - a dictionary containing Bin which is the ending row for the estimate of the PDF.
948 - Est - a dictionary of the estimatated polynomial:
949 - Coef - list of the three coefficients a0, a1, a2 for the polynomial a0+a1*x+a2*x^2 where x is the natural log of the probability.
950 - Valid - if the estimate is valid.
951 - Wf - a dictionary of the estimated PDF tail where:
952 - x - is the x axis containing the voltage.
953 - y - is the y axis containing the probability.
954 - Hist - the entire, partially estimated, PDF for this level vertically across the entire vertical slice. There is one probablity element
955 per row in the eye diagram.
956 - CDFFromLeft - The integrated PDF from the bottom to the top, forming the probability that a value could be below a given row.
957 - CDFFromRight - The integrated PDF from the top to the bottom, forming the probability that a value could be above a given row.
958 * Horizontal - a dictionary array, one element per eye, where each element in the dictionary is 0, 1, 2, etc.), containing, per eye:
959 - Data - an instance of class Waveform, spanning the entire UI horizontally, containing the probability at the decision point for each column.
960 This is used to plot the jitter bathtub curves.
961 Inside the 'Probabilities' dictionary are the following items:
962 * SymbolCodes - a list of the codes for the symbol. This is just 0, 1, 2, etc. depending on the bits per symbol.
963 * GrayCodes - a list of gray coded symbols corresponding to the symbol codes.
964 * Interpretation - A list of lists representing a Symbols x Symbols matrix, where Interpretation[x][y] represents the probability that a
965 symbol transmitted as symbol x is interpreted as symbol y.
966 * Symbol - the measured probability that a symbol was transmitted.
967 * ErrorRate - The error rates containing:
968 - Symbol - the symbol error rates as:
969 - PerSymbol - A list per symbol containing the probability of a symbol being erroneously decoded.
970 - Nominal - The total, nominal symbol error rate, assuming that each symbol was transmitted with equal probability.
971 - Measured - The total, measured symbol error rate, considering the measured probability that a given symbol was transmitted.
972 - Bit - The bit error rates containing:
973 - Standard - The un-gray coded bit errors:
974 - PerSymbol - a list per symbol containing the probability of one or more bit errors in the symbol.
975 - Nominal - The total, nominal bit error rate, assuming that each bit was transmitted with equal probability.
976 - Measured - the total, measured bit error rate, considering the measured probability that a given bit was transmitted.
977 - Gray - The gray coded bit errors:
978 - PerSymbol - a list per symbol containing the probability of one or more bit errors in the symbol.
979 - Nominal - The total, nominal bit error rate, assuming that each bit was transmitted with equal probability. Nominally, this should
980 be the nominal BER divided by the number of bits per symbol.
981 - Measured - the total, measured bit error rate, considering the measured probability that a given bit was transmitted.
982 @remark The Measure member function be called prior to this call.
983 """
984 from SignalIntegrity.Lib.TimeDomain.Waveform import TimeDescriptor,Waveform
985 numberOfEyes=int(self.Levels-1)
986 UI=1./self.BaudRate
987 (R,C)=self.rawBitmap.shape
988 # build bathtub curve waveforms
989 # one long vertical waveform for a bathtub curve
990 c=C//2
991 # a horizontal bathtub curve waveform for each eye
992 x=[self.minV+r/R*(self.maxV-self.minV) for r in range(R)]
993 y=[self.rawBitmap[r][c] for r in range(R)]
994 self.measDict['Bathtub']={'Vertical':{'Data':{'x':x,'y':y}},
995 'Horizontal':{}}
996
