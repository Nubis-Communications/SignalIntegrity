25class PulseWaveform(Waveform):
26 """pulse waveform"""
27 def __init__(self,td,Amplitude=1.,StartTime=0.,PulseWidth=0.,Risetime=0.):
28 """Constructor
29 constructs a waveform with a single pulse.
30 @param td instance of class TimeDescriptor containing time axis of waveform.
31 @param Amplitude (optional) float amplitude of pulse (defaults to unity).
32 @param StartTime (optional) float starting time of the pulse (defaults to zero).
33 @param PulseWidth (optional) float the width of the pulse (defaults to zero).
34 @param Risetime (optional) float risetime in seconds (defaults to 0.)
35 @note The amplitude can be positive or negative, with negative providing a negative
36 pulse.
37 @note if the pulse appears entirely within the samples, then the waveform will be all zero.
38 @note actual risetime used is the sample period if less than that is specified
39 @note risetime is applied on the rising and falling edge. risetimes greater than
40 1.72 times the pulse width will diminish the pulse amplitude.
41 @note the risetime is applied such that the pulse reaches half amplitude at the start
42 time specified. Please note the expected non-causality.
43 """
44 StopTime=StartTime+PulseWidth
45 stepup=StepWaveform(td,Amplitude,StartTime,Risetime)
46 stepdown=StepWaveform(td,Amplitude,StopTime,Risetime)
47 Waveform.__init__(self,td,[stepup[k]-stepdown[k] for k in range(len(stepup))])
