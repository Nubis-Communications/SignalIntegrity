53 def Values(self,unit=None):
54 """Values
55 @param unit (optional) string containing the unit for the frequencies
56 @return list of complex values corresponding to the frequency-domain elements in the
57 units specified.
58 @remark
59 Valid unit strings are:
60 - 'dB' - values in decibels.
61 - 'mag' - values in absolute magnitude.
62 - 'rad' - the argument or phase in radians.
63 - 'deg' - the argument or phase in degrees.
64 - 'real' - the real part of the values.
65 - 'imag' - the imaginary part of the values.
66
67 Returns the list of complex values if no unit specified.
68
69 Returns None if the unit is invalid.
70 """
71 if unit==None:
72 return list(self)
73 elif unit =='dB':
74 return [-3000. if (abs(self[n]) < 1e-15) else
75 20.*math.log10(abs(self[n]))
76 for n in range(len(self.m_f))]
77 elif unit == 'mag':
78 return [abs(self[n]) for n in range(len(self.m_f))]
79 elif unit == 'rad':
80 return [cmath.phase(self[n]) for n in range(len(self.m_f))]
81 elif unit == 'deg':
82 return [cmath.phase(self[n])*180./math.pi
83 for n in range(len(self.m_f))]
84 elif unit == 'real':
85 return [self[n].real for n in range(len(self.m_f))]
86 elif unit == 'imag':
87 return [self[n].imag for n in range(len(self.m_f))]
88 def ReadFromLines(self,data):
89 """reads in frequency domain content from the lines specified.
90 @param data list of lines to read from
91 @return self
92 """
93 if data is None or len(data) == 0:
94 FrequencyDomain.__init__(self)
95 return self
96 if data[0].strip('\n')!='UnevenlySpaced':
97 N = int(str(data[0]))
98 Fe = float(str(data[1]))
99 frl=[line.strip().split(' ') for line in data[2:]]
100 resp=[float(fr[0])+1j*float(fr[1]) for fr in frl]
101 self.m_f=EvenlySpacedFrequencyList(Fe,N)
102 list.__init__(self,resp)
103 else:
104 frl=[line.split(' ') for line in data[1:]]
105 f=[float(fr[0]) for fr in frl]
106 resp=[float(fr[1])+1j*float(fr[2]) for fr in frl]
107 self.m_f=GenericFrequencyList(f)
108 list.__init__(self,resp)
109 return self

126 def WriteToFileStream(self,f):
127 """write the frequency domain content to the file stream specified.
128 @param f file stream to write to
129 @return self
130 """
131 fl=self.FrequencyList()
132 if fl.CheckEvenlySpaced():
133 f.write(str(fl.N)+'\n')
134 f.write(str(fl.Fe)+'\n')
135 for v in self.Values():
136 f.write(str(v.real)+' '+str(v.imag)+'\n')
137 else:
138 f.write('UnevenlySpaced\n')
139 for n in range(len(fl)):
140 f.write(str(fl[n])+' '+str(self.Values()[n].real)+' '+
141 str(self.Values()[n].imag)+'\n')
142 return self

170 def LimitEndFrequency(self,endFrequency):
171 """limits the end frequency
172 @param endFrequency float end frequency to limit to
173 @return self
174 @remark if the end frequency is higher than the current end frequency,
175 the content is left unchanged.
176 @warning the end frequency might be slightly higher and this is not
177 a strict limit. The goal is for the frequencies to potentially bracket
178 the desired end frequency.
179 """
180 frequencies=self.Frequencies()
181 deltaf=frequencies[-1]/(len(self)-1)
182 numPts=int(math.ceil(endFrequency/deltaf))
183 if numPts >= len(self):
184 return self
185 fl=FrequencyList(frequencies[0:numPts+1])
186 fl.CheckEvenlySpaced()
187 FrequencyDomain.__init__(self,fl,self[0:numPts+1])
188 return self
