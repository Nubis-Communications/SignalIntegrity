24def IdealTransformer(a=1):
25 """symbolic ideal transformer
26
27 Ports 1 and 2 are the primary.
28
29 Ports 3 and 4 are the secondary.
30
31 The dot is on ports 1 and 3.
32
33 a is the turns ratio specified as (secondary/primary) windings
34
35 @param a integer, float or string turns ratio\n
36 @return list of list of string s-parameter matrix
37 containing LaTeX or ASCII strings for each element.
38 @note strings can be any valid LaTeX
39 @note this is the symbolic version of SignalIntegrity.Lib.Devices.IdealTransformer
40 """
41 if isinstance(a,str):
42 try:
43 a=int(a)
44 except ValueError:
45 try:
46 a=float(a)
47 except ValueError:
48 pass
49 if isinstance(a,int) or isinstance(a,float):
50 if a==1:
51 a='1'
52 asq='1'
53 denom='2'
54 else:
55 asq=str(a*a)
56 denom=str(a*a+1)
57 a=str(a)
58 elif isinstance(a,str):
59 asq=a+'^2'
60 denom=asq+' + 1'
61 one=' '+lfrac(' 1 ',denom)+' '
62 a2=' '+lfrac(' '+asq+' ',' '+denom+' ')+' '
63 a1=' '+lfrac(' '+a+' ',' '+denom+' ')+' '
64 na=' -'+lfrac(' '+a+' ',' '+denom+' ')+' '
65 return [[one,a2,a1,na],[a2,one,na,a1],[a1,na,a2,one],[na,a1,one,a2]]
