77 def RawMeasuredSParameters(self,wfList):
78 """RawMeasuredSParameters
79 @param wfList list of lists of waveforms of measurements.
80 @return instance of class SParameters containing the raw measured
81 s-parameters of the DUT.
82
83 Each element at index d in wfList is a list of waveforms where
84 the zero based dth waveform contains the incident and the others don't.
85 (i.e. they are the waveforms measured at the ports of the DUT when zero
86 based port d is driven).
87
88 For a P port DUT, wfList must be PxP.
89 @note for one port measurements, wfList can be a single measurement (as opposed
90 to a 1x1 list of list of one element).
91 """
92 # pragma: silent exclude
93 wfList=copy.deepcopy(wfList)
94 if isinstance(wfList, Waveform):
95 wfList=[wfList]
96 # pragma: silent include
97 ports=len(wfList)
98 S=[[None for _ in range(ports)] for _ in range(ports)]
99 for d in range(ports):
100 # pragma: silent exclude
101 if isinstance(wfList[d],Waveform):
102 wfList[d]=[wfList[d]]
103 # pragma: include
104 fc=self.Convert(wfList[d],d)
105 for o in range(ports):
106 S[o][d]=fc[o]
107 f=S[0][0].Frequencies()
108 return SParameters(f,
109 [[[S[r][c][n] for c in range(ports)] for r in range(ports)]
110 for n in range(len(f))])
111 def Convert(self,wfListProvided,incidentIndex=0):
112 # pragma: silent exclude
113 wfList=copy.deepcopy(wfListProvided)
114 if isinstance(wfList, Waveform):
115 wfList=[wfList]
116 # pragma: silent include
117 if self.step:
118 wfList=[wf.Derivative(removePoint=False,scale=False)
119 for wf in wfList]
120 if self.inverted:
121 wfList=[wf*-1. for wf in wfList]
122 # pragma: silent exclude
123 self.derivatives=copy.deepcopy(wfList)
124 # pragma: silent include
125 if self.denoise:
126 wfList=[WaveletDenoiser.DenoisedWaveform(
127 wf,isDerivative=self.step,
128 mult=self.sigmaMultiple,
129 pct=self.denoisePercent)
130 for wf in wfList]
131 # pragma: silent exclude
132 self.denoisedDerivatives=copy.deepcopy(wfList)
133 # pragma: silent include
134 if self.length!=0:
135 lengthSamples=int(self.length*
136 wfList[incidentIndex].td.Fs+0.5)
137 wfList=[wf*WaveformTrimmer(0,wf.td.K-lengthSamples)
138 for wf in wfList]
139 # pragma: silent exclude
140 self.TrimmedDenoisedDerivatives=copy.deepcopy(wfList)
141 # pragma: silent include
142 incwf=copy.deepcopy(wfList[incidentIndex])
143 # pragma: silent exclude
144 self.TrimmedDenoisedIncidentDesignated=copy.deepcopy(incwf)
145 # pragma: silent include
146 maxValueIndex=0
147 maxValue=incwf[0]
148 for k in range(1,len(incwf)):
149 if incwf[k]>maxValue:
150 maxValue=incwf[k]
151 maxValueIndex=k
152 forwardSamples=int(self.wfhwt*incwf.td.Fs)
153 reverseSamples=int(self.wrhwt*incwf.td.Fs)
154 raisedCosineSamples=int(self.wrcdr*incwf.td.Fs)
155 (incidentExtractionWindow,reflectExtractionWindow)=self._ExtractionWindows(
156 incwf.td,forwardSamples,reverseSamples,raisedCosineSamples,maxValueIndex)
157 # pragma: silent exclude
158 self.IncidentExtractionWindow=copy.deepcopy(incidentExtractionWindow)
159 self.ReflectExtractionWindow=copy.deepcopy(reflectExtractionWindow)
160 # pragma: silent include
161 incwf=Waveform(incwf.td,[x*w
162 for (x,w) in zip(incwf.Values(),incidentExtractionWindow.Values())])
163 wfList=[Waveform(wf.td,[x*w
164 for (x,w) in zip(wf.Values(),reflectExtractionWindow.Values())])
165 for wf in wfList]
166 #wfList[incidentIndex]=wfList[incidentIndex]-incwf
167 # pragma: silent exclude
168 self.IncidentWaveform=copy.deepcopy(incwf)
169 self.ReflectWaveforms=copy.deepcopy(wfList)
170 # pragma: silent include
171 incwffc=incwf.FrequencyContent(self.fd)
172 res=[wf.FrequencyContent(self.fd) for wf in wfList]
173 # pragma: silent exclude
174 self.IncidentFrequencyContent=copy.deepcopy(incwffc)
175 self.ReflectFrequencyContent=copy.deepcopy(res)
176 # pragma: silent include
177 for fc in res:
178 for n in range(len(fc)):
179 fc[n]=fc[n]/incwffc[n]
180 # pragma: silent exclude
181 if (len(res)==1) and (isinstance(wfListProvided,Waveform)):
182 res=res[0]
183 # pragma: include
184 return res
185 def _ExtractionWindows(self,td,forwardSamples,reverseSamples,
186 raisedCosineSamples,maxValueIndex):
187 incidentExtractionWindow=Waveform(td)
188 reflectExtractionWindow=Waveform(td)
189 for k in range(td.K):
190 if k<maxValueIndex-reverseSamples-raisedCosineSamples:
191 incidentExtractionWindow[k]=0.0
192 reflectExtractionWindow[k]=0.0
193 elif k<maxValueIndex-reverseSamples and raisedCosineSamples>0:
194 si=k-(maxValueIndex-reverseSamples)
195 f=float(si)/raisedCosineSamples
196 incidentExtractionWindow[k]=(math.cos(f*math.pi)+1.)/2.
197 reflectExtractionWindow[k]=0.0
198 elif k<=maxValueIndex+forwardSamples:
199 incidentExtractionWindow[k]=1.0
200 reflectExtractionWindow[k]=0.0
201 elif k<=maxValueIndex+forwardSamples+raisedCosineSamples and raisedCosineSamples>0:
202 si=k-(maxValueIndex+forwardSamples)
203 f=float(si)/raisedCosineSamples
204 incidentExtractionWindow[k]=(math.cos(f*math.pi)+1.)/2.
205 reflectExtractionWindow[k]=1.-incidentExtractionWindow[k]
206 elif k>=td.K-1-raisedCosineSamples and raisedCosineSamples>0:
207 si=k-(td.K-1-raisedCosineSamples)
208 f=float(si)/raisedCosineSamples
209 incidentExtractionWindow[k]=0.
210 reflectExtractionWindow[k]=(math.cos(f*math.pi)+1.)/2.
211 else:
212 incidentExtractionWindow[k]=0.
213 reflectExtractionWindow[k]=1.
214 if not self.taper:
215 reflectExtractionWindow=Waveform(td,[1.0-v for v in incidentExtractionWindow])
216 return (incidentExtractionWindow,reflectExtractionWindow)
