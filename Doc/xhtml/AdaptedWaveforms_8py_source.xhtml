36 strategy=wfl[0].adaptionStrategy
37
38 #upsample all of the waveforms first
39 # udfl is a list of (U,D) tuples containing upsample and decimation factor
40 # todo: this change was made to prevent a failure, presumably when decimation was needed (and would
41 # create an integer upsample factor of 0). Now, for example, it might have an upsample factor of 1 and
42 # a downsample factor of 2, instead of 0 for the integer upsample factor, which caused failure. But, it's
43 # unclear how the downsample factor is handled downstream of this and this needs to be investigated.
44 # in any case, all the unit tests pass with this change.
45 udfl=[Rat(wfl[0].td.Fs/wf.td.Fs) for wf in wfl]
46 wfl=[wf if uf[0] == 1 else wf*
47 (InterpolatorSinX(uf[0]) if strategy=='SinX' else InterpolatorLinear(uf[0]))
48 for (uf,wf) in zip(udfl,wfl)]
49 adl=[wfl[0].td/wf.td for wf in wfl]
50 fdl=[ad.D-int(ad.D) for ad in adl]
51 wfl=[wf if fd == 0.0 else wf*
52 (FractionalDelayFilterSinX(fd,True) if strategy=='SinX' else FractionalDelayFilterLinear(fd,True))
53 for (fd,wf) in zip(fdl,wfl)]
54 overlapping=wfl[0].td
55 for wf in wfl[1:]: overlapping=overlapping.Intersection(wf.td)
56 # overlapping now contains the overlapping waveform
57 adl=[overlapping/wf.td for wf in wfl]
58 trl=[WaveformTrimmer(max(0,int(round(ad.TrimLeft()))),max(0,int(round(ad.TrimRight())))) for ad in adl]
59 list.__init__(self,[wf*tr for (wf,tr) in zip(wfl,trl)])
