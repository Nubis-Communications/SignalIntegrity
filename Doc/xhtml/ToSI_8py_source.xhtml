90def ToSI(d,sa='',letterPrefixes=True,round=12):
91
92 if sa is None:
93 sa=''
94
95 if letterPrefixes:
96 incPrefixes = ['k', 'M', 'G', 'T', 'P', 'E', 'Z', 'Y']
97 decPrefixes = ['m', 'u', 'n', 'p', 'f', 'a', 'z', 'y']
98 spacer=' '
99 else:
100 incPrefixes = incexpPrefixes
101 decPrefixes = decexpPrefixes
102 spacer=''

104 if d==0.:
105 if letterPrefixes:
106 return '0'+spacer+sa
107 else:
108 return '0'
109
110 exponentInUnit=False
111 if len(sa)>=2:
112 if sa[0:2] in ['e-','e+']:
113 exponentInUnit=True
114 if not letterPrefixes:
115 exp=sa.split(' ')[0]
116 try:
117 s=str(d)+exp
118 number=float(s)
119 # if the number attempt passed, then we can return the string
120 return s
121 except ValueError:
122 d=d*float('1'+exp)
123 exponentInUnit=False
124
125 degree = int(math.floor(math.log10(math.fabs(d)) / 3))
126
127 prefix = ''
128
129 if not letterPrefixes:
130 sa=''
131
132 # certain units should not be scaled to engineering notation
133 prohibitedUnits = (sa[0:2] == 'dB') or ((sa == '') and letterPrefixes)
134
135 if degree!=0 and not exponentInUnit and not prohibitedUnits:
136 ds = degree/math.fabs(degree)
137 if ds == 1:
138 if degree - 1 < len(incPrefixes):
139 prefix = incPrefixes[degree - 1]
140 else:
141 prefix = incPrefixes[-1]
142 degree = len(incPrefixes)
143 elif ds == -1:
144 if -degree - 1 < len(decPrefixes):
145 prefix = decPrefixes[-degree - 1]
146 else:
147 prefix = decPrefixes[-1]
148 degree = -len(decPrefixes)

150 scaled = float(d * math.pow(1000, -degree))
151 s = ("{:."+str(round)+"g}"+spacer+"{}").format(scaled,prefix)
152 if not '.' in s:

161 # a milli-inch is called a mil

163 sa = 'il'
