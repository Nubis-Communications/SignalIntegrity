43
44 |x type |y type |outcome |

46 | Waveform | don't care | waveform is x provided (copy constructor) |

104 def __add__(self,other):
105 """overloads +
106 @param other instance of class Waveform or float, int, complex to add.
107 @return instance of class Waveform with other added to self
108 @note does not affect self
109 @note
110 valid types of other to add are:
111
112 - Waveform - if the other waveform has the same time descriptor, returns the waveform
113 with self and others values added together, otherwise adapts other to self and then
114 adds them.
115 - float,int,complex - adds the constant value to all values in self.
116 @throw SignalIntegrityExceptionWaveform if other cannot be added.
117 @see AdaptedWaveforms
118 """
119 if isinstance(other,Waveform):
120 if self.td == other.td:
121 return Waveform(self.td,[self[k]+other[k] for k in range(len(self))])
122 else:
123 [s,o]=AdaptedWaveforms([self,other])
124 return Waveform(s.td,[s[k]+o[k] for k in range(len(s))])
125 #return awf[0]+awf[1]
126 elif isinstance(other,(float,int,complex)):
127 return Waveform(self.td,[v+other.real for v in self])
128 # pragma: silent exclude
129 else:
130 raise SignalIntegrityExceptionWaveform('cannot add waveform to type '+str(other.__class__.__name__))

132 def __sub__(self,other):
133 """overloads -
134 @param other instance of class Waveform or float, int, complex to subtract.
135 @return instance of class Waveform with other subtracted from self
136 @note does not affect self
137 @note
138 valid types of other to subtract are:
139
140 - Waveform - if the other waveform has the same time descriptor, returns the waveform
141 with self and others values subtracted, otherwise adapts other to self and then
142 subtracts them.
143 - float,int,complex - subtracts the constant value from all values in self.
144 @throw SignalIntegrityExceptionWaveform if other cannot be subtracted.
145 @see AdaptedWaveforms
146 """

154 return Waveform(self.td,[v-other.real for v in self])

176 of other is:

178 - ImpulseResponse - returns self convolved with the impulse response.
179 - WaveformProcessor - returns self processed by the instance of WaveformProcessor.
180 - float,int,complex - returns the Waveform produced by multiplying all of the values in
181 self multiplied by the constant value supplied.

193 elif isinstance(other,ImpulseResponse):

226 def ReadFromFile(self,fileName):
227 """reads a waveform from a file
228 @param fileName string name of file to read
229 @return self
230 @note this DOES affect self
231 @note the normal waveform format is one number per line starting with the
232 horizontal offset followed by the number of points followed by the sample
233 rate. The remaining lines contain one waveform point per line. This is
234 the format output by SignalIntegrity. However, if the data is all on one line, then
235 the format is assumed to be LeCroy MathPack format with the first point
236 being the number of points, and the remaining points being time and value.
237 @note if the file extension is '.trc', then LeCroy waveform format is assumed
238 """
239 # pragma: silent exclude
240 _, file_extension = os.path.splitext(fileName)
241 if file_extension == '.trc':
242 self.ReadLeCroyWaveform(fileName)
243 return self
244 try:
245 # pragma: include outdent
246 with open(fileName,'rU' if sys.version_info.major < 3 else 'r') as f:
247 data=f.readlines()
248 # pragma: silent exclude
249 if len(data)==1:
250 data=data[0].split()
251 NumPts=int(float(data[0])+0.5)
252 HorOffset=float(data[1])
253 SampleRate=1./(float(data[3])-HorOffset)
254 Values=[float(data[k*2+2]) for k in range(NumPts)]
255 else:
256 # pragma: silent include outdent
257 HorOffset=float(data[0])
258 NumPts=int(float(data[1])+0.5)
259 SampleRate=float(data[2])
260 Values=[float(data[k+3]) for k in range(NumPts)]
261 # pragma: silent indent
262 self.td=TimeDescriptor(HorOffset,NumPts,SampleRate)
263 list.__init__(self,Values)
264 # pragma: silent exclude indent
265 except IOError:
266 raise SignalIntegrityExceptionWaveformFile(fileName+' not found')
267 # pragma: include
268 return self

294 def __eq__(self,other):
295 """overloads ==
296 @param other instance of other waveform.
297 @return boolean whether the waveforms are equal to each other.
298 @note an epsilon of 1e-6 is used for the compare.
299 """
300 if other == None:
301 return False
302 if len(self) != len(other):
303 return False
304 if self.td != other.td:
305 return False
306 for k in range(len(self)):
307 if abs(self[k]-other[k])>self.epsilonepsilon:
308 return False
309 return True

316 def Adapt(self,td):
317 """adapts waveform to time descriptor
318 Waveform adaption is performed using upsampling, decimation, fractional delay,
319 and waveform point trimming.
320 @param td instance of class TimeDescriptor to adapt waveform to
321 @return instance of class Waveform containing self adapted to the time descriptor
322 @note does not affect self.
323 @note the static member variable adaptionStrategy determines how to interpolate. 'SinX' means
324 to use sinx/x interpolation, 'Linear' means to use linear interpolation.
325 @see InterpolatorSinX
326 @see SignalIntegrity.TimeDomain.Filters.InterpolatorSinX.FractionalDelayFilterSinX
327 @see SignalIntegrity.TimeDomain.Filters.InterpolatorSinX.FractionalDelayFilterSinX
328 @see SignalIntegrity.TimeDomain.Filters.InterpolatorLinear.InterpolatorLinear
329 @see SignalIntegrity.TimeDomain.Filters.InterpolatorLinear.FractionalDelayFilterLinear
330 @see SignalIntegrity.TimeDomain.Filters.WaveformTrimmer.WaveformTrimmer
331 @see SignalIntegrity.TimeDomain.Filters.WaveformDecimator.WaveformDecimator
332 @see SignalIntegrity.Rat.Rat
333 """
334 # pragma: silent exclude
335 from SignalIntegrity.Lib.TimeDomain.Filters.InterpolatorSinX import InterpolatorSinX
336 from SignalIntegrity.Lib.TimeDomain.Filters.InterpolatorSinX import FractionalDelayFilterSinX
337 from SignalIntegrity.Lib.TimeDomain.Filters.InterpolatorLinear import InterpolatorLinear
338 from SignalIntegrity.Lib.TimeDomain.Filters.InterpolatorLinear import FractionalDelayFilterLinear
339 from SignalIntegrity.Lib.TimeDomain.Filters.WaveformTrimmer import WaveformTrimmer
340 from SignalIntegrity.Lib.TimeDomain.Filters.WaveformDecimator import WaveformDecimator
341 from SignalIntegrity.Lib.Rat import Rat
342 # pragma: include
343 wf=self
344 (upsampleFactor,decimationFactor)=Rat(td.Fs/wf.td.Fs)
345 if upsampleFactor>1:
346 # pragma: silent exclude
347 if wf.td.K*upsampleFactor > self.maximumWaveformSize:
348 raise SignalIntegrityExceptionWaveform('waveform too large to process')
349 # pragma: include
350 wf=wf*(InterpolatorSinX(upsampleFactor) if wf.adaptionStrategy=='SinX'
351 else InterpolatorLinear(upsampleFactor))
352 ad=td/wf.td
353 f=ad.D-int(math.floor(ad.D))
354 if not ((f<self.epsilonepsilon) or ((1-f)<self.epsilonepsilon)):
355 wf=wf*(FractionalDelayFilterSinX(f,True) if wf.adaptionStrategy=='SinX'
356 else FractionalDelayFilterLinear(f,True))
357 ad=td/wf.td
358 if decimationFactor>1:
359 decimationPhase=int(round(ad.TrimLeft())) % decimationFactor
360 wf=wf*WaveformDecimator(decimationFactor,decimationPhase)
361 ad=td/wf.td
362 tr=WaveformTrimmer(max(0,int(round(ad.TrimLeft()))),
363 max(0,int(round(ad.TrimRight()))))
364 wf=wf*tr
365 return wf
366 def Measure(self,time):
367 """measures a value at a given time
368 @param time float time to measure the value at.
369 @return value at time specified
370 @note will return None if time is not within the waveform
371 @note linearly interpolates nearest point
372 """
373 sample=(time-self.td.H)*self.td.Fs
374 k=int(math.floor(sample))
375 if k < 0 or k > (self.td.K-1): return None
376 frac=sample-k
377 res=frac*(self[k+1]-self[k])+self[k]
378 return res

413 def Integral(self,c=0.,addPoint=True,scale=True):
414 """integral of waveform
415 the integral is calculated using Riemann sums (as opposed to trapezoidal
416 integration.
417 @param c (optional) float value to add to the integral waveform
418 @param addPoint (optional) boolean whether to add a point to the waveform before
419 the first point. the value added is c.
420 @param scale (optional) boolean whether to multiply each sum by the sample period
421 providing a true integral. Otherwise, the values are simply summed.
422 """
423 td=copy(self.td)
424 i=[0 for k in range(len(self))]
425 T=1./td.Fs if scale else 1.
426 for k in range(len(i)):
427 if k==0:
428 i[k]=self[k]*T+c
429 else:
430 i[k]=i[k-1]+self[k]*T
431 td.H=td.H+(1./2.)*(1./td.Fs)
432 if addPoint:
433 td.K=td.K+1
434 td.H=td.H=td.H-1./td.Fs
435 i=[c]+i
436 return Waveform(td,i)
437 def Derivative(self,c=0.,removePoint=True,scale=True):
438 """derivative of waveform
439 the derivative is calculated using the difference divided by the sample period.
440 @param c (optional) this value is superfluous and not used.
441 @param removePoint (optional) boolean whether to remove the first point. If the
442 first point is not removed, it is zero.
443 @param scale (optional) boolean whether to divide each difference by the sample period
444 providing a true derivative. Otherwise, the values are simply subtracted.
445 @todo remove argument c.
446 """
447 td=copy(self.td)
448 vl=copy(self)
449 T=1./td.Fs if scale else 1.
450 for k in range(len(vl)):
451 if k==0:
452 vl[k]=0.
453 else:
454 vl[k]=(self[k]-self[k-1])/T
455 td.H=td.H-(1./2.)*(1./td.Fs)
456 if removePoint:
457 td.K=td.K-1
458 td.H=td.H+1./td.Fs
459 vl=vl[1:]
460 return Waveform(td,vl)
