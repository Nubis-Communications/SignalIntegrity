25class NetworkAnalyzer(SParameters):
26 """produces calibrated or uncalibrated s-parameters from a network analyzer.
27 The usual use of this class is to provide calibrated s-parameters, given raw s-parameters
28 and error terms, but it can also be used to go the reverse direction."""
29 def __init__(self,f,filename,etfilename,portListstring=None,calculate=True,callback=None,**kwargs):
30 """Constructor
31 Calculates calibrated or raw s-parameters from a network analyzer.
32 @param f list of frequencies to calculate s-parameters
33 @param filename string filename containing (usually raw) s-parameters.
34 @param etfilename string filename containing error-terms (see Calibration).
35 @param portListstring (optional, defaults to None) string containing comma delimited list
36 of one-based port numbers to use. This is so that the error-terms from a large port count calibration
37 can be used to make a smaller measurement on less ports, and for port reordering. Essentially, this
38 is the list of network analyzer ports used for the measurement.
39 @param calculate (optional, defaults to True) bool that defines the calculation direction. If calculate
40 is True, then the assumption is that the s-parameters provided are raw and the goal is to convert the
41 raw s-parameters to calibrated s-parameters. In the unusual situation where calculate is False, the direction
42 is reversed and the assumption is that the s-parameters provided are calibrated and the goal is to convert the
43 calibrated s-parameters to raw s-parameters.
44 @param callback function ptr (optional, defaults to None) callback function.
45 The callback function is used to pass down into s-parameter files that are actually
46 SignalIntegrity projects so that progress can be tracked and the UI thread can be kept
47 updated. The callback function should have a signature like Callback(self,number,name=None),
48 where the number is the progress in percent and the name is the name of the file being processed.
49 @param **kwargs dict (optional, defaults to {}) dictionary of arguments for the files
50 """
51 if portListstring!=None:
52 portlist=[int(p)-1 for p in portListstring.split(',')]
53 else: portlist=None
54 spraw=SParameterFile(filename,None,callback,**kwargs).Resample(f)
55 calibration=Calibration(0,0).ReadFromFile(etfilename,callback,**kwargs)
56 fixtures=calibration.Fixtures()
57 fixtures=[fixture.Resample(f) for fixture in fixtures]
58 calibration.InitializeFromFixtures(fixtures)
59 if calculate:
60 dut=calibration.DutCalculation(spraw,portlist)
61 else:
62 dut=calibration.DutUnCalculation(spraw,portlist)
63 SParameters.__init__(self,f,dut)
