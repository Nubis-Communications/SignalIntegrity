28 """base class for netlist based virtual probing solutions"""

38 def _ProcessVirtualProbeLine(self,line):
39 """processes a line of a netlist, handing virtual probe specific commands.
40 Lines that can be processed at this level are processed and lines that
41 are unknown are place in a list of unknown lines for upstream processing. This
42 enables derived classes to benefit from what this class knows how to process and
43 to simply add specific functionality. As a simple example, a derived simulator class
44 needs to add output probes, and this simple system description class knows nothing of
45 this.
46 netlist lines that are handled at this level are:
47 - 'output' - addition of an output probe.
48 - 'voltageoutput' - addition of a voltage output probe with a name.
49 - 'meas' - addition of a measurement probe.
50 - 'stim' - addition of a stimulus source.
51 - 'stimdef' - definition of a stimdef (the relationship between stims).
52 """
53 lineList=self.ReplaceArgs(LineSplitter(line))
54 if len(lineList) == 0: return
55 if lineList[0] == 'meas':
56 if self.m_sdm_sd.pMeasurementList is None: self.m_sdm_sd.pMeasurementList = []
57 for i in range(1,len(lineList),2):
58 self.m_sdm_sd.pMeasurementList.append((lineList[i],int(lineList[i+1])))
59 elif lineList[0] == 'output':
60 if self.m_sdm_sd.pOutputList is None: self.m_sdm_sd.pOutputList = []
61 for i in range(1,len(lineList),2):
62 self.m_sdm_sd.pOutputList.append((lineList[i],int(lineList[i+1])))
63 # pragma: silent exclude
64 elif lineList[0] in ['voltageoutput','eyeprobe']:
65 if self.m_sdm_sd.pOutputList is None: self.m_sdm_sd.pOutputList = []
66 for i in range(1,len(lineList),3):
67 self.m_sdm_sd.pOutputList.append((lineList[i+1],int(lineList[i+2]),lineList[i]))
68 # pragma: include
69 elif lineList[0] == 'stim':
70 for i in range((len(lineList)-1)//3):
71 self.m_sdm_sd.AssignM(lineList[i*3+2],int(lineList[i*3+3]),lineList[i*3+1])
72 elif lineList[0] == 'stimdef':
73 self.m_sdm_sd.pStimDef = [[float(e) for e in r] for r in [s.split(',')
74 for s in ''.join(lineList[1:]).strip(' ').strip('[[').
75 strip(']]').split('],[') ]]
76 else: self.m_ul.append(line)
77 def _ProcessLines(self):
78 """processes all of the lines in a netlist
79 @see _ProcessLine() for explanation of parameters and functionality.
80 """
81 SystemDescriptionParser._ProcessLines(self)
82 self.m_sdm_sd = VirtualProbe(self.m_sdm_sd)
83 lines=copy.deepcopy(self.m_ul); self.m_ul=[]
84 for line in lines: self._ProcessVirtualProbeLine(line)
85 return self
