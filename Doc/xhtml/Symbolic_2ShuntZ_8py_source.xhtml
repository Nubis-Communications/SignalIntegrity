24def ShuntZ(P,Z):
25 """symbolic shunt impedance
26 @param P integer number of ports (1,2,3,4)
27 if ports are 1, then returns SignalIntegrity.Symbolic.ShuntZ.ShuntZOnePort \n
28 if ports are 2, then returns SignalIntegrity.Symbolic.ShuntZ.ShuntZTwoPort \n
29 if ports are 3, then returns SignalIntegrity.Symbolic.ShuntZ.ShuntZThreePort \n
30 if ports are 4, then returns SignalIntegrity.Symbolic.ShuntZ.ShuntZFourPort \n
31 @param Z string impedance
32 @return list of list of string s-parameter matrix
33 containing LaTeX or ASCII strings for each element.
34 @note strings can be any valid LaTeX
35 """
36 if P==1:
37 return ShuntZOnePort(Z)
38 elif P==2:
39 return ShuntZTwoPort(Z)
40 elif P==3:
41 return ShuntZThreePort(Z)
42 elif P==4:
43 return ShuntZFourPort(Z)

45def ShuntZFourPort(Z):
46 """symbolic four port shunt impedance
47 @param Z string impedance
48 @return list of list of string s-parameter matrix
49 containing LaTeX or ASCII strings for each element.
50 @note strings can be any valid LaTeX
51 @note this is the symbolic version of SignalIntegrity.Lib.Devices.ShuntZ.ShuntZFourPort
52 @remark Ports 1 and 3 are connected to port 1 of the device D provided.\n
53 Ports 2 and 4 are connected to port 2 of the device D provided.\n
54 @todo check the port numbering
55 """
56 D='2\\cdot ('+Z+'+Z0 )'
57 return [[lfrac('-Z0',D),lfrac('Z0',D),lfrac('2\\cdot '+Z+'+Z0',D),lfrac('Z0',D)],
58 [lfrac('Z0',D),lfrac('-Z0',D),lfrac('Z0',D),lfrac('2\\cdot '+Z+'+Z0',D)],
59 [lfrac('2\\cdot '+Z+'+Z0',D),lfrac('Z0',D),lfrac('-Z0',D),lfrac('Z0',D)],
60 [lfrac('Z0',D),lfrac('2\\cdot '+Z+'+Z0',D),lfrac('Z0',D),lfrac('-Z0',D)]]

62def ShuntZThreePort(Z):
63 """symbolic three port shunt impedance
64 @param Z string impedance
65 @return list of list of string s-parameter matrix
66 containing LaTeX or ASCII strings for each element.
67 @note strings can be any valid LaTeX
68 @note this is the symbolic version of SignalIntegrity.Lib.Devices.ShuntZ.ShuntZThreePort
69 @remark Ports 1 and 2 are connected together and to port 1 impedance.\n
70 Port 3 is the other port of the impedance.\n
71 @todo check the port numbering
72 """
73 D='2\\cdot '+Z+'+3\\cdot Z0'
74 return [[lfrac('-Z0',D),lfrac('2\\cdot '+Z+'+2\\cdot Z0',D),lfrac('2\\cdot Z0',D)],
75 [lfrac('2\\cdot '+Z+'+2\\cdot Z0',D),lfrac('-Z0',D),lfrac('2\\cdot Z0',D)],
76 [lfrac('2\\cdot Z0',D),lfrac('2\\cdot Z0',D),lfrac('2\\cdot '+Z+'-Z0',D)]]

78def ShuntZTwoPort(Z):
79 """symbolic two port shunt impedance
80 @param Z string impedance
81 @return list of list of string s-parameter matrix
82 containing LaTeX or ASCII strings for each element.
83 @note strings can be any valid LaTeX
84 @note this is the symbolic version of SignalIntegrity.Lib.Devices.ShuntZ.ShuntZTwoPort
85 @remark Ports 1 and 2 are connected together and to one side of the impedance.\n
86 The other side of the impedance is tied to ground.
87 """
88 D='2\\cdot '+Z+' +Z0'
89 return [[lfrac('-Z0',D),lfrac('2\\cdot '+Z,D)],
90 [lfrac('2\\cdot '+Z,D),lfrac('-Z0',D)]]

92def ShuntZOnePort(Z):
93 """symbolic four port shunt impedance
94
95 This is simply a termination impedance to ground.
96
97 @param Z string impedance
98 @return list of list of string s-parameter matrix
99 containing LaTeX or ASCII strings for each element.
100 @note strings can be any valid LaTeX
101 @note this is the symbolic version of SignalIntegrity.Lib.Devices.TerminationZ.TerminationZ
102 """
103 return [[lfrac(' '+Z+' -Z0',' '+Z+' +Z0')]]
