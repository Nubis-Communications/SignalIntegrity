24class TLineTwoPortRLGCApproximate(SParameters):
25 """s-parameters of two port RLGC (telegrapher's) transmission line
26 calculated by approximating distributed parameters with a finite number
27 of sections specified."""
28 rtFraction=.01
29 def __init__(self,f, R, Rse, L, G, C, df, Z0=50., K=0, scale=1.):
30 """Constructor
31 @param f list of float frequencies
32 @param R float DC series resistance (ohms)
33 @param Rse float series skin-effect resistance (ohms/sqrt(Hz))
34 @param L float series inductance (H)
35 @param G float DC conductance to ground (S)
36 @param C float capacitance to ground (F)
37 @param df float dissipation factor (loss-tangent) of capacitance to ground
38 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
39 @param K (optional) integer number of sections (defaults to zero)
40 @param scale (optional) float amount to scale the line by (assuming all other values are per unit)
41 @note If K=0 is specified, it is modified to a value that will provided a very good numerical
42 approximation.
43
44 The calculation is such that round-trip propagation time (twice the electrical length)
45 of any one small section is no more than one percent of the fastest possible risetime.
46 """
47 R=R*scale; Rse=Rse*scale; L=L*scale; G=G*scale; C=C*scale; df=df
48 K=K*scale
49 if K==0:
50 """max possible electrical length"""
51 Td=math.sqrt(L*C)
52 Rt=0.45/f[-1] # fastest risetime
53 """sections such that fraction of risetime less than round trip
54 electrical length of one section"""
55 K=Td*2/(Rt*self.rtFraction)
56
57 self.m_K=K
