28class DeembedderParser(SystemDescriptionParser):
29 """base class for netlist based deembedding solutions"""
30 def __init__(self, f=None, args=None, Z0=50.):
31 """Constructor
32 frequencies may be provided at construction time (or not for symbolic solutions).
33 @param f (optional) list of frequencies
34 @param args (optional) string arguments for the circuit.
35 @param Z0 float (optional, defaults to 50.) reference impedance for the calculation
36 @remark Arguments are provided on a line as pairs of names and values separated by a space.
37 """
38 SystemDescriptionParser.__init__(self, f, args, Z0)
39 def _ProcessDeembedderLine(self,line):
40 """processes a line of a netlist, handing deembedding specific commands
41 Lines that can be processed at this level are processed and lines that
42 are unknown are place in a list of unknown lines for upstream processing. This
43 enables derived classes to benefit from what this class knows how to process and
44 to simply add specific functionality. As a simple example, a derived simulator class
45 needs to add output probes, and this simple system description class knows nothing of
46 this.
47 netlist lines that are handled at this level are:
48 - 'system' - addition of the system
49 - 'unknown' - addition of a device whose s-parameters are unknown.
50 Calls SystemDescriptionParser._ProcessLines(),
51 exludes 'connect' and 'port' in first call, then processes simulator lines, then
52 calls upstream one more time for the device connections, again excluding 'port'.
53 """
54 lineList=self.ReplaceArgs(LineSplitter(line))
55 if len(lineList) == 0: return
56 if lineList[0] == 'system':
57 dev=DeviceParser(self.m_f,None,None,lineList[1:],Z0=self.m_Z0)
58 if not dev.m_spf is None:
59 self.m_spc.append(('system',dev.m_spf))
60 elif lineList[0] == 'unknown':
61 self.m_sdm_sd.AddUnknown(lineList[1],int(lineList[2]))
62 else: self.m_ul.append(line)
63 def _ProcessLines(self):
64 """processes all of the lines in a netlist
65 @see _ProcessLine() for explanation of parameters and functionality.
66 """
67 SystemDescriptionParser._ProcessLines(self,['connect','port'])
68 self.m_sdm_sd = Deembedder(self.m_sdm_sd)
69 lines=copy.deepcopy(self.m_ul); self.m_ul=[]
70 for line in lines: self._ProcessDeembedderLine(line)
71 lines=copy.deepcopy(self.m_ul); self.m_ul=[]
72 for line in lines: SystemDescriptionParser._ProcessLine(self,line,[])
73 return self
