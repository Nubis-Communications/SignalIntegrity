45 def EnforcePassivity(self,maxSingularValue=1.):
46 """Enforces passivity on the s-parameters.
47 @param maxSingularValue (optional, defaults to 1) float maximumum singular value allowed
48 Enforces passivity by clipping all singular values to a maximum value.
49
50 the optional maximum value allows for adjusting devices with gain to a maximum
51 value. For passive devices, the maximum singular value is the default value of 1.
52 """
53 for n in range(len(self.m_d)):
54 (u,s,vh)=linalg.svd(self.m_d[n],full_matrices=1,compute_uv=1)
55 for si in range(len(s)): s[si]=min(maxSingularValue,s[si])
56 self.m_d[n]=dot(u,dot(diag(s),vh)).tolist()
57 return self
58 def IsCausal(self,threshold=0.):
59 """Checks whether the s-parameters are causal.
60 @param threshold (optional, defaults to 0) positive float threshold for causality detection
61 @return boolean True if the absolute value of all values in the impulse response
62 of each s-parameter before time zero are less
63 than the threshold provided otherwise returns False.
64
65 This is checked by generating the impulse response corresponding to the frequency
66 response of each s-parameter from and to port combination.
67 """
68 for toPort in range(self.m_P):
69 for fromPort in range(self.m_P):
70 fr=self.FrequencyResponse(toPort,fromPort)
71 ir=fr.ImpulseResponse()
72 if ir is None: return False
73 else:
74 t=ir.td
75 Ts=1./ir.td.Fs
76 for k in range(len(t)):
77 if t[k]<=-Ts:
78 if abs(ir[k])>threshold:
79 return False
80 return True
81 def EnforceCausality(self,preserveDC=False):
82 """Enforces causality by setting all of the values before time zero in the impulse
83 responses of the s-parameters to zero.
84 @param preserveDC (optional, defaults to False) boolean whether to preserve the DC point in causality enforcement
85 """
86 self.ResampleToEvenlySpaced()
87 for toPort in range(self.m_P):
88 for fromPort in range(self.m_P):
89 fr=self.FrequencyResponse(toPort+1,fromPort+1)
90 ir=fr.ImpulseResponse()
91 dc=sum(ir)
92 if ir is not None:
93 t=ir.td; Ts=1./ir.td.Fs
94 for k in range(len(t)):
95 if t[k]<=-Ts: ir[k]=0.
96 newdc=sum(ir)
97 if preserveDC and (newdc != 0):
98 ir=ir*(dc/newdc)
99 fr=ir.FrequencyResponse()
100 frv=fr.Response()
101 for n in range(len(frv)): self.m_d[n][toPort][fromPort]=frv[n]
102 self.ResampleToUnevenlySpaced()
103 return self
104 def WaveletDenoise(self,threshold=0.00001):
105 """Denoises the s-parameters
106 @param threshold (optional) float threshold for the wavelets (defaults to 0.00001).
107 Denoises the s-parameter by computing the wavelet transform of the impulse response
108 for each s-parameter from and to port combination and keeping only the wavelets whoe
109 absolute value is above the threshold."""
110 self.ResampleToEvenlySpaced()
111 from SignalIntegrity.Lib.Wavelets import WaveletDaubechies4
112 w=WaveletDaubechies4()
113 for toPort in range(self.m_P):
114 for fromPort in range(self.m_P):
115 fr=self.FrequencyResponse(toPort+1,fromPort+1)
116 ir=fr.ImpulseResponse()
117 if ir is not None:
118 irl=len(ir)
119 nirl=int(pow(2.,math.ceil(math.log(float(irl))/math.log(2.))))
120 ir=ir._Pad(nirl)
121 y=ir.Values()
122 Y=w.DWT(y)
123 Y=[0. if abs(Yv) <= threshold else Yv for Yv in Y]
124 y=w.IDWT(Y)
125 #ir.x=y
126 # @todo get rid of this hack
127 ir.__init__(ir.TimeDescriptor(),y)
128 ir=ir._Pad(irl)
129 fr=ir.FrequencyResponse()
130 frv=fr.Response()
131 for n in range(len(frv)):
132 self.m_d[n][toPort][fromPort]=frv[n]
133 self.ResampleToUnevenlySpaced()
134 return self

146 def DetermineImpulseResponseLength(self,epsilon=1e-6,allLengths=False):
147 """determines the impulse response lengths of the ports by comparing impulse response to threshold.
148 @param epsilon (optional, defaults to 1e-6) absolute threshold on impulse response.
149 @param allLengths (optional, defaults to False) whether to return the lengths of each port combination
150 @return returns a tuple containining (negativeTime,positiveTime) if allLengths is false (default) containing
151 the most negative time and most positive time for all impulse responses, otherwise if allLengths is true, it returns
152 a list of list of tuples as stated for each impulse response.
153 """
154 lengths=[[(None,None) for _ in range(self.m_P)] for _ in range(self.m_P)]
155 for toPort in range(self.m_P):
156 for fromPort in range(self.m_P):
157 (negativeTimeLimit,positiveTimeLimit)=(None,None)
158 fr=self.FrequencyResponse(toPort+1,fromPort+1)
159 ir=fr.ImpulseResponse()
160 if ir is not None:
161 t=ir.td
162 for k in range(len(t)):
163 if t[k]<0.:
164 if negativeTimeLimit is None:
165 if abs(ir[k])>epsilon:
166 negativeTimeLimit=t[k]
167 else:
168 if abs(ir[k])>epsilon:
169 positiveTimeLimit=t[k]
170 if negativeTimeLimit is None:
171 negativeTimeLimit=0.0
172 if positiveTimeLimit is None:
173 positiveTimeLimit=0.0
174 maxlength=ir.td.K/ir.td.Fs/2.
175 negativeTimeLimit=max(-maxlength,math.floor(negativeTimeLimit*ir.td.Fs)/ir.td.Fs)
176 positiveTimeLimit=min(maxlength,math.ceil(positiveTimeLimit*ir.td.Fs)/ir.td.Fs)
177 lengths[toPort][fromPort]=(negativeTimeLimit,positiveTimeLimit)
178 if allLengths: return lengths
179 negativeLengths=[lengths[toPort][fromPort][0] for toPort in range(self.m_P) for fromPort in range(self.m_P)]
180 positiveLengths=[lengths[toPort][fromPort][1] for toPort in range(self.m_P) for fromPort in range(self.m_P)]
181 if all(length is None for length in negativeLengths): minNegativeLengths = None
182 else:
183 negativeLengths=[length if length is not None else 0.0 for length in negativeLengths]
184 minNegativeLengths = min(negativeLengths)
185 if all(length is None for length in positiveLengths): maxPositiveLengths = None
186 else:
187 positiveLengths=[length if length is not None else 0.0 for length in positiveLengths]
188 maxPositiveLengths = max(positiveLengths)
189 return (minNegativeLengths,maxPositiveLengths)

221 def EnforceReciprocity(self):
222 """ Enforces reciprocity on the s-parameters
223 affects self
224 @return self
225 """

233 def EnforceBothPassivityAndCausality(self,causalityThreshold=0.,maxIterations=30,maxSingularValue=1.,preserveDC=False):
234 """Enforces both passivity and causality on the s-parameters.
235 Affects self.
236 For up to the maxIterations specified, alternately enforces passivity, to the maxSingularValue and
237 enforces causality (because each causes violations of the other, but alternating this way generally
238 converges. If causality is detected by being below the causalityThreshold, then iterations stop.
239 @param maxSingularValue (optional, defaults to 1) float maximumum singular value allowed
240 @param maxIterations (optional, defaults to 30) maximum iterations
241 @param causalityThreshold (optional, defaults to 0) positive float threshold for causality detection
242 @param preserveDC (optional, defaults to False) boolean whether to preserve the DC point in causality enforcement
243 @return self
244 @see EnforcePassivity
245 @see EnforceCausality
246 """
247 iterationCount=0
248 keepGoing=True
249 while keepGoing:
250 self.EnforcePassivity(maxSingularValue)
251 self.EnforceCausality(preserveDC)
252 iterationCount=iterationCount+1
253 if iterationCount >= maxIterations: keepGoing=False
254 elif self.IsCausal(causalityThreshold): keepGoing=False
255 else: keepGoing = True
256 return self

262 enforces causality (because each causes violations of the other, but alternating this way generally
263 converges. If causality is detected by being below the causalityThreshold, then iterations stop.
264 @param maxSingularValue (optional, defaults to 1) float maximumum singular value allowed
265 @param maxIterations (optional, defaults to 30) maximum iterations
266 @param causalityThreshold (optional, defaults to 0) positive float threshold for causality detection
267 @param preserveDC (optional, defaults to False) boolean whether to preserve the DC point in causality enforcement

271 @see EnforceCausality

273 self.EnforceReciprocity()
274 self.EnforceBothPassivityAndCausality(causalityThreshold, maxIterations, maxSingularValue,preserveDC)
275 return self
276 def RemoveImpulseResponseOffset(self,lengths=None):
277 """Removes offset in the impulse response.
278 @param lengths tuple or list of list of tuple impulse response lengths where the
279 tuple contains the negative time limit and the positive time limit. The offset to remove
280 is the mean value of all areas of the impulse repsonse outside these limits
281 @return self (with impulse response offsets removed)
282 @remark if the lengths are a single number, it is assumed to be the single
283 length, otherwise if the lengths is a list of list, it is length to be enforced
284 for each port-port connection.
285 """
286 self.ResampleToEvenlySpaced()
287 if lengths is None:
288 lengths=(-1e15,1e15)
289 if not isinstance(lengths,list):
290 lengths=[[lengths for _ in range(self.m_P)] for _ in range(self.m_P)]
291 for toPort in range(self.m_P):
292 for fromPort in range(self.m_P):
293 (negativeTimeLimit,positiveTimeLimit)=lengths[toPort][fromPort]
294 fr=self.FrequencyResponse(toPort+1,fromPort+1)
295 ir=fr.ImpulseResponse()
296 mean=0.
297 count=0
298 if ir is not None:
299 t=ir.td
300 for k in range(len(t)):
301 if t[k]<=negativeTimeLimit:
302 count+=1
303 mean+=ir[k]
304 if t[k]>=positiveTimeLimit:
305 count+=1
306 mean+=ir[k]
307 if count==0:
308 mean=(ir[0]+ir[-1])/2.
309 else:
310 mean=mean/count
311 for k in range(len(t)):
312 ir[k]-=mean
313 fr=ir.FrequencyResponse()
314 frv=fr.Response()
315 for n in range(len(frv)):
316 self.m_d[n][toPort][fromPort]=frv[n]
317 self.ResampleToUnevenlySpaced()
318 return self
319 def Taper(self,from_frequency,to_frequency=None):
320 """Tapers the frequency response.
321 The response is windowed as flat out to the from_frequency and tapered to zero using
322 a raised cosine between the from_frequency and the to_frequency.
323 @param from_frequency float frequency to remain flat to
324 @param to_frequency float frequency to taper to zero at
325 @return self (with response tapered)
326 @remark if the from_frequency is above the last frequency, the s-parameters are unaffected.
327 @remark if the to_frequency is None, then the last frequency is used.
328 @remark if the to_frequency is below the from_frequency, the s-parameters are unaffected.
329 @remark if the to_frequency is above the last frequency, it is tapered to zero at the last frequency.
330 """
331 if to_frequency is None:
332 to_frequency = self.m_f[-1]
333 to_frequency = min(to_frequency,self.m_f[-1])
334 if to_frequency < from_frequency:
335 return self
336 if from_frequency > self.m_f[-1]:
337 return self
338 window = [1. if f <= from_frequency else
339 0 if f >= to_frequency else 0.5+0.5*math.cos((f-from_frequency)/(to_frequency-from_frequency)*math.pi)
340 for f in self.m_f]
341 for to_port in range(self.m_P):
342 for from_port in range(self.m_P):
343 for n in range(len(self.m_f)):
344 self.m_d[n][to_port][from_port] *= window[n]
345 return self
