20from numpy import array,identity
21from numpy.linalg import inv
22import numpy as np
23
24from SignalIntegrity.Lib.Conversions.Z0KHelper import Z0KHelper

26def ReferenceImpedance(S,Z0f,Z0i=None,Kf=None,Ki=None):
27 """Changes the reference impedance and scaling factor
28 @param S s-parameter matrix to convert
29 @param Z0f the reference impedance to convert to
30 @param Z0i (optional) the reference impedance of the s-parameters (assumed 50 ohms)
31 @param Kf (optional) assumed to be sqrt(Z0f)
32 @param Ki (optional) assumed to be sqrt(Z0i)
33 @return the converted s-parameters
34 @see Z0KHelper to see how the reference impedance
35 and scaling factor are determined."""
36 (Z0f,Kf)=Z0KHelper((Z0f,Kf),len(S))
37 (Z0i,Ki)=Z0KHelper((Z0i,Ki),len(S))
38 I=array(identity(len(S)))
39 p=(array(Z0f)-array(Z0i)).dot(inv(array(Z0f)+array(Z0i)))
40 Kf=array(Ki).dot(inv(array(Kf)))
41 S=array(S)
42 result = (Kf.dot(inv(I-p)).dot(S-p).dot(
43 inv(I-p.dot(S))).dot(I-p).dot(inv(Kf))).tolist()
44 # pragma: silent exclude
45 if np.any(np.isnan(result)):
46 raise np.linalg.LinAlgError
47 # pragma: include
48 return result
