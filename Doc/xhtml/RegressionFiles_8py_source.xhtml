374 def CanonicalizePaths(lines,directory):
375 """Replaces a directory appearing inside netlist lines with a placeholder.
376 @param lines list of strings the netlist lines.
377 @param directory string the directory to stand in for.
378 @return list of strings the lines, with the directory replaced.
379 @remark A file referenced by a variable is written into the netlist as an
380 absolute path, so a stored netlist would otherwise record the machine it was
381 generated on and could never match anywhere else. Only the part of the path
382 that varies is replaced, so a reference to a different file still differs.
383 """

385 return lines
386 directory=directory.replace('\\','/').rstrip('/')
387 if directory=='':
388 return lines
389 alternatives=[re.escape(directory),re.escape(directory.replace('/','\\'))]
390 # drive letters and file names do not distinguish case on Windows
391 expression=re.compile('|'.join(alternatives),re.IGNORECASE if os.name=='nt' else 0)

393 @staticmethod
394 def _ConnectionPairs(line):
395 """The device/pin pairs of a connect line, or None if it is not one.
396 @param line string a netlist line.

557 @staticmethod
558 def SParametersAreEqual(lhs,rhs,epsilon=0.00001):
559 """Whether two sets of s-parameters agree to within epsilon.
560 @return bool True if they agree.
561 """
562 if lhs.m_P != rhs.m_P: return False
563 if lhs.m_Z0 != rhs.m_Z0: return False
564 if len(lhs) != len(rhs): return False
565 for n in range(len(lhs)):
566 if abs(lhs.m_f[n] - rhs.m_f[n]) > .1:
567 return False
568 lhsn=lhs[n]
569 rhsn=rhs[n]
570 for r in range(lhs.m_P):
571 for c in range(lhs.m_P):
572 if abs(lhsn[r][c] - rhsn[r][c]) > epsilon:
573 return False
574 return True
575 @staticmethod
576 def _SParameterDifference(sp,regression):
577 """Describes how far apart two sets of s-parameters are."""
578 if sp.m_P != regression.m_P:
579 return 'has '+str(sp.m_P)+' ports, reference has '+str(regression.m_P)
580 if sp.m_Z0 != regression.m_Z0:
581 return 'reference impedance is '+str(sp.m_Z0)+', reference has '+str(regression.m_Z0)
582 if len(sp) != len(regression):
583 return 'has '+str(len(sp))+' frequencies, reference has '+str(len(regression))
584 worst=0.; where=None
585 for n in range(len(sp)):
586 if abs(sp.m_f[n]-regression.m_f[n]) > .1:
587 return 'frequency '+str(n)+' is '+str(sp.m_f[n])+', reference has '+str(regression.m_f[n])
588 for r in range(sp.m_P):
589 for c in range(sp.m_P):
590 error=abs(sp[n][r][c]-regression[n][r][c])
591 if error > worst:
592 worst=error; where=(n,r,c)
593 if where is None:
594 return 'incorrect'
595 n,r,c=where

894 def DiffTrees(self,referenceRoot,candidateRoot,ignore=None,artifacts=None):
895 """Compares two sprayed regression trees file by file.
896 @param referenceRoot string the golden tree.
897 @param candidateRoot string the freshly generated tree.
898 @param ignore set of strings (optional) base names to skip (e.g. manifest.json).
899 @param artifacts collection of strings (optional) when given, only files of these
900 artifact types are compared; the reference's other artifacts are ignored (used by
901 the quick, measurements-only screen against a full golden archive).
902 @return list of RegressionResult, one per differing or missing file, ordered
903 so an upstream cause precedes its downstream effects.
904 """
905 ignore=set() if ignore is None else set(ignore)
906 artifacts=None if artifacts is None else set(artifacts)
907 referenceFiles=self._RelativeFiles(referenceRoot)
908 candidateFiles=self._RelativeFiles(candidateRoot)
909 results=[]
910 for relative in sorted(referenceFiles | candidateFiles):
911 if os.path.basename(relative) in ignore:
912 continue
913 if self._ArtifactOf(relative) == 'eye':
914 # eye bitmaps are archived for reference only, never compared
915 continue
916 if artifacts is not None and self._ArtifactOf(relative) not in artifacts:
917 # quick screen: only the requested artifact types participate
918 continue
919 if relative not in candidateFiles:
920 results.append(RegressionResult(self._ArtifactOf(relative),relative,False,
921 'missing from candidate'))
922 continue
923 if relative not in referenceFiles:
924 results.append(RegressionResult(self._ArtifactOf(relative),relative,False,
925 'unexpected in candidate'))
926 continue
927 result=self._CompareFiles(relative,
928 os.path.join(referenceRoot,relative),
929 os.path.join(candidateRoot,relative))
930 result.filename=relative
931 results.append(result)
932 results.sort(key=self._DiffSortKey)
