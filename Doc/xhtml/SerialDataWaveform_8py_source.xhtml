29 """a serial data NRZ waveform"""
30 rtvsT=0.5903445
31 def __init__(self,pattern,bitRate,amplitude=1.0,risetime=0.,delay=0.,tdOrFs=None):
32 """constructor
33 @param pattern list of bits in pattern which must be 0 or 1
34 @param bitRate float bitrate in bits/s
35 @param amplitude (optional, defaults to 1.0) float amplitude of zero centered serial data waveform (pk-pk amplitude is twice this value)
36 @param risetime (optional, defaults to 0.0) float risetime of a 1/2 cosine response.
37 @param delay (optional, defaults to 0,0) float delay in time into the pattern.
38 @param tdOrFs (optional, defaults to None) time descriptor or float sample rate. If None is specified, then
39 the descriptor is generated for one pattern length with a sample rate of 10x the bitrate. Otherwise, if a float
40 is specified, then a time descriptor for one pattern length is generated with the sample rate specified. Otherwise
41 the time descriptor specified is used.
42 @return self, a waveform.
43 @throw SignalIntegrityExceptionWaveform exception is raised if the risetime exceeds 59% of the unit interval as this causes
44 the calculation to fail.
45 @todo the failure for a given risetime is due to the simplicity of the algorithm which should be improved at some point.
46 this simplicity is that it only looks at adjacent bits to determine the intersymbol effect of the risetime.
47 @note the left edge of the first bit in the pattern is at the delay time. But because a cosine is used for
48 the risetime emulation, the 50% point of the cosine is reached at this edge.
49 @note the risetime is adjusted if it is too low relative to the sample period.
50 """
51 if tdOrFs is None:
52 sampleRate=10.*bitRate
53 td=self.TimeDescriptorTimeDescriptor(bitRate, sampleRate, len(pattern))
54 elif isinstance(tdOrFs,float):
55 sampleRate=tdOrFs
56 td=self.TimeDescriptorTimeDescriptor(bitRate, sampleRate, len(pattern))
57 else:
58 td=tdOrFs
59 T=max(risetime/self.rtvsTrtvsT,1/td.Fs)
60 patternTimeLength=len(pattern)/bitRate
61 unitInterval=1./bitRate
62
63 if td.Fs == bitRate and risetime == 0.:
64 Waveform.__init__(self,Waveform(td,[2.*amplitude*(pattern[v]-0.5) for v in range(len(td))]))
65 return
66
67 if T>unitInterval:
68 # todo: would like to handle this case in the future
69 raise SignalIntegrityExceptionWaveform('PRBS risetime too high\nfor waveform generation. '+\
70 'Limit is '+ToSI(unitInterval*self.rtvsTrtvsT,'s')+' for given bitrate')
71 v=[0. for _ in range(len(td))]
72 for k in range(len(td)):
73 t=td[k]
74 timeInPattern=((t-delay)%patternTimeLength+patternTimeLength)%patternTimeLength
75 thisBit=int(math.floor(timeInPattern/unitInterval))
76 timeInBit=timeInPattern-thisBit*unitInterval
77 thisBitValue=pattern[thisBit]
78 if timeInBit>=T/2. and (timeInBit-unitInterval)<=-T/2.:
79 v[k]=2.*amplitude*(thisBitValue-0.5)
80 elif timeInBit<T/2.:
81 previousBit=(thisBit-1+len(pattern))%len(pattern)
82 previousBitTransition=thisBitValue-pattern[previousBit]
83 if previousBitTransition==0:
84 v[k]=2.*amplitude*(thisBitValue-0.5)
85 else:
86 v[k]=amplitude*previousBitTransition*math.cos(math.pi*(timeInBit/T+3./2.))
87 elif (timeInBit-unitInterval)>-T/2.:
88 nextBit=(thisBit+1)%len(pattern)
89 nextBitTransition=pattern[nextBit]-thisBitValue
90 if nextBitTransition==0:
91 v[k]=2.*amplitude*(thisBitValue-0.5)
92 else:
93 v[k]=amplitude*nextBitTransition*math.cos(math.pi*((timeInBit-unitInterval)/T+3./2.))
94 Waveform.__init__(self,Waveform(td,v))
