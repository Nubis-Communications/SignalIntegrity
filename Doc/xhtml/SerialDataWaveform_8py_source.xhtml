29class SerialDataWaveform(Waveform):
30 """a serial data NRZ waveform"""
31 rtvsT=0.5903445
32 def __init__(self,pattern,bitRate,amplitude=1.0,risetime=0.,delay=0.,tdOrFs=None):
33 """constructor
34 @param pattern list of bits in pattern which must be 0 or 1
35 @param bitRate float bitrate in bits/s
36 @param amplitude (optional, defaults to 1.0) float amplitude of zero centered serial data waveform (pk-pk amplitude is twice this value)
37 @param risetime (optional, defaults to 0.0) float risetime of a 1/2 cosine response.
38 @param delay (optional, defaults to 0,0) float delay in time into the pattern.
39 @param tdOrFs (optional, defaults to None) time descriptor or float sample rate. If None is specified, then
40 the descriptor is generated for one pattern length with a sample rate of 10x the bitrate. Otherwise, if a float
41 is specified, then a time descriptor for one pattern length is generated with the sample rate specified. Otherwise
42 the time descriptor specified is used.
43 @throw SignalIntegrityExceptionWaveform exception is raised if the risetime exceeds 59% of the unit interval as this causes
44 the calculation to fail.
45 @todo the failure for a given risetime is due to the simplicity of the algorithm which should be improved at some point.
46 this simplicity is that it only looks at adjacent bits to determine the intersymbol effect of the risetime.
47 @note the left edge of the first bit in the pattern is at the delay time. But because a cosine is used for
48 the risetime emulation, the 50% point of the cosine is reached at this edge.
49 @note the risetime is adjusted if it is too low relative to the sample period.
50 """
51 if tdOrFs is None:
52 sampleRate=10.*bitRate
53 td=self.TimeDescriptor(bitRate, sampleRate, len(pattern))
54 elif isinstance(tdOrFs,float):
55 sampleRate=tdOrFs
56 td=self.TimeDescriptor(bitRate, sampleRate, len(pattern))
57 else:
58 td=tdOrFs
59 v=SerialDataWaveform._Render(td,bitRate,[pattern],[amplitude],risetime,delay)
60 Waveform.__init__(self,Waveform(td,v))
61 @staticmethod
62 def _Render(td,bitRate,patterns,amplitudes,risetime,delay):
63 """vectorized serial-data renderer
64 @param td instance of TimeDescriptor for the output waveform.
65 @param bitRate float bit (symbol) rate.
66 @param patterns list of equal-length binary patterns (one per bit position).
67 @param amplitudes list of float amplitudes, one for each pattern.
68 @param risetime float risetime of a 1/2 cosine response.
69 @param delay float delay in time into the pattern.
70 @return list of float sample values equal to the sum of the serial-data
71 waveforms for each (pattern,amplitude) pair.
72 @throw SignalIntegrityExceptionWaveform if the risetime exceeds the unit interval.
73 @note this reproduces the per-bit SerialDataWaveform algorithm exactly while
74 computing the shared sample geometry only once. Multi-level waveforms are the
75 sum of several single-bit waveforms, so rendering them in a single pass (rather
76 than building and summing a separate waveform per bit) is what makes them fast.
77 """
78 N=len(td)
79 unitInterval=1./bitRate
80 if td.Fs == bitRate and risetime == 0.:
81 acc=np.zeros(N)
82 for pattern,amplitude in zip(patterns,amplitudes):
83 acc=acc+2.*amplitude*(np.asarray(pattern,dtype=float)-0.5)
84 return acc.tolist()
85 T=max(risetime/SerialDataWaveform.rtvsT,1/td.Fs)
86 if T>unitInterval:
87 # todo: would like to handle this case in the future
88 raise SignalIntegrityExceptionWaveform('PRBS risetime too high\nfor waveform generation. '+\
89 'Limit is '+ToSI(unitInterval*SerialDataWaveform.rtvsT,'s')+' for given bitrate')
90 patternLength=len(patterns[0])
91 patternTimeLength=patternLength*unitInterval
92 t=np.arange(N)/td.Fs+td.H
93 timeInPattern=np.mod(np.mod(t-delay,patternTimeLength)+patternTimeLength,patternTimeLength)
94 thisBit=np.floor(timeInPattern/unitInterval).astype(np.int64)
95 np.clip(thisBit,0,patternLength-1,out=thisBit)
96 timeInBit=timeInPattern-thisBit*unitInterval
97 risingEdge=timeInBit<T/2.
98 fallingEdge=(timeInBit-unitInterval)>-T/2.
99 previousBit=(thisBit-1)%patternLength
100 nextBit=(thisBit+1)%patternLength
101 cosRising=np.cos(np.pi*(timeInBit/T+3./2.))
102 cosFalling=np.cos(np.pi*((timeInBit-unitInterval)/T+3./2.))
103 acc=np.zeros(N)
104 for pattern,amplitude in zip(patterns,amplitudes):
105 p=np.asarray(pattern,dtype=float)
106 thisBitValue=p[thisBit]
107 v=2.*amplitude*(thisBitValue-0.5)
108 previousBitTransition=thisBitValue-p[previousBit]
109 v=np.where(risingEdge & (previousBitTransition!=0.),
110 amplitude*previousBitTransition*cosRising,v)
111 nextBitTransition=p[nextBit]-thisBitValue
112 v=np.where(fallingEdge & (nextBitTransition!=0.),
113 amplitude*nextBitTransition*cosFalling,v)
114 acc=acc+v
115 return acc.tolist()

120 return TimeDescriptor(0.,numPoints,sampleRate)
