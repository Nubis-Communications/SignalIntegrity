42 def LaTeXSolution(self,**args):
43 """Calculates and stores internally a symbolic representation
44 of the LaTeX s-parameter solution for the network.
45 @param args named arguments (name=value).
46 @return self
47 the named arguments possible are:
48 - 'solvetype' - (optional) either 'block' for the block solution or 'direct' for
49 the direct solution (defaults to 'block').
50 - 'size' - either 'normal', 'big', or 'biggest' (defaults to 'normal').
51 @see Symbolic
52 Here, 'size' is used depending on the size of the solution.
53 If the size is normal, then the block solution is output in matrix form
54 on one line. If the size is big, the inverse of (I-Wxx) is shown on one line
55 and Wxx is used on the next single line solution, as this tends to become
56 large sometimes. The biggest size is used when the solution is really large
57 and it basically defines each matrix in the block solution on its own line
58 and then writes the final result using the names of the matrices."""
59 # pragma: silent exclude
60 self.CheckConnections()
61 # pragma: include
62 solvetype = args['solvetype'] if 'solvetype' in args else 'block'
63 size = args['size'] if 'size' in args else 'normal'
64 AN=self.PortBNames(); BN=self.PortANames()
65 if solvetype=='direct':
66 self._LaTeXSi()
67 BN=self.PortANames(); AN=self.PortBNames()
68 n=self.NodeVector()
69 SCI=Device.SymbolicMatrix('Si',len(n))
70 B=[[0]*len(BN) for p in range(len(BN))]
71 for r in range(len(BN)):
72 for c in range(len(BN)):
73 B[r][c]=SCI[n.index(BN[r])][n.index(AN[c])]
74 self._AddEq('\\mathbf{S} = '+self._LaTeXMatrix(B))
75 return self
76 XN=self.OtherNames(AN+BN)

78 sWba=self._LaTeXMatrix(Wba)
79 Wxx=self.WeightsMatrix(XN,XN)
80 if len(Wxx)==0:
81 self._AddEq('\\mathbf{S} = '+sWba)
82 return self
83 Wbx=self.WeightsMatrix(BN,XN)
84 Wxa=self.WeightsMatrix(XN,AN)
85 if AllZeroMatrix(Wbx) or AllZeroMatrix(Wxa):
86 self._AddEq('\\mathbf{S} = '+sWba)

91 XNzcWbx=[XN[zcWbx] for zcWbx in ZeroColumns(Wbx)]
