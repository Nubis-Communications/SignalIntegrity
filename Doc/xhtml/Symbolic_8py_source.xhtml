28class Symbolic():
29 """base class for all symbolic solutions."""
30 def __init__(self,**args):
31 """Constructor
32 @param args (optional) named args (name=value)
33 named arguments understood are:
34 - 'size' - 'normal' or 'small' (defaults to 'normal')
35 - 'docstart' - string for start of LaTeX document (defaults to a barebones
36 article class document start.
37 - 'docend' - string for end of document (defaults to normal LaTeX document end).
38 - 'eqprefix' - prefix for equation start (defaults to \\[).
39 - 'eqsuffix' - suffix for equation end (defaults to \\]).
40 - 'eqenv' - boolean whether to enclose the equation in the equation environment
41 Here, the size refers to the sizes of the output matrices. 'normal' means the
42 normal size matrices and 'small' means the \\smallmatrix environment.
43 """
44 self.m_lines=[]
45 size = args['size'] if 'size' in args else 'normal'
46 self.m_docStart = args['docstart'] if 'docstart' in args else\
47 '\\documentclass[10pt]{article}\n'+'\\usepackage{amsmath}\n'+\
48 '\\usepackage{bbold}\n'+'\\usepackage{bbm}\n'+'\\begin{document}'
49 self.m_docEnd = args['docend'] if 'docend' in args else '\\end{document}'
50 self.m_eqPrefix = args['eqprefix'] if 'eqprefix' in args else '\\[ '
51 self.m_eqSuffix = args['eqsuffix'] if 'eqsuffix' in args else ' \\]'
52 self.m_identity = '\\boldsymbol{\\mathbbm{I}} '
53 self.m_eqEnvironment = args['eqenv'] if 'eqenv' in args else True
54 self.m_small= (size == 'small')
