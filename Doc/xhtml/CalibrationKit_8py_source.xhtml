108 def ReadFromFile(self,filename):
109 """The file is read from the disk and the constants stored internally.
110 @param filename string name of calibration constant file to read.
111 @return self
112 """
113 with open(filename) as f:
114 lines=f.readlines()
115 actualLines=[]
116 for line in lines:
117 if line.strip()[0]!='%':
118 actualLines.append(line.strip())
119 # % C0 (fF) - OPEN
120 self.openC0=float(actualLines[0])*1e-15
121 # % C1 (1e-27 F/Hz) - OPEN
122 self.openC1=float(actualLines[1])*1e-27
123 # % C2 (1e-36 F/Hz^2) - OPEN
124 self.openC2=float(actualLines[2])*1e-36
125 # % C3 (1e-45 F/Hz^3) - OPEN
126 self.openC3=float(actualLines[3])*1e-45
127 # % offset delay (pS) - OPEN
128 self.openOffsetDelay=float(actualLines[4])*1e-12
129 # % real(Zo) of offset length - OPEN
130 self.openOffsetZ0=float(actualLines[5])
131 # % offset loss (Gohm/s) - OPEN
132 self.openOffsetLoss=float(actualLines[6])*1e9
133 # % L0 (pH) - SHORT
134 self.shortL0=float(actualLines[7])*1e-12
135 # % L1 (1e-24 H/Hz) - SHORT
136 self.shortL1=float(actualLines[8])*1e-24
137 # % L2 (1e-33 H/Hz^2) - SHORT
138 self.shortL2=float(actualLines[9])*1e-33
139 # % L3 (1e-42 H/Hz^3) - SHORT
140 self.shortL3=float(actualLines[10])*1e-42
141 # % offset delay (pS) - SHORT
142 self.shortOffsetDelay=float(actualLines[11])*1e-12
143 # % real(Zo) of offset length - SHORT
144 self.shortOffsetZ0=float(actualLines[12])
145 # % offset loss (Gohm/s) - SHORT
146 self.shortOffsetLoss=float(actualLines[13])*1e9
147 # % load resistance (ohm) - LOAD
148 self.loadZ=float(actualLines[14])
149 # % offset delay (pS) - LOAD
150 self.loadOffsetDelay=float(actualLines[15])*1e-12
151 # % real(Zo) of offset length - LOAD
152 self.loadOffsetZ0=float(actualLines[16])
153 # % offset loss (Gohm/s) - LOAD
154 self.loadOffsetLoss=float(actualLines[17])*1e9
155 # % offset delay (pS) - THRU
156 self.thruOffsetDelay=float(actualLines[18])*1e-12
157 # % real(Zo) of offset length - THRU
158 self.thruOffsetZ0=float(actualLines[19])
159 # % offset loss (Gohm/s) - THRU
160 self.thruOffsetLoss=float(actualLines[20])*1e9
161 self.f0=1e9
162 return self

299class CalibrationKit(object):
300 """Class for holding a calibration kit derived from calibration constants"""
301 def __init__(self,filename=None,f=None):
302 """Constructor
303 @param filename (optional) string filename of calibration constants file to read.
304 @param f (optional) list of frequencies to define the calibration standards for.
305 The calibration constants are initialized to their default as specified in CalibrationConstants.__init__().
306
307 If a filename is specified, the calibration constants are read from the disk.
308
309 If frequencies are provided, the frequencies are initialized and the calibration standards are initialized
310 to have the s-parameters of the standards as defined by the frequencies and calibration constants.
311
312 No calibration standards are generated yet if no frequencies are provided.
313 @see IntializeFrequency
314 """
315 self.Constants=CalibrationConstants()
316 self.m_f=None
317 if not filename is None:
318 self.ReadFromFile(filename)
319 if not f is None:
320 self.InitializeFrequency(f)
321 def InitializeFrequency(self,f):
322 """Initializes frequencies
323 @param f list of frequencies
324 Calibration standards are initialized to have the s-parameters of the standards as defined by the
325 frequencies and calibration constants.
326
327 Once intialized, the s-parameters of the calibration standards can be accessed and used.
328 @see openStandard
329 @see shortStandard
330 @see loadStandard
331 @see thruStandard
332 """
333 self.m_f=f
334 self.openStandard=OpenStandard(self.m_f,self.Constants.openOffsetDelay,
335 self.Constants.openOffsetZ0,self.Constants.openOffsetLoss,
336 self.Constants.f0,self.Constants.openC0,self.Constants.openC1,
337 self.Constants.openC2,self.Constants.openC3)
338 self.shortStandard=ShortStandard(self.m_f,self.Constants.shortOffsetDelay,
339 self.Constants.shortOffsetZ0,self.Constants.shortOffsetLoss,
340 self.Constants.f0,self.Constants.shortL0,self.Constants.shortL1,
341 self.Constants.shortL2,self.Constants.shortL3)
342 self.loadStandard=LoadStandard(self.m_f,self.Constants.loadOffsetDelay,
343 self.Constants.loadOffsetZ0,self.Constants.loadOffsetLoss,
344 self.Constants.f0,self.Constants.loadZ)
345 self.thruStandard=ThruStandard(self.m_f,self.Constants.thruOffsetDelay,
346 self.Constants.thruOffsetZ0,self.Constants.thruOffsetLoss,
347 self.Constants.f0)
348 return self
349 def ReadFromFile(self,filename):
350 """Reads the calibration constants from a file
351 @param filename string name of file to read calibration constants from
352 @return self
353 @see CalibrationConstants.ReadFromFile()
354 @see CalibrationConstants for file format
355 """
356 self.Constants=CalibrationConstants().ReadFromFile(filename)
357 return self

368 def WriteStandardsToFiles(self,filenamePrefix=''):
369 """Writes the standards to s-parameter files.
370 @param filenamePrefix string name of prefix for standards file names
371 @return self
372 Each standard is prefixed with the filenamePrefix provided concatenated with:
373 - 'Short.s1p' - for the short standard.
374 - 'Open.s1p' - for the open standard.
375 - 'Load.s1p' - for the load standard.
376 - 'Thru.s2p' - for the thru standard
377 @attention the member variables for the standards must exist.
378 @see shortStandard
379 @see openStandard
380 @see loadStandard
381 @see thruStandard
382 """
383 self.shortStandard.WriteToFile(filenamePrefix+'Short')
384 self.openStandard.WriteToFile(filenamePrefix+'Open')
385 self.loadStandard.WriteToFile(filenamePrefix+'Load')
386 self.thruStandard.WriteToFile(filenamePrefix+'Thru')
387 return self
388 def ReadStandardsFromFiles(self,filenamePrefix):
389 """Reads the standards from s-parameter files.
390 @param filenamePrefix string name of prefix for standards file names
391 @return self
392 Each standard is prefixed with the filenamePrefix provided concatenated with:
393 - 'Short.s1p' - for the short standard.
394 - 'Open.s1p' - for the open standard.
395 - 'Load.s1p' - for the load standard.
396 - 'Thru.s2p' - for the thru standard
397 @attention This decouples the standards from the calibration constants and is not preferred.
398 @see shortStandard
399 @see openStandard
400 @see loadStandard
401 @see thruStandard
402 """
403 self.shortStandard=SParameterFile(filenamePrefix+'Short.s1p')
404 self.openStandard=SParameterFile(filenamePrefix+'Open.s1p')
405 self.loadStandard=SParameterFile(filenamePrefix+'Load.s1p')
406 self.thruStandard=SParameterFile(filenamePrefix+'Thru.s2p')
407 return self
