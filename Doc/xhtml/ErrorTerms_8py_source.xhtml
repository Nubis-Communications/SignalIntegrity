27class ErrorTerms(object):
28 """Error terms for VNA and TDR based s-parameter calculations.
29 Error terms are, for P ports, a P x P matrix of lists of three error terms.
30 For the diagonal elements, the three error terms are ED, ER, and ES in that order
31 for the off diagonal elements, the three error terms are EX, ET and EL in that order
32 for r in 0...P-1, and c in 0...P-1, ET[r][c] = [ED[r],ER[r],ES[r]], when r==c
33 ET[r][c]=[EX[r][c],ET[r][c],EL[r][c]] when r !=c
34 ET[r][c] refers to the error terms at port r when driven at port c
35 in other words, if r==c, then:
36 ET[r][r][0] = EDr
37 ET[r][r][1] = ERr
38 ET[r][r][2] = ESr
39 and when r!=c, then:
40 ET[r][c][0]=EXrc
41 ET[r][c][1]=ETrc
42 ET[r][c][2]=ELrc
43 """
44 def __init__(self,ET=None):
45 """Constructor
46 @param ET (optional) instance of class ErrorTerms
47 """
48 self.ET=ET
49 if not ET is None:
50 self.numPorts=len(ET)
51 else:
52 self.numPorts=None
53 def Initialize(self,numPorts):
54 """Initialize
55 Initializes the number of ports and all of the three error terms for
56 each row and column of the error terms to zero.
57 @param numPorts integer number of ports for the error terms
58 @return self
59 """
60 self.numPorts=numPorts
61 self.ET=[[[None,None,None] for _ in range(self.numPorts)]
62 for _ in range(self.numPorts)]
63 return self
64 def CheckErrorTerms(self):
65 for other in range(self.numPorts):
66 for driven in range(self.numPorts):
67 if any(self.ET[other][driven][t]==None
68 for t in range(len(self.ET[other][driven]))):
69 raise SignalIntegrityExceptionMeasurement('Not all error terms calculated')
70 def InitializeFromFixtures(self,fixtureList):
71 """Initialize from list of fixtures
72 @param fixtureList list of list of list s-parameter fixture matrices
73 @return self
74
75 The number of matrices is the number of ports P and each matrix must be
76 2P x 2P corresponding to the fixture format
77 """
78 self.Initialize(len(fixtureList))
79 for r in range(len(fixtureList)):
80 for c in range(len(fixtureList)):
81 if r==c:
82 self.ET[r][r][0]=fixtureList[r][r][r]
83 self.ET[r][r][1]=fixtureList[r][r][r+self.numPorts]
84 self.ET[r][r][2]=fixtureList[r][r+self.numPorts][r+self.numPorts]
85 else:
86 self.ET[r][c][0]=fixtureList[c][r][c]
87 self.ET[r][c][1]=fixtureList[c][r][r+self.numPorts]
88 self.ET[r][c][2]=fixtureList[c][r+self.numPorts][r+self.numPorts]
89 return self
90 def __getitem__(self,item):
91 """overloads [item]
92 @param item integer row of the error term matrix to access
93 @remark This is typically used to access an error term where self[o][d][i]
94 would access the ith error term for port o with port d driven.
95 """
96 return self.ET[item]
97 def ReflectCalibration(self,hatGamma,Gamma,m):
98 """performs a reflect calibration
99 Computes the directivity, reverse transmission, and source match terms
100 for a given port and frequency from a list of measurements and actual standard
101 values and updates itself.
102 @param hatGamma list of complex measurements of reflect standards
103 @param Gamma list of complex actual values of the reflect standards
104 @param m integer index of port
105 @return self
106 """
107 # pragma: silent exclude
108 if len(Gamma)<3: raise SignalIntegrityExceptionMeasurement('Insufficient reflect measurements')
109 # pragma: include
110 A=[[1.,Gamma[r]*hatGamma[r],-Gamma[r]] for r in range(len(Gamma))]
111 B=[[hatGamma[r]] for r in range(len(Gamma))]
112 EdEsDeltaS=(pinv(array(A)).dot(array(B))).tolist()
113 Ed=EdEsDeltaS[0][0]
114 Es=EdEsDeltaS[1][0]
115 DeltaS=EdEsDeltaS[2][0]
116 Er=Ed*Es-DeltaS
117 self[m][m]=[Ed,Er,Es]
118 return self
119 def ThruCalibration(self,b1a1,b2a1,S,n,m):
120 """performs a thru calibration
121 Computes the forward transmission and load match terms
122 for a given driven and undriven port and frequency from a list of measurements and actual
123 standard values and updates itself.
124 @param b1a1 list or single complex value for ratio of reflect to incident at driven port.
125 @param b2a1 list or single complex value for ratio of reflect to incident at undriven port.
126 @param S list or single list of list matrix representing s-parameters of thru standard
127 @param n integer index of undriven port
128 @param m integer index of driven port
129 @return self
130 """
131 # pragma: silent exclude
132 if not isinstance(b1a1,list):
133 b1a1=[b1a1]
134 b2a1=[b2a1]
135 S=[S]
136 # pragma: include
137 [Ed,Er,Es]=self[m][m]
138 Ex=self[n][m][0]
139 # pragma: silent exclude
140 if any(errorterm==None for errorterm in [Ed,Er,Es,Ex]):
141 raise SignalIntegrityExceptionMeasurement('Thru calibration with missing reflect terms')
142 # pragma: include
143 A=zeros((2*len(b1a1),2)).tolist()
144 B=zeros((2*len(b1a1),1)).tolist()
145 for i in range(len(b1a1)):
146 Sm=S[i]
147 detS=det(array(Sm))
148 A[2*i][0]=(Es*detS-Sm[1][1])*(Ed-b1a1[i])-Er*detS
149 A[2*i][1]=0.
150 A[2*i+1][0]=(Es*detS-Sm[1][1])*(Ex-b2a1[i])
151 A[2*i+1][1]=Sm[1][0]
152 B[2*i][0]=(1.-Es*Sm[0][0])*(b1a1[i]-Ed)-Er*Sm[0][0]
153 B[2*i+1][0]=(1-Es*Sm[0][0])*(b2a1[i]-Ex)
154 ElEt=(pinv(array(A)).dot(array(B))).tolist()
155 (El,Et)=(ElEt[0][0],ElEt[1][0])
156 self[n][m]=[Ex,Et,El]
157 return self
158 def UnknownThruCalibration(self,Sm,Sest,firstPort,secondPort):
159 """Computes the unknown thru
160 for a given set of measurements of a thru and an estimate of the thru the actual value of the
161 thru is calculated.
162 @param Sm list of list raw sparameter measurement of the thru.
163 @param Sest list of list estimate of the thru
164 @param firstPort integer zero based port number of port 1 of thru standard
165 @param secondPort integer zero based port number of port 2 of thru standard
166 @return list of list sparameter value of the thru
167 @remark normally this algorithm is used to compute the error terms directly (which are actually
168 calculated here), but only the recovered thru value is returned. This is so that an actual thru
169 measurement can be created with this thru value. If the thru measurement is provided with this
170 calculated value of the thru, the error terms will be recalculated as calculated here, but returning
171 the value of the thru allows the estimate to be checked for validity, and more importantly, allows
172 for multiple thru measurements to be provided to allow for overconstrained, better calibrations.
173 Additionally, the s-parameters of the entire thru measurement (not just at a single frequency) can be
174 impulse response limited to provide an even more improved estimate of the thru.
175 """
176 # pragma: silent exclude
177 # comment this in for debugging - it allows one to compare ET and EL calculated here with what it would
178 # be with a known thru calibration
179# self.ThruCalibration(Sm[0][0], Sm[1][0],Sest,secondPort, firstPort)
180# self.ThruCalibration(Sm[1][1], Sm[0][1],[[Sest[1][1],Sest[1][0]],[Sest[0][1],Sest[0][0]]],firstPort,secondPort)
181 # pragma: include
182 [ED1,ER1,ES1]=self[firstPort][firstPort]
183 [ED2,ER2,ES2]=self[secondPort][secondPort]
184 [EX12,ET12,EL12]=self[firstPort][secondPort]
185 [EX21,ET21,EL21]=self[secondPort][firstPort]
186 # pragma: silent exclude
187 if any(errorterm==None for errorterm in [ED1,ER1,ES1,ED2,ER2,ES2,EX12,EX21]):
188 raise SignalIntegrityExceptionMeasurement('Unknown thru calibration with missing reflect terms')
189 # pragma: include
190 p=cmath.sqrt((Sm[0][1]-EX12)/(Sm[1][0]-EX21))
191 [EX12,ET12,EL12]=[EX12,cmath.sqrt(ER1)*cmath.sqrt(ER2)*p,ES1]
192 [EX21,ET21,EL21]=[EX21,cmath.sqrt(ER1)*cmath.sqrt(ER2)/p,ES2]
193 DutCalc1=ErrorTerms([[[ED1,ER1,ES1],[EX12,ET12,EL12]],
194 [[EX21,ET21,EL21],[ED2,ER2,ES2]]]).DutCalculation(Sm)
195 DutCalc2=ErrorTerms([[[ED1,ER1,ES1],[EX12,-ET12,EL12]],
196 [[EX21,-ET21,EL21],[ED2,ER2,ES2]]]).DutCalculation(Sm)
197 if norm(array(DutCalc1)-array(Sest)) < norm(array(DutCalc2)-array(Sest)):
198 return DutCalc1
199 else:
200 return DutCalc2

206 def ExCalibration(self,b2a1,n,m):
207 """Computes the crosstalk term
208 For a given driven and undriven port and frequency from a list of measurements and actual
209 standard values and updates itself.
210 @param b2a1 single complex value for ratio of reflect to incident at undriven port.
211 @param n integer index of undriven port
212 @param m integer index of driven port
213 @return self
214 """
215 [_,Et,El]=self[n][m]
216 Ex=b2a1
217 self[n][m]=[Ex,Et,El]

219 def TransferThruCalibration(self):
220 """Performs the transfer thru calibrations.
221 After all of the thru calibration calculations have been performed, it looks to see if there
222 are any port combinations where a thru was not connected and attempts to perform the 'transfer
223 thru' calibration that uses other thru measurements to form the thru calibration for a given
224 port combination."""
225 didOne=True
226 while didOne:
227 didOne=False
228 for otherPort in range(self.numPorts):
229 for drivenPort in range(self.numPorts):
230 if (otherPort == drivenPort):
231 continue
232 if ((self[otherPort][drivenPort][1]==None) and
233 (self[otherPort][drivenPort][2]==None)):
234 for mid in range(self.numPorts):
235 if ((mid != otherPort) and
236 (mid != drivenPort) and
237 ((self[otherPort][mid][1]!=None) or
238 (self[otherPort][mid][2]!=None)) and
239 ((self[mid][drivenPort][1]!=None) or
240 (self[mid][drivenPort][2]!=None))):
241 (_,Etl,_)=self[otherPort][mid]
242 (_,Etr,_)=self[mid][drivenPort]
243 (_,Erm,_)=self[mid][mid]
244 (_,_,Eso)=self[otherPort][otherPort]
245 (Ex,Et,El)=self[otherPort][drivenPort]
246 Et=Etl*Etr/Erm
247 El=Eso
248 self[otherPort][drivenPort]=[Ex,Et,El]
249 didOne=True
250 continue
251 return self
252 def Fixture(self,m,pl=None):
253 """Fixture
254 For a P port measurement, the s-parameters are for a 2*P port
255 fixture containing the error terms when port m driven going between
256 the insrument ports and the DUT ports where
257 the first P ports are the instrument port connections and the remaining
258 ports connect to the DUT.
259 @param m integer driven port
260 @param pl (optional) list of zero based port numbers of the DUT
261 @return a list of list s-parameter matrix
262 @remark If the portList is None, then it assumed to be a list [0,1,2,P-1] where P is the
263 number of ports in sRaw, otherwise ports can be specified where the DUT is connected.
264 m is assumed to be in integer zero based index in the ports list pl
265 """
266 if pl is None: pl = [p for p in range(self.numPorts)]
267 numPorts=len(pl)
268 E=[[zeros((numPorts,numPorts),complex).tolist(),
269 zeros((numPorts,numPorts),complex).tolist()],
270 [zeros((numPorts,numPorts),complex).tolist(),
271 zeros((numPorts,numPorts),complex).tolist()]]
272 for n in range(numPorts):
273 ETn=self[pl[n]][pl[m]]
274 E[0][0][n][m]=ETn[0]
275 E[0][1][n][n]=ETn[1]
276 E[1][1][n][n]=ETn[2]
277 E[1][0][m][m]=1.
278 return E
279 def DutCalculationAlternate(self,sRaw,pl=None,reciprocal=False):
280 """Alternate Dut Calculation
281 @deprecated This provides a DUT calculation according to the Wittwer method,
282 but a better,simpler method has been found.
283 @param sRaw list of list s-parameter matrix of raw measured DUT
284 @param pl (optional) list of zero based port numbers of the DUT
285 @param reciprocal (optional, defaults to False) whether to enforce reciprocity
286 @return list of list s-parameter matrix of calibrated DUT measurement
287 @remark if reciprocity is True, the reciprocity is enforced in the calculation
288 @see _EnforceReciprocity
289 @see DutCalculation
290 """
291 if pl is None: pl = [p for p in range(len(sRaw))]
292 numPorts=len(pl)
293 if numPorts==1:
294 (Ed,Er,Es)=self[pl[0]][pl[0]]
295 gamma=sRaw[0][0]
296 Gamma=(gamma-Ed)/((gamma-Ed)*Es+Er)
297 return [[Gamma]]
298 else:
299 A=zeros((numPorts,numPorts),complex).tolist()
300 B=zeros((numPorts,numPorts),complex).tolist()
301 I=(identity(numPorts)).tolist()
302 for m in range(numPorts):
303 E=self.Fixture(m,pl)
304 b=array([[sRaw[r][m]] for r in range(numPorts)])
305 Im=array([[I[r][m]] for r in range(numPorts)])
306 bprime=(pinv(array(E[0][1])).dot(b-array(E[0][0]).dot(Im))).tolist()
307 aprime=(array(E[1][0]).dot(Im)+array(E[1][1]).dot(array(bprime))).tolist()
308 for r in range(numPorts):
309 A[r][m]=aprime[r][0]
310 B[r][m]=bprime[r][0]
311 if not reciprocal: S=(array(B).dot(pinv(array(A)))).tolist()
312 else: S=self._EnforceReciprocity(A,B)
313 return S
314 def DutCalculation(self,sRaw,pl=None,reciprocal=False):
315 """Calculates a DUT
316 @param sRaw list of list s-parameter matrix of raw measured DUT
317 @param pl (optional) list of zero based port numbers of the DUT
318 @param reciprocal (optional, defaults to False) whether to enforce reciprocity
319 @return list of list s-parameter matrix of calibrated DUT measurement
320 @remark This provides a newer, simpler DUT calculation
321 @remark If the portList is None, then it assumed to be a list [0,1,2,P-1] where P is the
322 number of ports in sRaw, otherwise ports can be specified where the DUT is connected.
323 @remark if reciprocity is True, the reciprocity is enforced in the calculation
324 @see _EnforceReciprocity
325 @see DutCalculationAlternate
326 """
327 P=len(sRaw); Pr=range(P)
328 if pl is None: pl = [p for p in Pr]
329 B=[[(sRaw[r][c]-self[pl[r]][pl[c]][0])/self[pl[r]][pl[c]][1]
330 for c in Pr] for r in Pr]
331 A=[[B[r][c]*self[pl[r]][pl[c]][2]+(1 if r==c else 0) for c in Pr] for r in Pr]
332 if not reciprocal: S=(array(B).dot(pinv(array(A)))).tolist()
333 else: S=self._EnforceReciprocity(A,B)
334 return S
335 def _EnforceReciprocity(self,A,B):
336 """Given S*A=B, Calculates a DUT in S enforcing reciprocity
337 @param A list of list matrix
338 @param B list of list matrix
339 @return list of list s-parameter matrix of calibrated DUT measurement with reciprocity enforced
340 @see DutCalculation, DutCalculationAlternate
341 """
342 P=len(A); Pr=range(P); M=[[None for _ in Pr] for _ in Pr]
343 for c in Pr:
344 for r in Pr:
345 M[r][c]=M[c][r] if r < c else r-c+(0 if c==0 else M[P-1][c-1]+1)
346 L=[[0. for c in range(P*(P+1)//2)] for r in range(P*P)]
347 b=[None for r in range(P*P)]
348 for r in Pr:
349 for c in Pr:
350 b[r*P+c]=[B[r][c]]
351 for p in Pr: L[p*P+r][M[p][c]]=A[c][r]
352 sv=(pinv(array(L)).dot(array(b))).tolist()
353 S=[[sv[M[r][c]][0] for c in Pr] for r in Pr]
354 return S
355 def DutUnCalculation(self,S,pl=None):
356 """undoes the DUT calculation
357 @param S list of list s-parameter matrix a DUT
358 @param pl (optional) list of zero based port numbers of the DUT
359 @return list of list s-parameter matrix of raw measured s-parameters
360 @remark If the portList is None, then it assumed to be a list [0,1,2,P-1] where P is the
361 number of ports in sRaw, otherwise ports can be specified where the DUT is connected.
362 """
363 if pl is None: pl = [p for p in range(len(S))]
364 Sp=[[None for c in range(len(S))] for r in range(len(S))]
365 Si=pinv(array(S))
366 for c in range(len(S)):
367 E=self.Fixture(c,pl)
368 Em=[[array(E[0][0]),array(E[0][1])],[array(E[1][0]),array(E[1][1])]]
369 col=(Em[0][0].dot(Em[1][0])+Em[0][1].dot(
370 pinv(Si-Em[1][1])).dot(Em[1][0])).tolist()
371 for r in range(len(S)): Sp[r][c]=col[r][c]
372 return Sp
