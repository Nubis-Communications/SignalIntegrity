24class MCB(SParameters):
25 """s-parameters of four-port MCB transmission line built from two two-port lines
26 @note MCB is a Module Compliance Board as defined in the IEEE 802.3 standards."""
27 def __init__(self,f,Td=200e-12,lp=[1,2],rp=[3,4],Z0=50.):
28 """Constructor
29 @param f list of float frequencies
30 @param Td float electrical length (propagation time)
31 @param lp list of ints left ports in order (top to bottom)
32 @param rp list of ints right ports in order (top to bottom)
33 @param Z0 (optional) float or complex reference impedance (defaults to 50 ohms)
34 """
35 self.m_lp=lp
36 self.m_rp=rp
37 eq=('10**(-(0.0912+0.3102*math.sqrt(f/1e9)+0.008578*(f/1e9)'
38 '+0.000759*(f/1e9)**2)/20.)*cmath.exp(-j*2.*pi*f*'+repr(float(Td))+')')
39 self.m_line1=TransmissionLineEquation(f,eq,Z0)
40 self.m_line2=TransmissionLineEquation(f,eq,Z0)
41 SParameters.__init__(self,f,None,Z0)
42
43 def __getitem__(self,n):
44 """overloads [n]
45 @return list of list s-parameter matrix for the nth frequency element
46 """
47 # upper pair uses canonical ports 1,3; lower pair uses canonical ports 2,4
48 h1=self.m_line1[n]
49 h2=self.m_line2[n]
50 sc=[[0. for _ in range(4)] for _ in range(4)]
51 sc[0][0]=h1[0][0]; sc[0][2]=h1[0][1]; sc[2][0]=h1[1][0]; sc[2][2]=h1[1][1]
52 sc[1][1]=h2[0][0]; sc[1][3]=h2[0][1]; sc[3][1]=h2[1][0]; sc[3][3]=h2[1][1]
53 order=self.m_lp+self.m_rp
54 if len(order) != 4 or sorted(order) != [1,2,3,4]:
55 raise IndexError('port number incorrect')
56 inverse=[0]*4
57 for canonical_index,external_port in enumerate(order):
58 inverse[external_port-1]=canonical_index
59 return [[sc[inverse[r]][inverse[c]] for c in range(4)] for r in range(4)]
