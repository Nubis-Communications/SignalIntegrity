24class SimpleIIRFilter(object):
25 def __init__(self,timeconstant):
26 self.coef=1.-math.exp(-1./timeconstant)
27 self.first=True
28 self.output=0.
29 self.delta=0.
30 def Output(self,inputValue=None):
31 if inputValue is None:
32 return self.output
33 if self.first:
34 self.first=False
35 newOutput=inputValue
36 self.output=inputValue
37 else:
38 newOutput=inputValue*self.coef+self.output*(1-self.coef)
39 self.delta=newOutput-self.output
40 self.output=newOutput
41 return self.output
42 def Delta(self):
43 return self.delta

45class TwoTapBoxcarFilter(object):
46 def __init__(self):
47 self.z0=0.
48 self.z1=0.
49 self.output=0.
50 def Output(self,inputValue=None):
51 if inputValue is None:
52 return self.output
53 self.z1=self.z0
54 self.z0=inputValue
55 output=(self.z0+self.z1)/2.
56 self.delta=output-self.output
57 self.output=output
58 return self.output
59 def Delta(self):
60 return self.delta

143 def Continue(self):
144 IterationsExhausted = (self._IterationsTaken >= self._maxIterations)
145
146 MseToleranceMet = (self._Mse < self._tolerance)
147
148 if (self._IterationsTaken > self._lambdaTimeConstant):
149 LambdaFilterThresholdMet = self._FilteredLogDeltaLambda < self._lambdaUnchanging
150 else: LambdaFilterThresholdMet = False
151
152 if (self._IterationsTaken > self._mseTimeConstant):
153 MseFilterThresholdMet = self._FilteredLogDeltaMse < self._mseUnchanging
154 else: MseFilterThresholdMet = False
155
156 lambdaOkay=True
157
158 Stop = (
159 IterationsExhausted or
160 (not lambdaOkay) or
161 (LambdaFilterThresholdMet and MseFilterThresholdMet) or
162 MseToleranceMet
163 )
164
165 if Stop:
166 if IterationsExhausted:
167 self.why = 'iterations exhausted'
168 elif (not lambdaOkay):
169 self.why = 'lamda not okay'
170 elif (LambdaFilterThresholdMet and MseFilterThresholdMet):
171 self.why = 'filtered delta lamda and mse both went below threshold'
172 elif MseToleranceMet:
173 self.why = 'mse tolerance achieved'
174 else:
175 self.why = 'unknown: this is an error'
176
177 return not Stop
