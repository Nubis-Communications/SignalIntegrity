22class CTLE(SParameters):
23 """Continuous time linear equalizer"""
24 def __init__(self,f,gDC,gDC2,fz,fLF,fp1,fp2,Z0=50.):
25 """Constructor
26 @param f list of float frequencies
27 @param gDC float DC gain in dB
28 @param gDC2 float DC gain at peaking point in dB
29 @param fz float frequency of zero
30 @param fLF float LF frequency
31 @param fp1 float first pole frequency
32 @param fp2 float second pole frequency
33 @param Z0 float (optional, defaults to 50) reference impedance
34 """
35 self.powgDC=pow(10.,gDC/20.); self.powgDC2=pow(10.,gDC2/20.)
36 self.joverfz=1j/fz; self.joverfLF=1j/fLF; self.joverfp1=1j/fp1; self.joverfp2=1j/fp2
37 SParameters.__init__(self,f,None,Z0)
38 def _Evaluate(self,f):
39 """Evaluate CTLE
40 @param f float frequency
41 @return CTLE evaluated at f
42 """
43 N=(self.powgDC+self.joverfz*f)*(self.powgDC2+self.joverfLF*f)
44 D=(1.+self.joverfp1*f)*(1.+self.joverfp2*f)*(1.+self.joverfLF*f)
45 return N/D
46 def __getitem__(self,n):
47 return [[1.,0],[2.*self._Evaluate(self.m_f[n]),-1.]]
