23def SeriesC(C,f,Z0=None,df=0.,esr=0.):
24 """Series Capacitance
25 @param C float capacitance
26 @param f float frequency
27 @param Z0 (optional) float of complex reference impedance (defaults to 50 ohms)
28 @param df (optional) float dissipation factor (or loss-tangent) (defaults to 0)
29 @param esr (optional) float effective-series-resistance (defaults to 0)
30 @return the list of list s-parameter matrix for a series capacitance
31 """
32 G=C*2.*np.pi*f*(1j+df)
33 try: G=1./(1./G+esr)
34 except: pass
35 return SeriesG(G,Z0)
