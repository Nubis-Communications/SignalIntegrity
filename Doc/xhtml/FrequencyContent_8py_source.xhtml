33class FrequencyContent(FrequencyDomain):
34 """Handles frequency content of waveforms.
35 This is the frequency content view of a waveform. In other words, it assumes that a waveform is an actual waveform and
36 contains the complex values of sinusoids that, if added together, would make up the waveform. This is the
37 opposite of the FrequencyResponse() view.
38 @see FrequencyResponse
39 """
40 R=50.0
41 P=1e-3
42 LogRP10=10.*math.log10(R*P)
43 dB3=20*math.log10(math.sqrt(2))
44 dB6=20*math.log10(0.5)
45 def __init__(self,wf,fd=None):
46 """Constructor
47 @param wf in instance of class Waveform
48 @param fd (optional) an instance of class FrequencyList (defaults to None)
49 @remark initializes itself internally by computing the frequency content of the waveform.
50
51 If fd is None then the frequency descriptor is simply the frequency descriptor corresponding to the time
52 descriptor of the waveform and the frequency content is computed from the DFT.
53
54 Otherwise, the CZT is used to compute the frequency content and the time descriptor corresponds to the
55 frequency descriptor.
56
57 the time descriptor and frequency descriptor are retained so a waveform can be obtained from the frequency content.
58
59 @note the frequency content is scaled differently from the raw DFT or CZT outputs in that the absolute value of each
60 complex number in the frequency content represents the amplitude of a cosine wave. This is not true with the raw
61 DFT output and scaling things this way helps in the proper interpretation of the frequency content without having
62 to think about the vagaries of the DFT.
63
64 @see TimeDescriptor
65 @see FrequencyList
66 @see ChirpZTransform
67 """
68 # pragma: silent exclude
69 from SignalIntegrity.Lib.TimeDomain.Waveform.TimeDescriptor import TimeDescriptor
70 # pragma: include
71 td=wf.td
72 if fd is None:
73 X=fft.fft(wf.Values())
74 K=int(td.K)
75 Keven = DFTUtilities.Keven(td.K)
76 fd=td.FrequencyList()
77 else:
78 # pragma: silent exclude
79 if not fd.EvenlySpaced():
80 raise SignalIntegrityExceptionWaveform('cannot generate frequency content')
81 # pragma: include
82 K=fd.N*2
83 Keven=True
84 X=CZT(wf.Values(),td.Fs,0,fd.Fe,fd.N,True)
85 td=TimeDescriptor(td.H,fd.N*2,fd.Fe*2.)
86 FrequencyDomain.__init__(self,fd,[X[n]/K*\
87 (1. if (n==0 or ((n==fd.N) and Keven)) else 2.)*\
88 cmath.exp(-1j*2.*math.pi*fd[n]*td.H) for n in range(fd.N+1)])
89 self.td=td
90 def Values(self,unit=None):
91 """frequency content values
92 @param unit (optional) string containing the unit for the values desired.
93 @return a list of complex values representing the frequency content.
94 @remark
95 Valid frequency content units are:\n
96 - 'rms' - the root-mean-squared (rms) value.
97 - 'dBm' - the values in decibels were 0 dBm corresponds to the voltage needed to deliver
98 1 mW to a 50 ohm load. It's computed as 20*Log(rms)+13.010.
99 - 'dBmPerHz' - the spectral density in dBm/Hz.
100
101 If no unit is specified, the complex frequency content is returned.
102 If no valid frequency content units are found, then it defers to the FrequencyDomain base class.
103
104 @see FrequencyDomain.
105 """
106 if unit=='rms':
107 Keven = DFTUtilities.Keven(self.td.K)
108 A=FrequencyDomain.Values(self,'mag')
109 return [A[n]/(1 if (n==0 or ((n==self.m_fm_f.N) and Keven))
110 else math.sqrt(2)) for n in range(len(A))]
111 elif unit=='dBm':
112 return [-3000. if r < 1e-15 else 20.*math.log10(r)-self.LogRP10
113 for r in self.ValuesValues('rms')]
114 elif unit=='dBmPerHz':
115 Keven = DFTUtilities.Keven(self.td.K)
116 Deltaf=self.m_fm_f.Fe/self.m_fm_f.N
117 adder=-10*math.log10(Deltaf)
118 dBm=self.ValuesValues('dBm')
119 return [dBm[n]+adder+
120 (self.dB3 if (n==0 or ((n==self.m_fm_f.N) and Keven))
121 else 0) for n in range(len(dBm))]
122 # pragma: silent exclude
123 elif unit=='VPerRootHz':
124 Keven = DFTUtilities.Keven(self.td.K)
125 Deltaf=self.m_fm_f.Fe/self.m_fm_f.N
126 A = FrequencyDomain.Values(self,'mag')
127 rms = DFTUtilities.A_to_rms(A,Keven)
128 rho = DFTUtilities.rms_to_rho(rms,Deltaf,Keven)
129 return rho
130 # pragma: include
131 else: return FrequencyDomain.Values(self,unit)

138 def Waveform(self,td=None):
139 """Computes the time-domain waveform using IDFT methods
140 @param td (optional) instance of class TimeDescriptor declaring the time descriptor of the waveform to produce.
141 @return wf instance of class Waveform corresponding to the frequency content.
142 @note
143 If td is None then the time descriptor corresponding to the frequency descriptor is used.\n
144 The waveform produced is essentially the inverse process of class initialization.\n
145 @see WaveformFromDefinition()
146 """
147 # pragma: silent exclude
148 from SignalIntegrity.Lib.TimeDomain.Waveform.Waveform import Waveform
149 # pragma: include
150 Keven = DFTUtilities.Keven(self.td.K)
151 X=self.ValuesValues()
152 X=[X[n]*self.td.K*\
153 (1. if (n==0 or ((n==self.m_fm_f.N) and Keven)) else 0.5)*\
154 cmath.exp(1j*2.*math.pi*self.m_fm_f[n]*self.td.H)
155 for n in range(self.m_fm_f.N+1)]
156 if Keven:
157 X2=[X[self.m_fm_f.N-n].conjugate() for n in range(1,self.m_fm_f.N)]
158 else:
159 X2=[X[self.m_fm_f.N-n+1].conjugate() for n in range(1,self.m_fm_f.N+1)]
160 X.extend(X2)
161 x=[xk.real for xk in fft.ifft(X).tolist()]
162 wf=Waveform(self.td,x)
163 if not td is None:
164 wf=wf.Adapt(td)
165 return wf
166 def WaveformFromDefinition(self,td=None):
167 """Computes the time-domain waveform using sums of cosines
168 @param td instance of class TimeDescriptor declaring the time descriptor of the waveform to produce.
169 @return wf instance of class Waveform corresponding to the frequency content.
170 @note
171 If td is None then the time descriptor corresponding to the frequency descriptor is used.\n
172 The waveform produced is essentially the inverse process of __init__().\n
173 This function should produce the exact same result as the Waveform() method, and is slow, but clearly
174 written out to see how the waveform is produced by summing sinusoids. It used to essentially document
175 the class.\n
176 @see Waveform().
177 """
178 # pragma: silent exclude
179 from SignalIntegrity.Lib.TimeDomain.Waveform.Waveform import Waveform
180 from SignalIntegrity.Lib.TimeDomain.Waveform.SineWaveform import SineWaveform
181 # pragma: include
182 absX=self.ValuesValues('mag')
183 theta=self.ValuesValues('deg')
184 wf=Waveform(self.td)
185 for n in range(self.m_fm_f.N+1):
186 wf=wf+SineWaveform(self.td,Frequency=self.m_fm_f[n],
187 Amplitude=absX[n],Phase=theta[n]+90)
188 if not td is None:
189 wf=wf.Adapt(td)
190 return wf
