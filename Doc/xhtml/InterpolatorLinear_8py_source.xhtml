25class FractionalDelayFilterLinear(FirFilter):
26 """linear fractional delay filter"""
27 def __init__(self,F,accountForDelay=True):
28 """Constructor
29 applies a two-tap linear interpolating filter.
30 @param F float amount of delay to apply. The delay is in samples of the input waveform.
31 @param accountForDelay (optional) boolean whether to account for the delay
32 @remark if accountForDelay, then the filter provides a sample phase adjustment, meaning
33 that there is no actual delay applied to the waveform, but the time axis under
34 the waveform is shifted. This is the usual way to apply this filter and is used
35 to adapt waveforms on different time axes to each other.\n
36 if not accountForDelay, then the filter actually delays waveforms by the delay
37 specified.
38 """
39 # pragma: silent exclude
40 from SignalIntegrity.Lib.TimeDomain.Filters.FilterDescriptor import FilterDescriptor
41 # pragma: include
42 FirFilter.__init__(self,FilterDescriptor(1,
43 (F if F >= 0 else 1+F) if accountForDelay else 0,1),
44 [1-F,F] if F >= 0 else [-F,1+F])

46class InterpolatorLinear(FirFilter):
47 """linear interpolating filter"""
48 def __init__(self,U):
49 """Constructor
50 applies a linear interpolating filter.
51 @param U integer upsample factor of the filter.
52 """
53 # pragma: silent exclude
54 from SignalIntegrity.Lib.TimeDomain.Filters.FilterDescriptor import FilterDescriptor
55 # pragma: include
56 FirFilter.__init__(self,
57 FilterDescriptor(U,(U-1.)/float(U),2*(U-1.)/float(U)),
58 [float(u+1)/float(U) for u in range(U)]+
59 [1-float(u+1)/float(U) for u in range(U-1)])
60 def FilterWaveform(self,wf):
61 """overloads base class FilterWaveform
62 @param wf instance of class Waveform
63 @return instance of class Waveform containing the upsampled, interpolated wf
64 @remark
65 This method first classically upsamples the waveform by inserting zeros
66 between the samples and then passes the upsampled waveform through the linear
67 interpolation filter.
68 """
69 # pragma: silent exclude
70 from SignalIntegrity.Lib.TimeDomain.Waveform.Waveform import Waveform
71 # pragma: include
72 fd=self.FilterDescriptor()
73 us=[0. for k in range(len(wf)*fd.U)]
74 for k in range(len(wf)):
75 us[k*fd.U]=wf[k]
76 return FirFilter.FilterWaveform(self,Waveform(wf.td,us))
