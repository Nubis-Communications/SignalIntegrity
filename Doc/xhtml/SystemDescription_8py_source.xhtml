33 Often, this class will be used to construct the system description and

36 A system description is fundamentally a list of devices, which in turn are lists

159 def AddPort(self,DeviceName,DevicePort,SystemPort,AddThru=False):
160 """Adds a system port at a specified device port
161 @param DeviceName string name of device
162 @param DevicePort integer port number of device
163 @param SystemPort integer port number
164 @param AddThru boolean whether to force a thru between the system port and the device port.
165 Adds a system port to a device port for the solutions of s-parameters or for deembedding.
166
167 In the theory, sometimes it's necessary to add a thru between a system port and a device port.
168 But it turns out that this is only necessary when connecting a system port directly to a port
169 of an unknown device in a deembedding problem. Although the ability to force the addition of
170 a thru is retained, the code actually checks for this situation and adds a thru automatically
171 so currently, there are no situations known where AddThru needs to be set True.
172
173 @todo Consider removing the AddThru argument
174 """
175 PortName = self.port_refdes+str(SystemPort)
176 self.AddDevice(PortName,1,[[0.0]])
177 self.AssignM(PortName,1,'m'+str(SystemPort))
178 if not AddThru:
179 AddThru = self[self.IndexOfDevice(DeviceName)].Type == 'unknown'
180 if AddThru:
181 thruName=self.m_UniqueDevice.Name()
182 self.AddDevice(thruName,2,Thru())
183 self.ConnectDevicePort(PortName,1,thruName,1)
184 self.ConnectDevicePort(thruName,2,DeviceName,DevicePort)
185 else:
186 self.ConnectDevicePort(PortName,1,DeviceName,DevicePort)
