25class DFTUtilities(object):
26 """A set of DFT utilities for dealing with the DFT for handling frequency
27 and time-domain information properly.
28
29 @remark
30 All methods are static. They operate on either:
31 - a full conjugate-symmetric DFT vector of length K, or
32 - a single-sided spectrum of length N+1 where N = K // 2.
33
34 The @a Keven flag indicates the parity of the time-record length K:
35 - Keven = True : K = 2*N
36 - Keven = False : K = 2*N + 1
37
38 Reference: Pupalaikis, Peter J., S-parameters for Signal Integrity,
39 Cambridge University Press, p. 371.
40 """
41
42 # ---- basic counts / axes -------------------------------------------------
43
44 @staticmethod
45 def N(K):
46 """Number of frequency points minus one.
47 @param K integer time-record length.
48 @return integer N = floor(K/2), the index of the highest frequency bin
49 in a single-sided spectrum.
50 """
51 return K // 2
52
53 @staticmethod
54 def K(N, Keven=True):
55 """Time-record length from the frequency bin count.
56 @param N integer number of frequency bins minus one.
57 @param Keven (optional) bool, True (default) if K is even (K=2N),
58 False if K is odd (K=2N+1).
59 @return integer K = 2*N if Keven else 2*N + 1.
60 @see N
61 """
62 return 2 * N + (0 if Keven else 1)
63
64 @staticmethod
65 def Keven(K):
66 """Parity test for a time-record length K.
67 @param K integer (or integer-valued) time-record length.
68 @return bool, True if K is even, False if odd.
69 @remark K is cast to int first so that callers passing numpy
70 integer/float scalars (e.g. from a TimeDescriptor) get a correct
71 integer parity test rather than an accidental float modulus.
72 @see N
73 @see K
74 """
75 return int(K) % 2 == 0
76
77 @staticmethod
78 def EndFrequency(K, Fs):
79 """Highest frequency of a single-sided spectrum.
80 @param K integer time-record length.
81 @param Fs float sample rate (Hz).
82 @return float Fe = (N/K) * Fs, where N = K//2.
83 @note Equals Fs/2 only when K is even.
84 @see SampleRate
85 """
86 return (K // 2) * Fs / K
87
88 @staticmethod
89 def SampleRate(K, Fe):
90 """Sample rate from the end frequency.
91 @param K integer time-record length.
92 @param Fe float end frequency (Hz).
93 @return float Fs = (K/N) * Fe, where N = K//2.
94 @note Inverse of EndFrequency().
95 @see EndFrequency
96 """
97 return K * Fe / (K // 2)
98
99 @staticmethod
100 def DeltaFrequency(K, Fs):
101 """Frequency bin spacing.
102 @param K integer time-record length.
103 @param Fs float sample rate (Hz).
104 @return float df = Fs / K.
105 """
106 return Fs / K
107
108 @staticmethod
109 def Frequency(n, K, Fs):
110 """Frequency of the n-th DFT bin.
111 @param n integer bin index (0..K-1).
112 @param K integer time-record length.
113 @param Fs float sample rate (Hz).
114 @return float frequency in Hz, f = n * Fs / K.
115 """
116 return n * Fs / K
117
118 # ---- DFT length conversions ---------------------------------------------
119
120 @staticmethod
121 def Full_to_Half(X):
122 """Single-sided spectrum from a full conjugate-symmetric DFT.
123 @param X list of complex values, the full DFT of length K (even or odd).
124 @return list of complex values, the first N+1 = K//2 + 1 entries.
125 @remark
126 A Keven flag is unnecessary because N = K // 2 yields the correct
127 half length for both parities:
128 - K = 2N (Keven=True) -> half length N+1
129 - K = 2N+1 (Keven=False) -> half length N+1
130 @see Half_to_Full
131 """
132 N = len(X) // 2
133 return list(X[:N + 1])
134
135 @staticmethod
136 def Half_to_Full(X,Keven=True):
137 N = len(X) - 1
138 K = DFTUtilities.K(N, Keven)
139 F = [0 for _ in range(K)]
140 for n in range(N + 1):
141 F[n] = X[n]
142 if Keven:
143 for sigma in range(1, N):
144 F[N+sigma] = X[N-sigma].conjugate()
145 else:
146 for sigma in range(1, N+1):
147 F[N+sigma] = X[N-sigma+1].conjugate()
148 return F

151
152 @staticmethod
153 def X_to_A(X, Keven = True):
154 """Single-sided amplitude A[n] from single-sided DFT X[n].
155 @param X list of complex single-sided DFT values, length N+1.
156 @param Keven bool, True if the underlying K is even (K=2N),
157 False if odd (K=2N+1).
158 @return list of float amplitudes A[n] = |X[n]| / K * factor, where
159 factor is 1 for the DC bin (n=0) and (when Keven) the Nyquist bin
160 (n=N), and 2 otherwise.
161 @see AtoX
162 """
163 N = len(X) - 1
164 K = 2 * N if Keven else 2 * N + 1
165 return [abs(X[n]) / K *
166 (1. if (n == 0 or (n == N and Keven)) else 2.)
167 for n in range(N + 1)]
168
169 # ---- amplitude -> DFT ---------------------------------------------------
170
171 @staticmethod
172 def A_to_X(A, Keven = True, random_phase=True):
173 """Single-sided DFT X[n] from amplitude A[n].
174 @param A list of float per-bin amplitudes, length N+1.
175 @param Keven bool, True if the underlying K is even (K=2N),
176 False if odd (K=2N+1).
177 @param random_phase (optional) bool, True (default) to assign each
178 interior bin an independent phase uniformly distributed in
179 [-pi, pi); False to use zero phase (returns real magnitudes).
180 @return list of complex DFT values of length N+1 with
181 |X[n]| = A[n] * K / factor, where factor is 1 for the DC bin (n=0)
182 and (when Keven) the Nyquist bin (n=N), and 2 otherwise.
183 @note The DC bin (n=0) and, when Keven, the Nyquist bin (n=N) are
184 kept real so that Hermitian extension yields a real-valued
185 time-domain signal.
186 @note Inverse of X_to_A().
187 @see X_to_A
188 """
189 N = len(A) - 1
190 K = 2 * N if Keven else 2 * N + 1
191 X = []
192 for n in range(N + 1):
193 factor = 1. if (n == 0 or (n == N and Keven)) else 2.
194 mag = A[n] * K / factor
195 if random_phase and not (n == 0 or (n == N and Keven)):
196 phase = random.uniform(-math.pi, math.pi)
197 else:
198 phase = 0.0
199 X.append(mag * cmath.exp(1j * phase))
200 return X

204 @staticmethod
205 def A_to_rms(A, Keven = True):
206 """RMS values from single-sided amplitudes.
207 @param A list of float amplitudes A[n], length N+1.
208 @param Keven bool, True if the underlying K is even.
209 @return list of float rms values, rms[n] = A[n] for DC and (when
210 Keven) Nyquist bins, else A[n] / sqrt(2).
211 @see rms_to_A
212 """
213 N = len(A) - 1
214 return [A[n] / (1. if (n == 0 or (n == N and Keven)) else math.sqrt(2.))
215 for n in range(N + 1)]
216
217 @staticmethod
218 def rms_to_A(rms, Keven = True):
219 """Single-sided amplitudes from RMS values.
220 @param rms list of float per-bin rms values, length N+1.
221 @param Keven bool, True if the underlying K is even.
222 @return list of float amplitudes, A[n] = rms[n] for DC and (when
223 Keven) Nyquist bins, else rms[n] * sqrt(2).
224 @note Inverse of A_to_rms().
225 @see A_to_rms
226 """
227 N = len(rms) - 1
228 return [rms[n] * (1. if (n == 0 or (n == N and Keven)) else math.sqrt(2.))
229 for n in range(N + 1)]

231 # ---- dBm ----------------------------------------------------------------

233 @staticmethod
234 def rms_to_dBm(rms):
235 """Convert per-bin RMS values to dBm (referenced to 50 ohm, 1 mW).
236 @param rms list of float per-bin rms values (Vrms).
237 @return list of float dBm values, computed as
238 20*log10(rms) - 10*log10(R*P), with R = 50 ohm and P = 1 mW.
239 @note Values below 1e-15 Vrms clamp to -3000 dBm to avoid log of zero.
240 """
241 LogRP10 = 10. * math.log10(50.0 * 1e-3)
242 return [-3000. if r < 1e-15 else 20. * math.log10(r) - LogRP10
243 for r in rms]

247 @staticmethod
248 def rms_to_rho(rms, delta_f, Keven = True):
249 """Spectral density (Vrms/sqrt(Hz)) from per-bin RMS values.
250 @param rms list of float per-bin rms values, length N+1.
251 @param delta_f float bin spacing (Hz).
252 @param Keven bool, True if the underlying K is even.
253 @return list of float spectral densities rho[n]; an extra factor of
254 sqrt(2) is applied at the DC bin (and, when Keven, the Nyquist bin)
255 to account for their one-sided contribution.
256 @see rho_to_rms
257 """
258 N = len(rms) - 1
259 sqrt_df = math.sqrt(delta_f)
260 return [rms[n] *
261 (math.sqrt(2.) if (n == 0 or (n == N)) else 1.) /
262 sqrt_df for n in range(N + 1)]
263
264 @staticmethod
265 def rho_to_rms(rho, delta_f, Keven = True):
266 """Per-bin RMS values from spectral density.
267 @param rho list of float spectral densities (Vrms/sqrt(Hz)), length N+1.
268 @param delta_f float bin spacing (Hz).
269 @param Keven bool, True if the underlying K is even.
270 @return list of float per-bin rms values; an extra 1/sqrt(2) is
271 applied at the DC bin (and, when Keven, the Nyquist bin), undoing
272 the boost applied by rms_to_rho.
273 @note Inverse of rms_to_rho().
274 @see rms_to_rho
275 """
276 N = len(rho) - 1
277 sqrt_df = math.sqrt(delta_f)
278 return [rho[n] /
279 (math.sqrt(2.) if (n == 0 or (n == N)) else 1.) *
280 sqrt_df for n in range(N + 1)]

284 @staticmethod
285 def TotalSpectralContentRMS(rms):
286 """Total RMS across a single-sided spectrum.
287 @param rms list of float per-bin rms values.
288 @return float total rms = sqrt(sum(rms[n]^2)).
289 """
290 return math.sqrt(sum(r * r for r in rms))
291
292 @staticmethod
293 def TotalSpectralContentdBm(dBm):
294 """Total power (dBm) across a single-sided spectrum.
295 @param dBm list of float per-bin dBm values.
296 @return float total power in dBm, obtained by summing the linear
297 powers and converting back to dBm.
298 """
299 LogRP10 = 10. * math.log10(50.0 * 1e-3)
300 return 10. * math.log10(
301 sum(10. ** ((d + LogRP10) / 10.) for d in dBm)) - LogRP10

305 @staticmethod
306 def ConvertSpectralDensity(value, from_units, to_units, bw=None, reference='voltage'):
307 """Convert a spectral quantity between unit representations.
308
309 Supported voltage units (reference='voltage'):
310 - 'dBm/Hz' power spectral density (50 ohm, 1 mW reference)
311 - 'V/sqrt(Hz)' amplitude spectral density
312 - 'V^2/GHz' voltage power spectral density per GHz
313 - 'Vrms' integrated rms voltage over bandwidth @a bw
314 (assumes a flat/white spectrum across @a bw)
315
316 Supported current units (reference='current'):
317 - 'dBm/Hz' power spectral density (50 ohm, 1 mW reference)
318 - 'A/sqrt(Hz)' amplitude spectral density
319 - 'A^2/GHz' current power spectral density per GHz
320 - 'Arms' integrated rms current over bandwidth @a bw
321 (assumes a flat/white spectrum across @a bw)
322
323 @param value float value to convert.
324 @param from_units source unit (voltage or current set per @a reference).
325 @param to_units destination unit (voltage or current set per @a reference).
326 @param bw float bandwidth in Hz; required only when the rms
327 ('Vrms'/'Arms') unit appears as input or output.
328 @param reference 'voltage' (default) or 'current'. Selects the unit
329 set and the power law used for the dBm leg. For a
330 voltage source P = asd^2 / R (R divides); for a
331 current source P = asd^2 * R (R multiplies).
332 @return converted value in @a to_units.
333 @throws ValueError on unknown units, or missing @a bw when needed.
334 @note
335 Only the dBm leg differs between voltage and current references; the
336 sqrt(Hz), ^2/GHz, and rms legs are structurally identical because they
337 do not involve the reference impedance.
338 """
339 R = 50.
340 P = 1e-3
341 if reference == 'current':
342 asd_units = 'A/sqrt(Hz)'
343 sq_units = 'A^2/GHz'
344 rms_units = 'Arms'
345 else:
346 asd_units = 'V/sqrt(Hz)'
347 sq_units = 'V^2/GHz'
348 rms_units = 'Vrms'
349 valid = ('dBm/Hz', asd_units, sq_units, rms_units)
350 if from_units not in valid:
351 raise ValueError(f'Unknown spectral density unit: {from_units}')
352 if to_units not in valid:
353 raise ValueError(f'Unknown spectral density unit: {to_units}')
354 if from_units == to_units:
355 return value
356 if (from_units == rms_units or to_units == rms_units) and bw is None:
357 raise ValueError(f"'bw' is required when converting to/from '{rms_units}'")
358
359 # Step 1: normalize input to amplitude spectral density (asd_units)
360 if from_units == asd_units:
361 asd = value
362 elif from_units == 'dBm/Hz':
363 # P = asd^2 / R (voltage) or P = asd^2 * R (current)
364 if reference == 'current':
365 asd = math.sqrt(P * 10. ** (value / 10.) / R)
366 else:
367 asd = math.sqrt(R * P * 10. ** (value / 10.))
368 elif from_units == sq_units:
369 asd = 0. if value <= 0. else math.sqrt(value / 1e9)
370 else: # rms_units
371 asd = value / math.sqrt(bw)
372
373 # Step 2: convert amplitude spectral density to requested output
374 if to_units == asd_units:
375 return asd
376 if to_units == sq_units:
377 return asd * asd * 1e9
378 if to_units == 'dBm/Hz':
379 if asd <= 0.:
380 return -3000.
381 if reference == 'current':
382 return 10. * math.log10(asd * asd * R / P)
383 return 10. * math.log10(asd * asd / R / P)
384 return asd * math.sqrt(bw) # rms_units
