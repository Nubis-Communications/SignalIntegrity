23class ParserArgs():
24 """argument handling base class for parsers"""
25 def AssignArguments(self,args):
26 """assigns arguments
27 @param args list of arguments as name,value pairs
28 The arguments are stored as a dictionary of values corresponding to the keyword names
29 """
30 self.m_vars=dict()
31 if args is None:
32 self.m_args=[]
33 else:
34 args=LineSplitter(args)
35 self.m_args = dict([('$'+args[i]+'$',args[i+1])
36 for i in range(0,len(args),2)])
37 def ReplaceArgs(self,lineList):
38 """replaces arguments in a line list with the actual values
39 @param lineList list of tokens in a line to process
40 If any token is surrounded by '$' and it's name is in the variable dictionary, then
41 the token is replaced with the value in the variable dictionary.
42 """
43 replacedOne=False
44 for i in range(len(lineList)):
45 if lineList[i] in self.m_vars:
46 replacedOne=True
47 lineList[i] = self.m_vars[lineList[i]]
48 if replacedOne:
49 lineList=';'.join(lineList).split(';')
50 return lineList
51 def ReplaceArgsInLine(self,line):
52 """replaces variables in a raw netlist line with the actual values.
53 @param line string netlist line to process
54 @return string the line with any variables replaced with their values.
55 @remark This is used for lines that are not processed immediately, but are
56 instead saved for later processing (i.e. the unknown lines list). Such lines
57 must have their variables resolved at the time that the variables are known,
58 otherwise the variables are lost.
59 @remark Unlike ReplaceArgs(), which operates on a token list produced by
60 LineSplitter() (and therefore strips quotes), this operates on the raw line
61 such that the line is returned unmolested if no variables are replaced.
62 """
63 if not hasattr(self,'m_vars') or not self.m_vars: return line
64 tokens=line.split(' ')
65 replacedOne=False
66 for i in range(len(tokens)):
67 if tokens[i] in self.m_vars:
68 tokens[i]=self.m_vars[tokens[i]]
69 replacedOne=True
70 return ' '.join(tokens) if replacedOne else line
71 def ProcessVariables(self,lineList):
72 """processes lines insofar as arguments are concerned.
73 @param lineList list of tokens in a line to process
74 If the first token is 'var', then the keyword,value pairs are placed in
75 the variable dictionary
76 """
77 if lineList[0] == 'var':
78 variables=dict([(lineList[i*2+1],lineList[i*2+2])
79 for i in range((len(lineList)-1)//2)])
80 for key in self.m_args:
81 if key in variables:
82 variables[key]=self.m_args[key]
83 self.m_vars=dict(list(self.m_vars.items())+list(variables.items()))
84 return True
85 else:
86 return False
