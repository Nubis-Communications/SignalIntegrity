25class TransferMatricesProcessor(CallBacker):
26 """process transfer matrices
27 Usually transfer matrices are produced in simulation and virtual probing
28 solutions. The resultant waveform processing for the final results are produced
29 using this class.
30 @see TransferMatrices"""
31 def __init__(self,transferMatrices,callback=None):
32 """Constructor
33 @param transferMatrices instance of class TransferMatrices.
34 @param callback (optional) reference to function that takes float percentage
35 used for periodically calling to show progress and allow aborting.
36 @see CallBacker
37 """
38 self.TransferMatrices=transferMatrices
39 # pragma: silent exclude
40 CallBacker.__init__(self,callback)
41 # pragma: include
42 def PrecalculateFrequencyResponses(self):
43 return self.TransferMatrices.FrequencyResponses()
44 def PrecalculateImpulseResponses(self,td=None):
45 return self.TransferMatrices.ImpulseResponses(td)
46 def ProcessWaveforms(self,wfl,td=None,adaptToLargest=False):
47 """processes input waveforms and produces output waveforms
48 @param wfl list of Waveform input waveforms to process. If numbers are in the
49 waveform list, they are assumed to be DC waveforms, that are treated specially.
50 @param td (optional) instance of class TimeDescriptor. If this is included,
51 all final waveforms are adapted to this descriptor. Otherwise, the sample rates of
52 each waveform are used in the computation of the impulse responses.
53 @param adaptToLargest bool (optional, defaults to False) causes waveform adaption
54 during summation to choose the waveform with the largest absolute value of a point.
55 This helps when models have frequency responses near the end frequency and causes
56 the adaption to resample the smaller waveforms, which have less effect.
57 @remark Externally, the order of the input and output waveforms are known. The
58 input waveforms must be provided in that order and the output waveforms are
59 produced in that order.
60 """
61 # pragma: silent exclude
62 from SignalIntegrity.Lib.TimeDomain.Waveform.Waveform import Waveform
63 # pragma: include
64 if td is None:
65 td = [wflm.td.Fs if isinstance(wflm,Waveform) else None for wflm in wfl]
66 ir = self.TransferMatrices.ImpulseResponses(td)
67 fr = self.TransferMatrices.FrequencyResponses() # for DC inputs
68 result=[]
69 for o in range(len(ir)):
70 acc=[]
71 for i in range(len(ir[o])):
72 acc.append(ir[o][i].FirFilter().FilterWaveform(wfl[i])
73 if isinstance(wfl[i],Waveform)
74 else (wfl[i]*fr[o][i][0]).real)
75 # pragma: silent exclude
76 if self.HasACallBack():
77 progress=(float(o)/len(ir)+float(i)/(len(ir)*len(ir[o])))*100.0
78 if not self.CallBack(progress):
79 raise SignalIntegrityExceptionSimulator('calculation aborted')
80 if adaptToLargest and len(acc)>1:
81 largestValue=0.0; largestIndex=0
82 for wfi in range(len(acc)):
83 if isinstance(acc[wfi],Waveform):
84 absMax=max(acc[wfi].Values('abs'))
85 if absMax>=largestValue:
86 largestIndex=wfi; largestValue=absMax
87 acc=[acc[largestIndex]]+acc[0:largestIndex]+acc[largestIndex+1:]
88 # if the first element of the accumulator is not a waveform (a number, assuming a DC value),
89 # then rearrange so a waveform comes first.
90 if not isinstance(acc[0],Waveform):
91 for wfi in range(len(acc)):
92 if isinstance(acc[wfi],Waveform):
93 acc=[acc[wfi]]+acc[0:wfi]+acc[wfi+1:]
94 break
95 # pragma: include
96 result.append(sum(acc))
97 return result
