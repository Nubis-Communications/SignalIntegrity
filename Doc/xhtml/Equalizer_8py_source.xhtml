24class CTLE(SParameters):
25 """Continuous time linear equalizer"""
26 def __init__(self,f,gDC,gDC2,fz,fLF,fp1,fp2,Z0=50.):
27 """Constructor
28 @param f list of float frequencies
29 @param gDC float DC gain in dB
30 @param gDC2 float DC gain at peaking point in dB
31 @param fz float frequency of zero
32 @param fLF float LF frequency
33 @param fp1 float first pole frequency
34 @param fp2 float second pole frequency
35 @param Z0 float (optional, defaults to 50) reference impedance
36 """
37 self.powgDC=pow(10.,gDC/20.); self.powgDC2=pow(10.,gDC2/20.)
38 self.joverfz=1j/fz; self.joverfLF=1j/fLF; self.joverfp1=1j/fp1; self.joverfp2=1j/fp2
39 SParameters.__init__(self,f,None,Z0)
40 def _Evaluate(self,f):
41 """Evaluate CTLE
42 @param f float frequency
43 @return CTLE evaluated at f
44 """
45 N=(self.powgDC+self.joverfz*f)*(self.powgDC2+self.joverfLF*f)
46 D=(1.+self.joverfp1*f)*(1.+self.joverfp2*f)*(1.+self.joverfLF*f)
47 return N/D
48 def __getitem__(self,n):
49 return [[1.,0],[2.*self._Evaluate(self.m_f[n]),-1.]]

51class FFE(SParameters):
52 """Linear feed-forward equalizer"""
53 def __init__(self,f,Td,taps,numPreCursorTaps,Z0=50.):
54 """Constructor
55 @param f list of float frequencies
56 @param Td float time delay of each tap
57 @param taps list of float values of the taps
58 @param numPreCursorTaps integer number of precursor taps
59 @param Z0 float (optional, defaults to 50) reference impedance
60 @remark Usually Td is the reciprocal of the baud rate or sometimes, half the reciprocal.
61 """
62 self.Td=Td
63 self.taps=taps
64 self.PC=numPreCursorTaps
65 SParameters.__init__(self,f,None,Z0)
66 def _Evaluate(self,f):
67 """Evaluate FFE
68 @param f float frequency
69 @return FFE evaluated at f
70 """
71 K=len(self.taps)
72 return sum([self.taps[k]*cmath.exp(1j*2.*cmath.pi*f*-(k-self.PC)*self.Td) for k in range(K)])
73 def __getitem__(self,n):
74 return [[1.,0],[2.*self._Evaluate(self.m_f[n]),-1.]]
