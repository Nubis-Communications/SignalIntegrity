26class FrequencyResponseFilter(SParameters):
27 """class for frequency response filter"""
28 def __init__(self,filename,normalizedDCGain=None,**kwargs):
29 """Constructor
30 @param filename string file name of frequency response file to read.
31 @param normalizedDCGain float (optional, defaults to None) DC gain to normalize response to.
32 @param **kwargs dict (optional, defaults to {}) dictionary of arguments for the file
33
34 Reads the frequency response file directly and produces SParameters.
35
36 The s-parameters are for a two-port device that is "amplifier-like", meaning it has infinite
37 input impedance, zero output impedance, the frequency response forms s21, and has infinite reverse
38 isolation.
39
40 If normalizedDCGain is 0 or None, the DC gain is not normalized, otherwise the filter is normalized
41 by setting the DC value of the frequency response equal to unity.
42 """
43 try:
44 fr=FrequencyResponse().ReadFromFile(filename)
45 except:
46 raise SignalIntegrityExceptionWaveformFile('frequency response could not be produced by '+filename)
47 if not ((normalizedDCGain == None) or (normalizedDCGain == 0)):
48 fr=FrequencyResponse(fr.Frequencies(),[r/fr[0] for r in fr])
49 fl=fr.Frequencies()
50 SParameters.__init__(self,fl,[[[1.,0],[2.*r,-1]] for r in fr])
