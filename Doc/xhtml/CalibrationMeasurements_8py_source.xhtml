22class CalibrationMeasurement(object):
23 """Base class for calibration measurements"""
24 def __init__(self,type,name=None):
25 """Constructor
26 @param type string representing the type of the measurement.
27 @param name (optional) string representing the name of the measurement
28 @remark
29 The name of the measurement is not used for anything, but can be used to identify
30 the name of a calibration measurement externally.
31
32 valid types of calibration measurements are:
33
34 - 'reflect' - a reflect measurement taken on a port with a reflect standard
35 (like short, open, or load).
36 - 'thru' - a thru calibration measurement taken between two ports.
37 - 'xtalk' - a crosstalk calibration measurement typically taken between two ports that
38 are completely unconnected.
39 @see ReflectCalibrationMeasurement
40 @see ThruCalibrationMeasurement
41 @see XtalkCalibrationMeasurement
42 """
43 self.type=type
44 self.name=name
