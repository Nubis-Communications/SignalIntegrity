21from SignalIntegrity.Lib.Exception import SignalIntegrityExceptionSimulator
22from SignalIntegrity.Lib.CallBacker import CallBacker

24class NoiseTransferMatricesProcessor(CallBacker):
25 """Processes transfer matrices for noise spectral densities.
26 Given transfer matrices and a list of input spectral densities, produces
27 output spectral densities by multiplying each input by its corresponding
28 frequency response magnitude and combining contributions via root-sum-square
29 (assuming uncorrelated noise sources).
30
31 @see TransferMatrices
32 @see SpectralDensity
33 """
34 def __init__(self, transferMatrices, callback=None):
35 """Constructor
36 @param transferMatrices instance of class TransferMatrices.
37 @param callback (optional) reference to function that takes float percentage
38 used for periodically calling to show progress and allow aborting.
39 @see CallBacker
40 """
41 self.TransferMatrices = transferMatrices
42 CallBacker.__init__(self, callback)
43
44 def ProcessNoise(self, sdl):
45 """Processes input noise spectral densities through transfer matrices.
46 @param sdl list of SpectralDensity instances, one per input source.
47 @return list of SpectralDensity instances, one per output.
48 @remark
49 For each output o, the output spectral density is computed as:
50 SD_out[o](f) = sqrt( sum_i ( |H[o][i](f)| * SD_in[i](f) )^2 )
51 This assumes uncorrelated noise sources so contributions add in
52 root-sum-square fashion.
53 """
54 fr = self.TransferMatrices.FrequencyResponses()
55 if fr is None:
56 return None
57 self.Contributions = [[None for _ in range(self.TransferMatrices.Inputs)]
58 for _ in range(self.TransferMatrices.Outputs)]
59 self.Result = []
60 for o in range(self.TransferMatrices.Outputs):
61 outputSD = None
62 for i in range(self.TransferMatrices.Inputs):
63 contribution = sdl[i] * fr[o][i]
64 self.Contributions[o][i] = contribution
65 if outputSD is None:
66 outputSD = contribution
67 else:
68 outputSD = outputSD + contribution
69 if self.HasACallBack():
70 progress = (float(o) / self.TransferMatrices.Outputs +
71 float(i) / (self.TransferMatrices.Outputs *
72 self.TransferMatrices.Inputs)) * 100.0
73 if not self.CallBack(progress):
74 raise SignalIntegrityExceptionSimulator('calculation aborted')
75 self.Result.append(outputSD)
76 return self.Result
