21# Parallel computation of per-frequency solutions.
22#
23# The transfer-matrix solutions performed by SimulatorNumericParser and
24# VirtualProbeNumericParser, and the system s-parameter solution performed by
25# SystemSParametersNumericParser, loop over every frequency and solve an
26# independent problem at each one. Because the per-frequency solves are
27# independent, they can be distributed across multiple processor cores. This
28# module provides that machinery in a form that is safe on Windows (which uses
29# the 'spawn' start method and therefore requires module-level, picklable worker
30# callables).
31#
32# The worker interface is *batch ready*: a worker receives a chunk (list) of
33# frequency payloads so that frequency batching can be tuned later simply by
34# changing the chunk size, with no change to the worker signature. With the
35# default chunk size of 1 the behavior matches a straightforward
36# one-task-per-frequency mapping.
37

42#: Number of worker processes to use *when parallel execution has been allowed*.
43#:
44#: Parallel per-frequency execution is **gated** by the ``allowParallel``
45#: argument to :func:`Solve` (which itself originates from the
46#: ``AllowParallelization`` calculation property, defaulting to False). This
47#: attribute does NOT turn parallelism on; it only chooses how many workers are
48#: used once it is allowed. The gate is separate because, for the vast majority
49#: of real problems, the per-frequency solve is cheap relative to the cost of
50#: standing up a process pool and shipping the problem (deepcopy + pickle of the
51#: system description and every per-frequency matrix) to the workers. In those
52#: cases the process/IPC overhead dominates and the "parallel" path is actually
53#: slower than serial -- sometimes by 2x or more. Measured example: a
54#: 1000-point TDR calibration that solves in 110s serially takes 219s when
55#: forced parallel, because the tiny solves are swamped by the repeated pool
56#: startup and data transfer. Even when parallel is allowed, the cost model
57#: below still decides per-solve whether it actually pays off.
58#:
59#: Values: ``None`` means "choose automatically" from os.cpu_count(); an int > 1
60#: uses that many workers; 1 (or anything <= 1) forces serial even when allowed.
61#: The environment variable SIGNALINTEGRITY_NUM_WORKERS overrides this at run
62#: time (an integer count, or 0/"auto" for os.cpu_count()).
63DefaultNumberOfWorkers = None
64
65#: Environment variable that overrides DefaultNumberOfWorkers at run time so
66#: parallelism can be enabled without editing code. Accepts an integer worker
67#: count, or "auto"/"0" for os.cpu_count(). Unset means "use
68#: DefaultNumberOfWorkers".
69NumberOfWorkersEnvVar = 'SIGNALINTEGRITY_NUM_WORKERS'

71#: Minimum number of frequencies before parallel execution is attempted. Below
72#: this the process startup / data transfer overhead is not worth it and the
73#: serial path is used.
74MinimumFrequenciesForParallel = 8
75
76#: Estimated serial solve time (in seconds) below which parallel execution is
77#: never attempted. The per-frequency solve is timed once and extrapolated; if
78#: the whole calculation is predicted to take less than this, the process
79#: startup / pickling overhead would dominate and the serial path is used
80#: instead. This is only a coarse floor; the finer decision is made by the
81#: cost model below (see EstimatedPoolStartupSeconds).
82MinimumEstimatedSerialSeconds = 2.0
83
84#: Estimated fixed cost (in seconds) of standing up a worker process pool. On
85#: Windows (the 'spawn' start method) each worker is a fresh interpreter that
86#: re-imports SignalIntegrity and unpickles the base problem, which measures at
87#: roughly 2 seconds for a pool of several workers. Parallel execution only
88#: pays off when the time it *saves* clearly exceeds this fixed cost, so the
89#: driver compares the two before spawning anything. This guards against the
90#: pathological case of many moderate solves (for example a calibration run)
91#: each spinning up and tearing down a pool whose startup dwarfs the tiny gain.
92EstimatedPoolStartupSeconds = 2.0
93
94#: Safety factor applied to the predicted parallel savings. The estimated time
95#: saved by going parallel must exceed EstimatedPoolStartupSeconds by at least
96#: this factor before a pool is created. A value > 1 keeps borderline problems
97#: serial, where the estimate is least reliable and a wrong guess is most costly.
98ParallelStartupSafetyFactor = 2.0

100#: Number of frequencies handled per worker task. 1 reproduces
101#: one-task-per-frequency behavior. Larger values amortize process startup,
102#: pickling and per-solver setup costs. A value of ``None`` means "choose
103#: automatically" based on the number of frequencies and workers.
104DefaultChunkSize = None

106#: Environment variables that control the number of threads used by the various
107#: BLAS / linear-algebra back-ends that numpy may sit on top of. Each worker
108#: process performs its own linear-algebra solves, and those back-ends *already*
109#: try to use every core. If that is left unchecked then N worker processes
110#: each spawn N BLAS threads, oversubscribing the machine by a factor of N and
111#: making the parallel run dramatically slower than the serial one. To get a
112#: real speedup, each worker is restricted to a single BLAS thread so that the
113#: parallelism comes purely from the process pool.
114_ThreadLimitEnvVars = (
115 'OMP_NUM_THREADS',
116 'OPENBLAS_NUM_THREADS',
117 'MKL_NUM_THREADS',
118 'NUMEXPR_NUM_THREADS',
119 'VECLIB_MAXIMUM_THREADS',
120)
121
122#: Number of BLAS threads each worker process is allowed to use. 1 avoids
123#: oversubscription (see _ThreadLimitEnvVars). Increase only if you deliberately
124#: run fewer worker processes than cores and want each to use the remainder.
125ThreadsPerWorker = 1

127class _WorkerThreadLimit(object):
128 """Context manager that restricts the BLAS thread count seen by worker
129 processes spawned while it is active.
130
131 Worker processes created with the (default on Windows) 'spawn' start method
132 inherit a *copy* of the parent's environment and import numpy fresh, reading
133 these variables when their BLAS back-end initializes. Setting the variables
134 here therefore limits the children without disturbing the parent's own,
135 already-initialized numpy. The parent's environment is restored on exit.
136 """
137 def __init__(self, threadsPerWorker):
138 self._value = str(max(1, int(threadsPerWorker)))
139 self._saved = {}
140 def __enter__(self):
141 for name in _ThreadLimitEnvVars:
142 self._saved[name] = os.environ.get(name)
143 os.environ[name] = self._value
144 return self
145 def __exit__(self, *excInfo):
146 for name, previous in self._saved.items():
147 if previous is None:
148 os.environ.pop(name, None)
149 else:
150 os.environ[name] = previous
151 return False

153class _SuppressMainReimport(object):
154 """Context manager that stops spawned worker processes from re-importing the
155 parent's ``__main__`` module.
156
157 With the 'spawn' start method (the default on Windows) every worker is a
158 brand-new interpreter that, as part of its bootstrap, re-imports the parent's
159 ``__main__`` module (see ``multiprocessing.spawn.get_preparation_data``). The
160 workers created here need *nothing* from ``__main__``: the worker callable
161 (:func:`_SolveChunk`) and every payload object (the system description, the
162 per-frequency matrices, numpy arrays) live in ordinary importable
163 SignalIntegrity/library modules. So that re-import is pure overhead -- and,
164 critically, its cost is unbounded and depends entirely on *what launched the
165 process*:
166
167 * Run a single ``.si`` file or one test module directly and ``__main__`` is
168 tiny, so each worker starts almost instantly.
169 * Run the same solve as part of a larger driver -- a test runner performing
170 unittest discovery, PyDev/Eclipse, pytest, or a top-level script that does
171 ``from Test... import *`` for a whole suite -- and ``__main__`` is *that*
172 driver. Re-importing it in a worker drags in the entire suite's imports
173 (dozens of test modules, the matplotlib GUI backend, ...) before the worker
174 can do any work. That happens for *every* worker, in *every* pool, for
175 *every* solve, which is exactly the pathological slowdown observed when
176 these parallel solves run inside a big test run instead of on their own.
177
178 ``multiprocessing`` decides how to reconstruct ``__main__`` in the child from
179 two attributes of the parent's ``__main__`` module: ``__spec__`` (import it by
180 module name) and ``__file__`` (execute it by path). Clearing both for the
181 duration of pool creation makes the preparation data carry no main-init
182 directive at all, so the workers skip the re-import completely. The parent's
183 own ``__main__`` is restored on exit, so this is invisible to the caller.
184 """
185 _SENTINEL = object()
186 def __init__(self):
187 self._main = None
188 self._savedSpec = self._SENTINEL
189 self._savedFile = self._SENTINEL
190 def __enter__(self):
191 import sys
192 self._main = sys.modules.get('__main__')
193 if self._main is not None:
194 self._savedSpec = getattr(self._main, '__spec__', self._SENTINEL)
195 self._savedFile = getattr(self._main, '__file__', self._SENTINEL)
196 # __spec__ = None suppresses the "import by module name" path;
197 # removing __file__ suppresses the "execute by path" path. With both
198 # gone, get_preparation_data emits no main-init directive.
199 try:
200 self._main.__spec__ = None
201 except Exception:
202 pass
203 if self._savedFile is not self._SENTINEL:
204 try:
205 del self._main.__file__
206 except Exception:
207 pass
208 return self
209 def __exit__(self, *excInfo):
210 if self._main is not None:
211 if self._savedSpec is not self._SENTINEL:
212 try:
213 self._main.__spec__ = self._savedSpec
214 except Exception:
215 pass
216 if self._savedFile is not self._SENTINEL:
217 try:
218 self._main.__file__ = self._savedFile
219 except Exception:
220 pass
221 return False

227#: Per-process worker state, cached across tasks and reused between solves.
228#:
229#: Because the process pool is now *persistent* (created once and reused for
230#: every per-frequency solve -- see _GetPersistentExecutor), a worker cannot be
231#: handed the base problem once via a pool ``initializer``: successive solves use
232#: different system descriptions. Instead every task carries the base problem
233#: tagged with a monotonically increasing ``generation`` id. A worker rebuilds
234#: its solver only when it first sees a new generation, so within one solve the
235#: (structurally cached) solver is still built once per worker and reused for all
236#: of that solve's frequencies -- exactly as before -- but the expensive process
237#: spawn and interpreter/import startup is paid only once for the whole run.
238_WorkerState = {}

240def _SolveChunk(payload):
241 """Solves the per-frequency result for a chunk of frequencies.
242 @param payload tuple ``(generation, kind, sd, names, Z0, solvetype, chunk)``
243 where ``chunk`` is a list of ``(n, matrices, system)`` tuples (n is the
244 frequency index, matrices is a list of per-device s-parameter matrices
245 aligned with names, and system is the per-frequency system s-parameter matrix
246 used by the 'deembedder' kind -- None for the other kinds) and the remaining
247 fields describe the base problem for this solve.
248 @return list of (n, result) tuples.
249 @remark The base problem (kind, sd, names, Z0, solvetype) is identical for
250 every chunk of a given solve, so it is cached per worker keyed on the
251 generation id and the solver is rebuilt only when the generation changes.
252 This keeps the per-frequency work identical to the original one-solver-per-
253 worker behavior while allowing the pool itself to be reused across solves.
254 """
255 (generation, kind, sd, names, Z0, solvetype, chunk) = payload
256 if _WorkerState.get('generation') != generation:
257 _WorkerState['generation'] = generation
258 _WorkerState['kind'] = kind
259 _WorkerState['sd'] = sd
260 _WorkerState['names'] = names
261 _WorkerState['Z0'] = Z0
262 _WorkerState['solvetype'] = solvetype
263 # The solver (when applicable) shares the same Device objects that
264 # AssignSParameters mutates, and its structural caches are valid across
265 # all frequencies of this solve, so a single solver is reused for every
266 # frequency until the next solve (a new generation) arrives.
267 _WorkerState['solver'] = _MakeSolver(kind, sd)
268 sd = _WorkerState['sd']
269 names = _WorkerState['names']
270 Z0 = _WorkerState['Z0']
271 kind = _WorkerState['kind']
272 solver = _WorkerState['solver']
273 solvetype = _WorkerState['solvetype']
274 results = []
275 for (n, matrices, system) in chunk:
276 results.append((n, _SolveOneFrequency(kind, sd, names, matrices, Z0,
277 solver, solvetype, system)))
278 return results

294#: The persistent, process-wide worker pool and the number of workers it was
295#: created with. The pool is created lazily on the first parallel solve and then
296#: reused for every subsequent solve, so the ~2 second cost of spawning worker
297#: interpreters and importing numpy/SignalIntegrity into each is paid *once* for
298#: the whole run instead of once per solve. This is the dominant cost when many
299#: independent solves run back to back (for example a calibration sweep or a test
300#: suite that solves dozens of small schematics); rebuilding a fresh pool for
301#: each of those solves made "parallel" execution slower than serial even though
302#: the individual solves genuinely parallelize.
303_PersistentExecutor = None
304_PersistentExecutorWorkers = None

306#: Monotonic id identifying the current solve's base problem to the workers.
307#: Incremented for every parallel solve; a worker rebuilds its cached solver when
308#: it sees a generation it has not seen before (see _SolveChunk).
309_SolveGeneration = 0

311def _GetPersistentExecutor(workers, mainGuard):
312 """Returns a process pool with ``workers`` workers, creating (and warming) it
313 on first use and reusing it thereafter.
314 @param workers int number of worker processes required.
315 @param mainGuard context manager active while any *new* workers are spawned
316 (restricts BLAS threads and suppresses the __main__ re-import in children).
317 @return a ProcessPoolExecutor, or None if one could not be created.
318 @remark If a pool with a different worker count already exists it is shut
319 down and replaced. New workers are spawned eagerly (warmed up) while
320 ``mainGuard`` is active so the spawn-time environment (thread limits, no
321 __main__ re-import) is guaranteed to apply to them.
322 """
323 global _PersistentExecutor, _PersistentExecutorWorkers
324 from concurrent.futures import ProcessPoolExecutor
325 if (_PersistentExecutor is not None and
326 _PersistentExecutorWorkers != workers):
327 try:
328 _PersistentExecutor.shutdown(wait=True)
329 except Exception:
330 pass
331 _PersistentExecutor = None
332 _PersistentExecutorWorkers = None
333 created = False
334 if _PersistentExecutor is None:
335 _PersistentExecutor = ProcessPoolExecutor(max_workers=workers)
336 _PersistentExecutorWorkers = workers
337 created = True
338 if created:
339 # Force every worker to spawn now, while mainGuard is active, so the
340 # child interpreters inherit the restricted BLAS thread count and skip
341 # re-importing __main__. Without this warm-up the workers would instead
342 # spawn lazily during the first real submit loop -- outside the guard and
343 # timed as part of that solve.
344 with mainGuard:
345 try:
346 list(_PersistentExecutor.map(_WarmUpWorker, range(workers)))
347 except Exception:
348 pass
349 return _PersistentExecutor

365def _ResolveNumberOfWorkers(numWorkers, numFrequencies):
366 """Determines the effective number of worker processes to use.
367
368 Resolution order:
369 1. an explicit ``numWorkers`` argument (from the caller),
370 2. the ``SIGNALINTEGRITY_NUM_WORKERS`` environment variable,
371 3. the module attribute ``DefaultNumberOfWorkers`` (serial by default).
372 ``None`` at any level means "choose automatically" from os.cpu_count(). A
373 resolved value <= 1 forces serial execution.
374 """
375 if numWorkers is None:
376 numWorkers = _WorkersFromEnv()
377 if numWorkers is None:
378 numWorkers = DefaultNumberOfWorkers
379 if numWorkers is None:
380 numWorkers = os.cpu_count() or 1
381 return max(1, min(numWorkers, numFrequencies))

383def _WorkersFromEnv():
384 """Reads the worker count from the environment, or returns None if unset.
385 Accepts an integer, or "auto"/"0"/"" for automatic (os.cpu_count())."""
386 raw = os.environ.get(NumberOfWorkersEnvVar)
387 if raw is None:
388 return None
389 raw = raw.strip().lower()
390 if raw in ('', 'auto', '0'):
391 return os.cpu_count() or 1
392 try:
393 return int(raw)
394 except ValueError:
395 return None

397def _ResolveChunkSize(chunkSize, numFrequencies, numWorkers):
398 """Determines the effective chunk size."""
399 if chunkSize is None:
400 chunkSize = DefaultChunkSize
401 if chunkSize is None:
402 # Aim for several chunks per worker for reasonable load balancing while
403 # still amortizing per-task overhead.
404 chunkSize = max(1, numFrequencies // (numWorkers * 4))
405 return max(1, chunkSize)

407def _BuildChunks(spc, numFrequencies, chunkSize, startIndex=0, systemMatrices=None):
408 """Builds the list of per-frequency payload chunks.
409 @param startIndex (optional) first frequency index to include (frequencies
410 before it, for example a timing probe, are skipped).
411 @param systemMatrices (optional) list of per-frequency system s-parameter
412 matrices used by the 'deembedder' kind; None (the default) supplies None as
413 the per-frequency system for every frequency.
414 @return list of chunks, each a list of (n, [spc[d][1][n] for d], system_n)
415 tuples.
416 """
417 numDevices = len(spc)
418 chunks = []
419 n = startIndex
420 while n < numFrequencies:
421 chunk = []
422 for nn in range(n, min(n + chunkSize, numFrequencies)):
423 matrices = [spc[d][1][nn] for d in range(numDevices)]
424 system = systemMatrices[nn] if systemMatrices is not None else None
425 chunk.append((nn, matrices, system))
426 chunks.append(chunk)
427 n += chunkSize
428 return chunks

430def _MakeSolver(kind, sd):
431 """Creates the reusable per-process solver for kinds that support reuse."""
432 if kind == 'simulator':
433 from SignalIntegrity.Lib.SystemDescriptions import SimulatorNumeric
434 return SimulatorNumeric(sd)
435 if kind == 'systemsparameters':
436 from SignalIntegrity.Lib.SystemDescriptions import SystemSParametersNumeric
437 return SystemSParametersNumeric(sd)
438 if kind == 'deembedder':
439 from SignalIntegrity.Lib.SystemDescriptions.DeembedderNumeric import DeembedderNumeric
440 return DeembedderNumeric(sd)
441 return None

443def _SolveOneFrequency(kind, sd, names, matrices, Z0, solver, solvetype=None,
444 system=None):
445 """Assigns the per-device s-parameters for one frequency and solves it.
446 @param system (optional) the per-frequency system s-parameter matrix used by
447 the 'deembedder' kind (the s-parameters of the known overall system).
448 """
449 for d in range(len(names)):
450 if names[d] is not None:
451 sd.AssignSParameters(names[d], matrices[d])
452 if kind == 'simulator':
453 return solver.TransferMatrix(Z0=Z0)
454 if kind == 'systemsparameters':
455 return solver.SParameters(solvetype=solvetype)
456 if kind == 'deembedder':
457 # The deembedder solver reads its assigned per-device s-parameters from
458 # sd each call, so the (structurally cache-free) solver is safely reused
459 # across frequencies. It returns the unknown device s-parameters (a
460 # single matrix for one unknown, or a list of matrices for several).
461 return solver.CalculateUnknown(system)
462 from SignalIntegrity.Lib.SystemDescriptions import VirtualProbeNumeric
463 return VirtualProbeNumeric(sd).TransferMatrix()

465def _SolveSerial(kind, sd, spc, numFrequencies, Z0, callback, abortException,
466 result=None, startIndex=0, solver=None, solvetype=None,
467 systemMatrices=None):
468 """Serial fallback that reproduces the original per-frequency loop.
469 @param result (optional) pre-allocated result list; already-computed entries
470 (for example a timing probe) are preserved and not recomputed.
471 @param startIndex (optional) first frequency index to compute.
472 @param solver (optional) reusable solver to continue with.
473 @param solvetype (optional) solve method string, used by 'systemsparameters'.
474 @param systemMatrices (optional) list of per-frequency system s-parameter
475 matrices, used by 'deembedder'.
476 """
477 names = [spc[d][0] for d in range(len(spc))]
478 if solver is None:
479 solver = _MakeSolver(kind, sd)
480 if result is None:
481 result = [None] * numFrequencies
482 for n in range(startIndex, numFrequencies):
483 matrices = [spc[d][1][n] for d in range(len(spc))]
484 system = systemMatrices[n] if systemMatrices is not None else None
485 result[n] = _SolveOneFrequency(kind, sd, names, matrices, Z0, solver,
486 solvetype, system)
487 if callback is not None:
488 progress = (n + 1) / numFrequencies * 100.0
489 if not callback(progress):
490 raise abortException
491 return result

493def Solve(kind, sd, spc, numFrequencies, Z0,
494 callback=None, abortException=None,
495 numWorkers=None, chunkSize=None, solvetype=None,
496 allowParallel=False, systemMatrices=None):
497 """Computes the per-frequency result, in parallel when beneficial.
498 @param kind string, one of 'simulator', 'virtualprobe', 'systemsparameters'
499 or 'deembedder'.
500 @param sd instance of the (already checked) system description.
501 @param spc list of (deviceName, sParametersOverFrequency) tuples.
502 @param numFrequencies integer number of frequencies.
503 @param Z0 float reference impedance for the calculation.
504 @param callback (optional) callable taking a float progress percentage and
505 returning False to abort.
506 @param abortException (optional) exception instance to raise when the
507 callback requests an abort.
508 @param numWorkers (optional) number of worker processes; None chooses
509 automatically, 1 forces serial.
510 @param chunkSize (optional) number of frequencies per worker task; None
511 chooses automatically.
512 @param solvetype (optional) solve method string, used by 'systemsparameters'.
513 @param allowParallel (optional) master gate for parallel execution. When
514 False (the default) the calculation always runs serially and no worker pool
515 is ever created. When True, parallel execution is *permitted* but still
516 subject to the size/cost model below, which decides per-solve whether it
517 actually pays off. This value originates from the AllowParallelization
518 calculation property.
519 @param systemMatrices (optional) list of per-frequency system s-parameter
520 matrices used by the 'deembedder' kind (the s-parameters of the known overall
521 system, one matrix per frequency, or None). Ignored by the other kinds.
522 @return list of per-frequency results ordered by frequency index. For the
523 'deembedder' kind each entry is the unknown device s-parameters at that
524 frequency (a single matrix for one unknown, or a list of matrices for
525 several), which the caller reorganizes per unknown device.
526 @remark Falls back to serial execution when not allowed, for small problems,
527 when a single worker is requested, or if parallel execution cannot be started
528 (for example if the problem cannot be pickled).
529 """
530 # Master gate: unless parallel execution has been explicitly allowed, run
531 # the original serial loop and never stand up a worker pool.
532 if not allowParallel:
533 return _SolveSerial(kind, sd, spc, numFrequencies, Z0,
534 callback, abortException, solvetype=solvetype,
535 systemMatrices=systemMatrices)
536
537 # Re-entrancy guard against the Windows 'spawn' fork-bomb. Each worker is a
538 # fresh interpreter that re-imports __main__ (the caller's script). If that
539 # script fails to guard its top-level work with `if __name__ == '__main__':`,
540 # every worker re-runs the whole script, which lands back here and would try
541 # to spawn yet another pool -- an exponential explosion of processes that
542 # repeats the calculation. Detect that we are inside a spawned child (it has
543 # a parent process) and force serial there so a nested pool can never be
544 # created. Note: this only stops the library from compounding the problem;
545 # the caller's script must still use the `if __name__ == '__main__':` guard
546 # (or multiprocessing.freeze_support()) to avoid re-running its own top-level
547 # code in each worker.
548 try:
549 import multiprocessing
550 if multiprocessing.parent_process() is not None:
551 return _SolveSerial(kind, sd, spc, numFrequencies, Z0,
552 callback, abortException, solvetype=solvetype,
553 systemMatrices=systemMatrices)
554 except (ImportError, AttributeError):
555 # parent_process() exists on Python 3.8+; if unavailable, skip the guard.
556 pass
557
558 effectiveWorkers = _ResolveNumberOfWorkers(numWorkers, numFrequencies)
559
560 if (effectiveWorkers <= 1 or
561 numFrequencies < MinimumFrequenciesForParallel):
562 return _SolveSerial(kind, sd, spc, numFrequencies, Z0,
563 callback, abortException, solvetype=solvetype,
564 systemMatrices=systemMatrices)
565
566 names = [spc[d][0] for d in range(len(spc))]
567
568 # Probe: time a single frequency solve serially and extrapolate. If the
569 # whole calculation is cheap, the multiprocessing overhead would dominate,
570 # so stay serial. The probe result is kept and reused, so no work is lost.
571 solver = _MakeSolver(kind, sd)
572 result = [None] * numFrequencies
573 probeMatrices = [spc[d][1][0] for d in range(len(spc))]
574 probeSystem = systemMatrices[0] if systemMatrices is not None else None
575 # Warm up before timing. The very first solve pays one-time costs that are
576 # not representative of the steady-state per-frequency cost: building the
