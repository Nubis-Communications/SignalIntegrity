22class Spline(object):
23 def __init__(self,x,y):
24 N=len(x)
25 r = range(0,N-1)
26 h = [x[i+1]-x[i] for i in r]
27 b = [(y[i+1]-y[i])/h[i] for i in r]
28 u = [0,2.*(h[0]+h[1])]
29 v = [0,6.*(b[1]-b[0])]
30 for i in range(2,N-1):
31 u.append(2.*(h[i]+h[i-1])-h[i-1]*h[i-1]/u[i-1])
32 v.append(6.*(b[i]-b[i-1])-h[i-1]*v[i-1]/u[i-1])
33 zi = reversed(range(1,N-1))
34 z = [0 for i in range(N)]
35 for i in zi:
36 z[i]=(v[i]-h[i]*z[i+1])/u[i]
37 r=range(0,N-1)
38 A = [y[i] for i in r]
39 B = [-h[i]*z[i+1]/6.-h[i]*z[i]/3.+b[i] for i in r]
40 C = [z[i]/2. for i in r]
41 D = [1./6./h[i]*(z[i+1]-z[i]) for i in r]
42 self.m_t = [x[i] for i in r]
43 P=[]
44 #these are the actual polynomial coefficients for P[0]+P[1]*x+P[2]*x**2+P[3]*x**3
45 for i in r:
46 Pi=[]
47 Pi.append(A[i]+x[i]*(-B[i]+x[i]*(C[i]-x[i]*D[i])))
48 Pi.append(B[i]+x[i]*(-2.*C[i]+x[i]*3.*D[i]))
49 Pi.append(C[i]-3.*D[i]*x[i])
50 Pi.append(D[i])
51 P.append(Pi)
52 self.m_A=[]
53 #these are the polynomial coefficients for A[0]+A[1]*(x-t)+A[2]*(x-t)**2+A[3]*(x-t)**3
54 #where t is the lower boundary of the interval
55 for i in r:
56 Ai=[]
57 Ai.append(P[i][0]+x[i]*(P[i][1]+x[i]*(P[i][2]+x[i]*P[i][3])))
58 Ai.append(P[i][1]+x[i]*(2.*P[i][2]+x[i]*3.*P[i][3]))
59 Ai.append(P[i][2]+x[i]*3.*P[i][3])
60 Ai.append(P[i][3])
61 self.m_A.append(Ai)
62 def Interval(self,x,newWay=True):
63 try:
64 import bisect
65 except:
66 newWay=False
67 if newWay:
68 import bisect
69 if x < self.m_t[0]:
70 return 0
71 elif x < self.m_t[-1]:
72 return bisect.bisect_right(self.m_t,x)-1
73 else:
74 return len(self.m_t)-1
75 else:
76 if x<self.m_t[0]:
77 return 0
78 for i in range(1,len(self.m_t)):
79 if x < self.m_t[i]:
80 return i-1
81 return len(self.m_t)-1

83 i = self.Interval(x)
84 xi = (x-self.m_t[i])
85 A=self.m_A[i]
86 return A[0]+xi*(A[1]+xi*(A[2]+xi*A[3]))
