22def VoltageAmplifier(P,G,Zi,Zo,Z0=50):
23 """VoltageAmplifier
24 2-4 Port Voltage Amplifiers
25 @param P integer number of ports (2-4)
26 @param G float voltage gain
27 @param Zi float or complex input impedance
28 @param Zo float or complex output impedance
29 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
30 @return list of list s-parameter matrix for a voltage amplifier
31 @remark
32 The voltage amplifier can be two, three or four ports.\n
33 @see VoltageAmplifierFourPort
34 @see VoltageAmplifierThreePort
35 @see VoltageAmplifierTwoPort
36 """
37 if P==2:
38 return VoltageAmplifierTwoPort(G,Zi,Zo,Z0=50.)
39 elif P==3:
40 return VoltageAmplifierThreePort(G,Zi,Zo,Z0=50.)
41 elif P==4:

45def VoltageAmplifierFourPort(G,Zi,Zo,Z0=50.):
46 """VoltageAmplifierFourPort
47 Four port voltage amplifier
48 @param G float voltage gain
49 @param Zi float or complex input impedance
50 @param Zo float or complex output impedance
51 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
52 @return list of list s-parameter matrix for a voltage amplifier
53 @remark
54 The voltage sense element senses the voltage with the plus terminal at port 1
55 and and the minus terminal at port 2 shunted with impedance of Zi.\n
56 The voltage generated by the amplifier has the plus terminal at port 3 and
57 and the minus terminal at port 4 with Zo in series.\n
58 """
59 S11=Zi/(Zi+2.*Z0)
60 S12=2.*Z0/(Zi+2.*Z0)
61 S13=0
62 S14=0
63 S21=S12
64 S22=S11
65 S23=0
66 S24=0
67 S31=2.*Zi*Z0*G/((Zi+2.*Z0)*(Zo+2.*Z0))
68 S32=-S31
69 S33=Zo/(Zo+2.*Z0)
70 S34=2.*Z0/(Zo+2.*Z0)
71 S41=S32
72 S42=S31
73 S43=S34
74 S44=S33
75 return [[S11,S12,S13,S14],
76 [S21,S22,S23,S24],
77 [S31,S32,S33,S34],
78 [S41,S42,S43,S44]]

80def VoltageAmplifierThreePort(G,Zi,Zo,Z0=50.):
81 """VoltageAmplifierThreePort
82 Three port voltage amplifier
83 @param G float voltage gain
84 @param Zi float or complex input impedance
85 @param Zo float or complex output impedance
86 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
87 @return list of list s-parameter matrix for a voltage amplifier
88 @remark
89 The three port voltage amplifier is the same as the four port voltage amplifier with
90 ports two and three connected together and exposed as a single port.\n
91 The voltage sense element senses the voltage with the plus terminal at port 1
92 and and the minus terminal at port 3 shunted with impedance of Zi.\n
93 The voltage generated by the amplifier has the plus terminal at port 2 and
94 and the minus terminal at port 3 with Zo in series.\n
95 """
96 D=-Zo*Zi-2.*Zo*Z0-2.*Zi*Z0-3.*Z0*Z0+G*Zi*Z0
97 S11=(-Zo*Zi-2.*Zi*Z0+Z0*Z0+G*Zi*Z0)/D
98 S12=-2.*Z0*Z0/D
99 S13=-2.*Z0*(Zo+Z0)/D
100 S21=-2.*Z0*(G*Zi+Z0)/D
101 S22=(Z0*Z0-2.*Zo*Z0+G*Zi*Z0-Zo*Zi)/D
102 S23=2.*Z0*(G*Zi-Zi-Z0)/D
103 S31=2.*Z0*(-Z0+G*Zi-Zo)/D
104 S32=-2.*Z0*(Zi+Z0)/D
105 S33=(-Zo*Zi+Z0*Z0-G*Zi*Z0)/D
106 return [[S11,S12,S13],
107 [S21,S22,S23],
108 [S31,S32,S33]]

110def VoltageAmplifierTwoPort(G,Zi,Zo,Z0=50.):
111 """VoltageAmplifierTwoPort
112 Two port voltage amplifier
113 @param G float voltage gain
114 @param Zi float or complex input impedance
115 @param Zo float or complex output impedance
116 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
117 @return list of list s-parameter matrix for a voltage amplifier
118 @remark
119 The two port voltage amplifier is the same as the three port voltage amplifier with
120 port 3 grounded.\n
121 The voltage sense element senses the voltage to ground at port 1 shunted to ground
122 with impedance of Zi.\n
123 The voltage generated by the amplifier is at port 2 referenced to ground
124 with Zo in series with the output.\n
125 """
126 return [[(Zi-Z0)/(Zi+Z0),0.],[2.*G*Zi*Z0/((Zi+Z0)*(Zo+Z0)),(Zo-Z0)/(Zo+Z0)]]
