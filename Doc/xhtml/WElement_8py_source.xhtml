28class MaxwellMatrix(list):
29 """A Maxwell matrix is a lower triangular matrix that is related to the mutual matrix."""
30 def __init__(self,MM):
31 """Constructor
32 @param MM lower triangular list of list Maxwell matrix
33 """
34 list.__init__(self,MM)
35 def MutualMatrix(self):
36 """Mutual matrix
37 @return a lower triangular mutual matrix
38 """
39 wires=len(self)
40 Mm=[[None for _ in range(w+1)] for w in range(wires)]
41 for r in range(wires):
42 Mm[r][r]=sum([self[r][c] if r >= c else self[c][r] for c in range(wires)])
43 for c in range(r): Mm[r][c]=-self[r][c]
44 return MutualMatrix(Mm)

54 """Maxwell matrix
55 @return a lower triangular Maxwell matrix
56 """
57 wires=len(self)
58 MM=[[None for _ in range(w+1)] for w in range(wires)]
59 for r in range(wires):
60 MM[r][r]=sum([(self[r][c] if r >= c else self[c][r]) for c in range(wires)])

62 return MaxwellMatrix(MM)

64class WElement(SParameters):
65 """s-parameters of of a W element is
66 calculated by approximating distributed parameters with a finite number
67 of sections specified."""
68 rtFraction=.01
69 def __init__(self,f, wires, R, Rse, df, Cm, Gm, Lm, Z0=50., K=0, scale=1.):
70 """Constructor
71 ports are 1,2,3,4, etc. on left and W, W+1, W+2, W+4, etc. on right where W=wires
72 and wire 1 is port 1 to P, wire 2 is port 2 to (P+1), etc. This is a P=2*w port device.
73 @param f list of float frequencies
74 @param wires integer number of wires
75 @param R list of float DC resistance of wire (ohms)
76 @param Rse list of float skin-effect resistance of wire (ohms/sqrt(Hz))
77 @param df list of list of float dissipation factor (loss-tangent) of capacitance (unitless)
78 @param Cm list of list of float mutual capacitance (self capacitance on diagonal) (F)
79 @param Gm list of list of float mutual conductance (self conductance on diagonal) (S)
80 @param Lm list of list of float mutual inductance (self inductance on diagonal) (H)
81 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
82 @param K (optional) float number of sections (defaults to 0)
83 @param scale (optional) float amount to scale the line by (assuming all other values are per unit)
84 @note If K=0 is specified, it is modified to a value that will provided a very good numerical
85 approximation.
86
87 The calculation is such that round-trip propagation time (twice the electrical length)
88 of any one small section is no more than one percent of the fastest possible risetime.
89 """
90 R=[r*scale for r in R]; Rse=[rse*scale for rse in Rse]; Cm=[[cm*scale for cm in cmr] for cmr in Cm];
91 Gm=[[gm*scale for gm in gmr] for gmr in Gm]; Lm=[[lm*scale for lm in lmr] for lmr in Lm]
92 K=K*scale
93 if K==0:
94 maxL=0.0; maxLm=0.0
95 for r in range(len(Lm)):
96 for c in range(len(Lm[r])):
97 if r==c: maxL=max(maxL,Lm[r][c])
98 else: maxLm=max(maxLm,Lm[r][c])
99 maxC=0.0; maxCm=0.0
100 for r in range(len(Cm)):
101 for c in range(len(Cm[r])):
102 if r==c: maxC=max(maxC,Cm[r][c])
103 else: maxCm=max(maxCm,Cm[r][c])
104 """max possible electrical length and fastest risetime"""
105 Td=math.sqrt((maxL+maxLm)*(maxC+2*maxCm)); Rt=0.45/f[-1]
106 """sections such that fraction of risetime less than round trip electrical
107 length of one section"""
108 K=Td*2/(Rt*self.rtFraction)
109 self.m_K=K

126 for o in range(w+1,wires):

128 sdp.AddLines(['device M'+suffix+' 4','device C'+suffix+' 2','device G'+suffix+' 2'])
129 sdp.AddLines(['port '+str(w+1)+' R_'+str(w+1)+' 1' for w in range(wires)])
130 sdp.AddLines(['port '+str(w+wires+1)+' C_'+str(w+1)+' 1' for w in range(wires)])

132 # connect all of the devices
133 sdp.AddLines(['connect R_'+str(w+1)+' 2 Rse_'+str(w+1)+' 1' for w in range(wires)])
134 sdp.AddLines(['connect Rse_'+str(w+1)+' 2 L_'+str(w+1)+' 1' for w in range(wires)])
135 for w in range(wires):
136 DeviceFromString='L_'+str(w+1)
137 DeviceFromPin=2
138 CapacitorConductanceConnectionString='C_'+str(w+1)+' 1 G_'+str(w+1)+' 1'
139 for o in range(wires):
140 if (o==w):
141 continue
142 if (o>w):
143 actualFrom=w
144 actualTo=o
145 actualInputPin=1
146 nextOutputPin=2
147 CapacitorConductanceConnectionString+=' C_'+str(w+1)+'_'+str(o+1)+' 1 G_'+str(w+1)+'_'+str(o+1)+' 1'
148 else:
149 actualFrom=o
150 actualTo=w
151 actualInputPin=3
152 nextOutputPin=4
153 CapacitorConductanceConnectionString+=' C_'+str(o+1)+'_'+str(w+1)+' 2 G_'+str(o+1)+'_'+str(w+1)+' 2'
154
155 DeviceToString='M_'+str(actualFrom+1)+'_'+str(actualTo+1)
156 sdp.AddLine('connect '+DeviceFromString+' '+str(DeviceFromPin)+' '+DeviceToString+' '+str(actualInputPin))
157 DeviceFromString=DeviceToString
158 DeviceFromPin=nextOutputPin
159 # finally, connect this last device output to the capacitances and conductances
160 sdp.AddLine('connect '+DeviceFromString+' '+str(DeviceFromPin)+' '+CapacitorConductanceConnectionString)
161 # make the system description and assign the s-parameters
162 self.m_sspn=SystemSParametersNumeric(sdp.SystemDescription())
163 self.NetListLines=sdp.m_lines
164 # frequency independent get assigned directly
165 for w in range(wires):
166 self.m_sspn.AssignSParameters('R_'+str(w+1),SeriesZ(R[w]/K,Z0))
167 self.m_sspn.AssignSParameters('G_'+str(w+1),TerminationG(Gm[w][w]/K,Z0))
168 for o in range(w+1,wires):
169 self.m_sspn.AssignSParameters('G_'+str(w+1)+'_'+str(o+1),SeriesG(Gm[o][w]/K,Z0)) # [o][w] not [w][o] to allow lower triangular
170 # now for frequency dependent ones
171 self.m_spdl=[]
172 for w in range(wires):
173 self.m_spdl.append(('Rse_'+str(w+1),dev.SeriesRse(f,Rse[w]/K,Z0)))
174 self.m_spdl.append(('C_'+str(w+1),dev.TerminationC(f,Cm[w][w]/K,Z0,df[w][w])))
175 self.m_spdl.append(('L_'+str(w+1),dev.SeriesL(f,Lm[w][w]/K,Z0)))
176 for o in range(w+1,wires):
177 self.m_spdl.append(('M_'+str(w+1)+'_'+str(o+1),dev.Mutual(f,Lm[o][w]/K,Z0))) # see note about [o][w] order above
178 self.m_spdl.append(('C_'+str(w+1)+'_'+str(o+1),dev.SeriesC(f,Cm[o][w]/K,Z0,df[o][w])))
179 SParameters.__init__(self,f,None,Z0)

187 """overloads [n]
188 @return list of list s-parameter matrix for the nth frequency element
189 """
190 for ds in self.m_spdl: self.m_sspn.AssignSParameters(ds[0],ds[1][n])
191 sp=self.m_sspn.SParameters()
192 if sp == 1: return sp
193 lp=[w+1 for w in range(len(sp)//2)]; rp=[w+len(sp)//2+1 for w in range(len(sp)//2)]
194 return T2S(linalg.fractional_matrix_power(S2T(sp,lp,rp),self.m_K),lp,rp)

196class WElementFile(WElement):
197 """W element file."""
198 def __init__(self,f,filename,df=0.,Z0=50., K=0, scale=1.):
199 """Constructor
200 @param f list of frequencies
201 @param filename string W element file.
202 @param df float (optional, defaults to 0.) dissipation factor to be added to W elements.
203 @param Z0 float (optional, defaults to 50.) reference impedance.
204 @param K int (optional, defaults to 0.) number of sections to be used for approximation.
205 @param scale float (optional, defaults to 1.) scale factor to scale result by.
206 @remark dissipation factor does not seem to be included in W elements, which is surprising, which
207 neccessitates the addition through an argument until I better understand this.
208 @remark If sections are specified as 0, the number of sections used for the approximation will be determined
209 automatically based on capacitances and inductances supplied, and the final end frequency for the s-parameters.
210 """
211 self.Z0=Z0
212 self.K=K
213 self.scale=scale
214 with open(filename,'rU' if sys.version_info.major < 3 else 'r') as fi:
215 lines=fi.readlines()
216 self.numbersList=[]
217 for line in lines:
218 if len(line)==0: continue
219 elif line[0]=='*': continue
220 elif len(line.strip(' '))==0: continue
221 tokens=line.split()
222 if len(tokens)>0:
223 self.numbersList.extend(tokens)
224 self.idx=0
225 self.wires=int(self._ReadToken())
226 self.L0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
227 CM=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
228 # CM is a Maxwell capacitance matrix. It needs to be converted to a mutual capacitance matrix
229 self.C0=MaxwellMatrix(CM).MutualMatrix()
230 self.R0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
231 self.G0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
232 self.Rs=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
233 self.Gd=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
234 self.df=[[df for _ in range(w+1)] for w in range(self.wires)]
235 WElement.__init__(self,f,self.wires,[self.R0[w][w] for w in range(self.wires)],[self.Rs[w][w] for w in range(self.wires)],
236 self.df,self.C0,self.G0,self.L0,self.Z0,self.K,self.scale)
237 def _ReadToken(self):
238 """Reads a single numeric token in string form from the W element file.
239 @return string numeric token.
240 """
241 value=self.numbersList[self.idx]
242 self.idx+=1
243 return value
