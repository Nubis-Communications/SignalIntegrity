29class MaxwellMatrix(list):
30 """A Maxwell matrix is a lower triangular matrix that is related to the mutual matrix."""
31 def __init__(self,MM):
32 """Constructor
33 @param MM lower triangular list of list Maxwell matrix
34 """
35 list.__init__(self,MM)
36 def MutualMatrix(self):
37 """Mutual matrix
38 @return a lower triangular mutual matrix
39 """
40 wires=len(self)
41 Mm=[[None for _ in range(w+1)] for w in range(wires)]
42 for r in range(wires):
43 Mm[r][r]=sum([self[r][c] if r >= c else self[c][r] for c in range(wires)])
44 for c in range(r): Mm[r][c]=-self[r][c]
45 return MutualMatrix(Mm)

55 """Maxwell matrix
56 @return a lower triangular Maxwell matrix
57 """
58 wires=len(self)
59 MM=[[None for _ in range(w+1)] for w in range(wires)]

65class WElement(SParameters):
66 """s-parameters of of a W element is
67 calculated by approximating distributed parameters with a finite number
68 of sections specified."""
69 rtFraction=.01
70 useLegacySolve=False
71 """bool whether to use the legacy interconnected-system solve for each section.
72
73 The default (False) computes each section's s-parameters directly from small
74 (wires x wires) series-impedance and shunt-admittance matrices, which is
75 numerically identical to the legacy solve but dramatically faster. Set to
76 True to fall back to the original interconnected s-parameter system solve."""
77 def __init__(self,f, wires, R, Rse, df, Cm, Gm, Lm, Z0=50., K=0, scale=1.):
78 """Constructor
79 ports are 1,2,3,4, etc. on left and W, W+1, W+2, W+4, etc. on right where W=wires
80 and wire 1 is port 1 to P, wire 2 is port 2 to (P+1), etc. This is a P=2*w port device.
81 @param f list of float frequencies
82 @param wires integer number of wires
83 @param R list of float DC resistance of wire (ohms)
84 @param Rse list of float skin-effect resistance of wire (ohms/sqrt(Hz))
85 @param df list of list of float dissipation factor (loss-tangent) of capacitance (unitless)
86 @param Cm list of list of float mutual capacitance (self capacitance on diagonal) (F)
87 @param Gm list of list of float mutual conductance (self conductance on diagonal) (S)
88 @param Lm list of list of float mutual inductance (self inductance on diagonal) (H)
89 @param Z0 (optional) float reference impedance (defaults to 50 ohms)
90 @param K (optional) float number of sections (defaults to 0)
91 @param scale (optional) float amount to scale the line by (assuming all other values are per unit)
92 @note If K=0 is specified, it is modified to a value that will provided a very good numerical
93 approximation.
94
95 The calculation is such that round-trip propagation time (twice the electrical length)
96 of any one small section is no more than one percent of the fastest possible risetime.
97 """
98 R=[r*scale for r in R]; Rse=[rse*scale for rse in Rse]; Cm=[[cm*scale for cm in cmr] for cmr in Cm];
99 Gm=[[gm*scale for gm in gmr] for gmr in Gm]; Lm=[[lm*scale for lm in lmr] for lmr in Lm]
100 K=K*scale
101 if K==0:
102 maxL=0.0; maxLm=0.0
103 for r in range(len(Lm)):
104 for c in range(len(Lm[r])):
105 if r==c: maxL=max(maxL,Lm[r][c])
106 else: maxLm=max(maxLm,Lm[r][c])
107 maxC=0.0; maxCm=0.0
108 for r in range(len(Cm)):
109 for c in range(len(Cm[r])):
110 if r==c: maxC=max(maxC,Cm[r][c])
111 else: maxCm=max(maxCm,Cm[r][c])
112 """max possible electrical length and fastest risetime"""
113 Td=math.sqrt((maxL+maxLm)*(maxC+2*maxCm)); Rt=0.45/f[-1]
114 """sections such that fraction of risetime less than round trip electrical
115 length of one section"""
116 K=Td*2/(Rt*self.rtFraction)
117 self.m_K=K
118 # build the netlist
119 # pragma: silent exclude

121 from SignalIntegrity.Lib.Devices import SeriesZ
122 from SignalIntegrity.Lib.Devices import TerminationG
123 import SignalIntegrity.Lib.SParameters.Devices as dev

128 sdp=SystemDescriptionParser().AddLines(['device R_'+str(w+1)+' 2' for w in range(wires)])
129 sdp.AddLines(['device Rse_'+str(w+1)+' 2' for w in range(wires)])
130 sdp.AddLines(['device L_'+str(w+1)+' 2' for w in range(wires)])
131 sdp.AddLines(['device G_'+str(w+1)+' 1' for w in range(wires)])
132 sdp.AddLines(['device C_'+str(w+1)+' 1' for w in range(wires)])
133 for w in range(wires-1):
134 for o in range(w+1,wires):
135 suffix='_'+str(w+1)+'_'+str(o+1)
136 sdp.AddLines(['device M'+suffix+' 4','device C'+suffix+' 2','device G'+suffix+' 2'])
137 sdp.AddLines(['port '+str(w+1)+' R_'+str(w+1)+' 1' for w in range(wires)])
138 sdp.AddLines(['port '+str(w+wires+1)+' C_'+str(w+1)+' 1' for w in range(wires)])
139
140 # connect all of the devices
141 sdp.AddLines(['connect R_'+str(w+1)+' 2 Rse_'+str(w+1)+' 1' for w in range(wires)])
142 sdp.AddLines(['connect Rse_'+str(w+1)+' 2 L_'+str(w+1)+' 1' for w in range(wires)])
143 for w in range(wires):
144 DeviceFromString='L_'+str(w+1)
145 DeviceFromPin=2
146 CapacitorConductanceConnectionString='C_'+str(w+1)+' 1 G_'+str(w+1)+' 1'
147 for o in range(wires):
148 if (o==w):
149 continue
150 if (o>w):
151 actualFrom=w
152 actualTo=o
153 actualInputPin=1
154 nextOutputPin=2
155 CapacitorConductanceConnectionString+=' C_'+str(w+1)+'_'+str(o+1)+' 1 G_'+str(w+1)+'_'+str(o+1)+' 1'
156 else:
157 actualFrom=o
158 actualTo=w
159 actualInputPin=3
160 nextOutputPin=4
161 CapacitorConductanceConnectionString+=' C_'+str(o+1)+'_'+str(w+1)+' 2 G_'+str(o+1)+'_'+str(w+1)+' 2'
162
163 DeviceToString='M_'+str(actualFrom+1)+'_'+str(actualTo+1)
164 sdp.AddLine('connect '+DeviceFromString+' '+str(DeviceFromPin)+' '+DeviceToString+' '+str(actualInputPin))
165 DeviceFromString=DeviceToString
166 DeviceFromPin=nextOutputPin
167 # finally, connect this last device output to the capacitances and conductances
168 sdp.AddLine('connect '+DeviceFromString+' '+str(DeviceFromPin)+' '+CapacitorConductanceConnectionString)
169 # make the system description and assign the s-parameters
170 self.m_sspn=SystemSParametersNumeric(sdp.SystemDescription())
171 self.NetListLines=sdp.m_lines
172 # frequency independent get assigned directly
173 for w in range(wires):
174 self.m_sspn.AssignSParameters('R_'+str(w+1),SeriesZ(R[w]/K,Z0))
175 self.m_sspn.AssignSParameters('G_'+str(w+1),TerminationG(Gm[w][w]/K,Z0))
176 for o in range(w+1,wires):
177 self.m_sspn.AssignSParameters('G_'+str(w+1)+'_'+str(o+1),SeriesG(Gm[o][w]/K,Z0)) # [o][w] not [w][o] to allow lower triangular
178 # now for frequency dependent ones
179 self.m_spdl=[]
180 for w in range(wires):
181 self.m_spdl.append(('Rse_'+str(w+1),dev.SeriesRse(f,Rse[w]/K,Z0)))
182 self.m_spdl.append(('C_'+str(w+1),dev.TerminationC(f,Cm[w][w]/K,Z0,df[w][w])))
183 self.m_spdl.append(('L_'+str(w+1),dev.SeriesL(f,Lm[w][w]/K,Z0)))
184 for o in range(w+1,wires):
185 self.m_spdl.append(('M_'+str(w+1)+'_'+str(o+1),dev.Mutual(f,Lm[o][w]/K,Z0))) # see note about [o][w] order above
186 self.m_spdl.append(('C_'+str(w+1)+'_'+str(o+1),dev.SeriesC(f,Cm[o][w]/K,Z0,df[o][w])))
187 # frequency-independent per-section matrices for the fast analytic solve.
188 # the lower-triangular [o][w] (o>=w) storage of the mutual/inductance/
189 # capacitance/conductance matrices is expanded to full symmetric matrices.
190 self.m_wires=wires
191 self.m_Rsec=[R[w]/K for w in range(wires)]
192 self.m_Rsesec=[Rse[w]/K for w in range(wires)]
193 self.m_Lsec=[[Lm[max(a,b)][min(a,b)]/K for b in range(wires)] for a in range(wires)]
194 self.m_Csec=[[Cm[max(a,b)][min(a,b)]/K for b in range(wires)] for a in range(wires)]
195 self.m_Gsec=[[Gm[max(a,b)][min(a,b)]/K for b in range(wires)] for a in range(wires)]
196 self.m_dfsec=[[df[max(a,b)][min(a,b)] for b in range(wires)] for a in range(wires)]
197 SParameters.__init__(self,f,None,Z0)

199 """NetList
200 @return the netlist formed for a W element.
201 @remark this is really for testing only.
202 """
203 return self.NetListLines
204 def SectionSParameters(self,n):
205 """s-parameters of a single section (1/K of the line) at frequency index n.
206
207 Computed directly from the section series-impedance matrix (series R, skin-
208 effect resistance and the full mutual inductance matrix) and the shunt nodal
209 admittance matrix (self/mutual capacitance - including loss tangent - and
210 self/mutual conductance). This is numerically identical to interconnecting
211 the individual lumped-element s-parameter devices, but avoids the large
212 per-frequency matrix solve.
213 @param n integer frequency index
214 @return list of list s-parameter matrix of the section (2*wires ports)
215 """
216 wires=self.m_wires; Z0=self.m_Z0
217 w=2.0*math.pi*self.m_f[n]; sqrtf=math.sqrt(self.m_f[n])
218 Lsec=self.m_Lsec; Csec=self.m_Csec; Gsec=self.m_Gsec; dfsec=self.m_dfsec
219 # series impedance matrix (coupled inductors in series with R and Rse)
220 Zs=np.array([[(self.m_Rsec[a]+self.m_Rsesec[a]*(1+1j)*sqrtf+1j*w*Lsec[a][a])
221 if a==b else (1j*w*Lsec[a][b])
222 for b in range(wires)] for a in range(wires)],dtype=complex)
223 # shunt nodal admittance matrix (caps/conductances to ground and between nodes)
224 Yp=np.zeros((wires,wires),dtype=complex)
225 for a in range(wires):
226 acc=w*Csec[a][a]*(1j+dfsec[a][a])+Gsec[a][a]
227 for b in range(wires):
228 if b==a: continue
229 ymut=w*Csec[a][b]*(1j+dfsec[a][b])+Gsec[a][b]
230 Yp[a][b]=-ymut
231 acc+=ymut
232 Yp[a][a]=acc
233 Ys=np.linalg.inv(Zs)
234 N=wires
235 Y=np.zeros((2*N,2*N),dtype=complex)
236 Y[:N,:N]=Ys; Y[:N,N:]=-Ys; Y[N:,:N]=-Ys; Y[N:,N:]=Ys+Yp
237 I=np.identity(2*N)
238 return ((I-Z0*Y).dot(np.linalg.inv(I+Z0*Y))).tolist()
239 def __getitem__(self,n):
240 """overloads [n]
241 @return list of list s-parameter matrix for the nth frequency element
242 """
243 if self.useLegacySolve:
244 for ds in self.m_spdl: self.m_sspn.AssignSParameters(ds[0],ds[1][n])
245 sp=self.m_sspn.SParameters()
246 else:
247 sp=self.SectionSParameters(n)
248 if sp == 1: return sp
249 lp=[w+1 for w in range(len(sp)//2)]; rp=[w+len(sp)//2+1 for w in range(len(sp)//2)]
250 return T2S(linalg.fractional_matrix_power(S2T(sp,lp,rp),self.m_K),lp,rp)

252class WElementFile(WElement):
253 """W element file."""
254 def __init__(self,f,filename,df=0.,Z0=50., K=0, scale=1.):
255 """Constructor
256 @param f list of frequencies
257 @param filename string W element file.
258 @param df float (optional, defaults to 0.) dissipation factor to be added to W elements.
259 @param Z0 float (optional, defaults to 50.) reference impedance.
260 @param K int (optional, defaults to 0.) number of sections to be used for approximation.
261 @param scale float (optional, defaults to 1.) scale factor to scale result by.
262 @remark dissipation factor does not seem to be included in W elements, which is surprising, which
263 neccessitates the addition through an argument until I better understand this.
264 @remark If sections are specified as 0, the number of sections used for the approximation will be determined
265 automatically based on capacitances and inductances supplied, and the final end frequency for the s-parameters.
266 """
267 self.Z0=Z0
268 self.K=K
269 self.scale=scale
270 with open(filename,'rU' if sys.version_info.major < 3 else 'r') as fi:
271 lines=fi.readlines()
272 self.numbersList=[]
273 for line in lines:
274 if len(line)==0: continue
275 elif line[0]=='*': continue
276 elif len(line.strip(' '))==0: continue
277 tokens=line.split()
278 if len(tokens)>0:
279 self.numbersList.extend(tokens)
280 self.idx=0
281 self.wires=int(self._ReadToken())
282 self.L0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
283 CM=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
284 # CM is a Maxwell capacitance matrix. It needs to be converted to a mutual capacitance matrix
285 self.C0=MaxwellMatrix(CM).MutualMatrix()
286 self.R0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
287 self.G0=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
288 self.Rs=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
289 self.Gd=[[float(self._ReadToken()) for _ in range(w+1)] for w in range(self.wires)]
290 self.df=[[df for _ in range(w+1)] for w in range(self.wires)]
291 WElement.__init__(self,f,self.wires,[self.R0[w][w] for w in range(self.wires)],[self.Rs[w][w] for w in range(self.wires)],
292 self.df,self.C0,self.G0,self.L0,self.Z0,self.K,self.scale)
