25class TimeDescriptor(object):
26 """time-axis for waveforms"""
27 def __init__(self,HorOffset,NumPts,SampleRate):
28 """Constructor
29 @param HorOffset float time of first point
30 @param NumPts integer number of points
31 @param SampleRate float sample rate (1/sample period)
32 """
33 self.H = HorOffset
34 self.K = int(NumPts)
35 self.Fs=SampleRate
36 def Duration(self):
37 """waveform duration
38 @return duration of the waveform
39 """
40 return self.K/self.Fs
41 def __len__(self):
42 """overloads len()
43 @return number of time points
44 """
45 return self.K
46 def __getitem__(self,item):
47 """overloads [item]
48 @param item integer or float index
49 @return float time at the index item specified.
50 """
51 return item/self.Fs+self.H
52 def __eq__(self,other):
53 """overloads ==
54 @param other instance of class TimeDescriptor
55 @return boolean whether the other descriptor is the same.
56 @note the sample rate is compared with an epsilon of 1e-15.
57 @note the horizontal offset is compared with an epsilon of 0.001% of the sample period.
58 """
59 if abs(self.Fs - other.Fs) > 1e-15: return False
60 if abs(self.H - other.H) > .00001*1./self.Fs: return False
61 if self.K != other.K: return False
62 return True
63 def __ne__(self,other):
64 """overloads !=
65 @param other instance of class TimeDescriptor
66 @return boolean whether other descriptor is not the same.
67 @see TimeDescriptor.__eq__
68 """
69 return not self == other
70 def Times(self,unit=None):
71 """returns list of times
72 @param unit (optional) string of units for time values.
73 @return list of times corresponding to self.
74 @note valid time units are:
75 - None - return simply the list of time values.
76 - 'ps' - return the time values divided by 1e-12.
77 - 'ns' - return the time values divided by 1e-9.
78 - 'us' - return the time values divided by 1e-6.
79 - 'ms' - return the time values divided by 1e-3.
80 """
81 if unit==None:
82 return [self[k] for k in range(len(self))]
83 elif isinstance(unit,float):
84 return [self[k]/unit for k in range(len(self))]
85 elif unit=='ps':
86 return [self[k]/1.e-12 for k in range(len(self))]
87 elif unit=='ns':
88 return [self[k]/1.e-9 for k in range(len(self))]
89 elif unit =='us':
90 return [self[k]/1.e-6 for k in range(len(self))]
91 elif unit == 'ms':
92 return [self[k]/1.e-3 for k in range(len(self))]
93 def ApplyFilter(self,F):
94 """calculates effect of filter
95 calculates the new time descriptor of a waveform filtered by a filter
96 with the filter descriptor provided.
97 @param F instance of class FilterDescriptor
98 @return instance of class TimeDescriptor containing the time descriptor
99 affected by the filter.
100 @note does not affect self.
101 """
102 return TimeDescriptor(
103 HorOffset=self.H+(F.S-F.D)/self.Fs,
104 NumPts=int(max(0,(self.K-F.S)*F.U)),
105 SampleRate=self.Fs*F.U)
106 def __mul__(self,F):
107 """overloads *
108 This is an abstraction.
109 calculates the new time descriptor of a waveform filtered by a filter
110 with the filter descriptor provided.
111 @param F instance of class FilterDescriptor
112 @return instance of class TimeDescriptor containing the time descriptor
113 affected by the filter.
114 @note does not affect self.
115 @see ApplyFilter()
116 """
117 return self.ApplyFilter(F)
118 def __div__(self,other):
119 return self.__truediv__(other)
120 def __truediv__(self,other):
121 """overloads /
122 This is an abstraction and is polymorphic.
123 if other is a filter descriptor, it assumes self is an outputwf such that
124 inputwf*other=outputwf and it calculates the inputwf as outputwf*other^-1.
125 This is the same as finding the input waveform that if filtered by other would
126 produce self.
127 if other is an input wf time descriptor, it assume that self is an output wf such that
128 inputwf*filter=outputwf and it calculates the filter as inputwf^-1*outputwf.
129 This is the same as finding the filter that, when applied to other, would produce self.
130 @param other instance of class TimeDescriptor or FilterDescriptor.
131 @return instance of class TimeDescriptor or FilterDescriptor depending on the situations
132 explained.
133 """
134 if isinstance(other,FilterDescriptor):
135 return TimeDescriptor(
136 HorOffset=self.H+other.U/self.Fs*(other.D-other.S),
137 NumPts=self.K/other.U+other.S,
138 SampleRate=self.Fs/other.U)
139 elif isinstance(other,TimeDescriptor):
140 UpsampleFactor=self.Fs/other.Fs
141 return FilterDescriptor(
142 UpsampleFactor,
143 DelaySamples=other.K-self.K/UpsampleFactor-
144 (self.H-other.H)*other.Fs,
145 StartupSamples=other.K-self.K/UpsampleFactor)
146 def DelayBy(self,D):
147 """delays by time
148 @param D float delay time
149 @return instance of class TimeDescriptor containing self delayed by the time.
150 @note does not affect self.
151 """
152 return TimeDescriptor(self.H+D,self.K,self.Fs)
153 def FrequencyList(self):
154 """corresponding frequency list
155 calculates a list of frequencies for a frequency domain waveform that would correspond
156 to the time descriptor self.
157 @return instance of EvenlySpacedFrequencyList containing list of frequencies that
158 correspond to this time descriptor.
159 """
160 K=int(self.K)
161 N=K//2
162 Fe=float(self.Fs)*N/K
163 return EvenlySpacedFrequencyList(Fe,N)
164 def Intersection(self,other):
165 """intersection of two waveforms
166 The intersection is the portion of two time-axes that overlap
167 @param other instance of class TimeDescriptor
168 @return the intersection of the two waveform descriptors
169 """
170 return TimeDescriptor(
171 HorOffset=max(self.H,other.H),
172 NumPts=max(0,min(self.TimeOfPoint(self.K),
173 other.TimeOfPoint(other.K))-max(self.H,other.H))*self.Fs,
174 SampleRate=self.Fs)
175 def TimeOfPoint(self,k):
176 """time corresponding to point index
177 @param k integer (or float) index of time point
178 @return float time of that point
179 @note the index can be fractional
180 @note the index does not have to correspond to a point inside the list of times.
181 """
182 return self.H+float(k)/self.Fs
183 def IndexOfTime(self,t,Integer=True):
184 """point index corresponding to time
185 @param t float time of point
186 @param Integer (optional, defaults to True) whether to return nearest integer.
187 Otherwise, returns float with possible fractional index
188 @return index of point corresponding to time
189 @note the index may not correspond to a point inside the list of times.
190 """
191 index=(t-self.H)*self.Fs
192 if Integer: index=int(index)
193 return index

195 """prints an ascii version of the time descriptor"""
196 print('HorOffset: '+str(self.H))

198 print('SampleRate: '+str(self.Fs))
