42 def SetEvenlySpaced(self,Fe,N):
43 """sets evenly spaced
44 @param Fe float end frequency for the frequency list
45 @param N integer number of points (-1) or the frequency list (i.e. the number of points
46 in the new frequency list will be N+1.
47 @return self
48 @remark
49 Initializes the frequency list to be evenly spaced with N+1 points from n=0..N where each
50 frequency is f[n]=n/N*Fe

80 """Frequencies
81 @param unit optional string containing unit to use
82 @return list of frequencies in the frequency list in the unit specified
83 @remark Valid units are:
84 - GHz - each frequency element is divided by 1e9.
85 - MHz - each frequency element is divided by 1e6.
86 - kHz - each frequency element is divided by 1e3
87
88 If no unit is supplied or if it's None, then the frequencies are provided as is.
89
90 if the unit supplied is otherwise invalid, None is returned.
91 """
92 if unit == None: return list(self)
93 elif isinstance(unit,float): return (self/unit).Frequencies()
94 elif unit == 'GHz': return (self/1.e9).Frequencies()
95 elif unit == 'MHz': return (self/1.e6).Frequencies()
96 elif unit == 'kHz': return (self/1.e3).Frequencies()
97 def CheckEvenlySpaced(self,epsilon=0.01):
98 """checks and sets whether evenly spaced
99 @param epsilon (optional) float difference tolerated in whether a frequency compared is the same.
100 (Defaults to 0.01).
101 @return boolean whether the frequency list is evenly spaced.
102 @note if the m_EvenlySpaced internal variable indicates it's evenly spaced, it simply returns True.
103
104 Checks the frequencies to see if they conform to N+1 frequencies with for n=0..N, each frequency
105 in the list is f[n]=n/N*Fe within the epsilon.
106 """
107 if self.m_EvenlySpaced: return True
108 for n in range(self.N+1):
109 try:
110 if abs(self[n]-self.Fe/self.N*n) > epsilon:
111 self.m_EvenlySpaced=False
112 return False
113 except:
114 return False
115 self.SetEvenlySpaced(self.Fe,self.N)
116 return True

213class LogarithmicallySpacedFrequencyList(FrequencyList):
214 """a logarithmically spaced frequency list"""
215 def __init__(self,start_frequency,end_frequency,points_per_decade):
216 """Constructor
217 @param start_frequency float start frequency
218 @param end_frequency float end frequency
219 @param points_per_decade integer number of points per decade
220 Intitializes the frequency list with the list of frequencies
221
222 The list of frequencies is constructed by first determining the list of frequencies in a decade based
223 on the number of points per decade specified. Then, for each decade, all of the points are retained
224 that are above the start frequency and below the end frequency. Finally, the start and end frequencies
225 are tacked onto the list.
226 """
227 import math
228 decade_list = [10.**(k/points_per_decade) for k in range(points_per_decade+1)]
229 frequency_list=[start_frequency]
230 for d in range(math.floor(math.log10(start_frequency)),math.ceil(math.log10(end_frequency))):
231 for k in range(points_per_decade+1):
232 this_frequency = decade_list[k] * math.pow(10.,d)
233 if (this_frequency > frequency_list[-1]) and (this_frequency < end_frequency):
234 frequency_list.append(this_frequency)
235 if end_frequency > frequency_list[-1]:
236 frequency_list.append(end_frequency)
237 FrequencyList.__init__(self,frequency_list)
